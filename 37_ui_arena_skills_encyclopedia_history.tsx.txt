'bg-teal-600 text-white shadow-sm' : 'text-[var(--text-muted)] hover:text-[var(--text-main)] bg-[var(--bg-body)]'}`}>Tất Cả ({historyLogs.length})</button>

'bg-blue-600 text-white shadow-sm' : 'text-[var(--text-muted)] hover:text-blue-400 bg-[var(--bg-body)]'}`}>📖 Cốt Truyện ({historyLogs.filter(l => l.type === 'story').length})</button>

'bg-emerald-600 text-white shadow-sm' : 'text-[var(--text-muted)] hover:text-emerald-400 bg-[var(--bg-body)]'}`}>⚡ Hành Động ({historyLogs.filter(l => l.type === 'action').length})</button>

'bg-orange-600 text-white shadow-sm' : 'text-[var(--text-muted)] hover:text-orange-400 bg-[var(--bg-body)]'}`}>🔔 Hệ Thống ({historyLogs.filter(l => l.type === 'system').length})</button>

'bg-orange-900/10 border-orange-900/30 items-center' : 'bg-[var(--bg-surface)] border-[var(--border-main)] items-start'}`}>

                            {log.type === 'system' ? (
                                <span className="text-xs font-mono font-bold text-orange-400 text-center" style={{ fontSize: `${Math.max(10, historyFontSize - 4)}px` }}>{String(log.text)}</span>
