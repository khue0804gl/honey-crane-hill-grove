  const handlePvpRealtimeAction = async (actionText) => {
      if (isPvpResolving || !currentCoopRoom?.pvpMatch) return;
      setIsPvpResolving(true); setPvpInput('');
      
      const match = currentCoopRoom.pvpMatch;
      const roomRef = doc(db, 'artifacts', appId, 'public', 'data', 'coop_rooms', currentCoopRoom.id);
      
      // Update log to show intent immediately
      const myName = match.p1.uid === user?.uid ? match.p1.name : match.p2.name;
      const intentLog = { type: 'action', text: actionText, uid: user?.uid, name: myName };
      await setDoc(roomRef, { pvpMatch: { ...match, logs: [...match.logs, intentLog], lastTurnTime: Date.now() } }, { merge: true });

      const isP1 = match.p1.uid === user?.uid;
      const myStats = isP1 ? match.p1 : match.p2;
      const oppStats = isP1 ? match.p2 : match.p1;

      let pvpKamuiR = "";
      if (myStats.traits.includes('Thuấn Di Bản Năng')) pvpKamuiR += "\n- [BÊN A có Thuấn Di Bản Năng V12.20]: Miễn toàn bộ ST Vật lý. Ra đòn TẤT TRÚNG nếu Thức A >= 1.2 Tốc B. Chịu thêm 40% ST Hỏa/Nổ/Độc.";
      if (oppStats.traits.includes('Thuấn Di Bản Năng')) pvpKamuiR += "\n- [BÊN B có Thuấn Di Bản Năng V12.20]: Miễn toàn bộ ST Vật lý. Ra đòn TẤT TRÚNG nếu Thức B >= 1.2 Tốc A. Chịu thêm 40% ST Hỏa/Nổ/Độc.";

      const sysPrompt = `BẠN LÀ THIÊN ĐẠO TRỌNG TÀI PVP THỜI GIAN THỰC.
[BÊN A (ĐANG TỚI LƯỢT)]: ${myStats.name} (Cảnh giới: ${myStats.level}) | HP: ${myStats.hp}/${myStats.maxHp} | Năng lượng (${myStats.energyName}): ${myStats.mp}/${myStats.maxMp}
Chỉ số: Công ${myStats.stats.atk}, Thủ ${myStats.stats.def}, Tốc ${myStats.stats.spd}, Thức ${myStats.stats.spp}
Vũ Khí/Skill: ${myStats.skills} | Item: ${myStats.items} | Khí vận: ${myStats.traits}

[BÊN B (ĐỐI THỦ)]: ${oppStats.name} (Cảnh giới: ${oppStats.level}) | HP: ${oppStats.hp}/${oppStats.maxHp} | Năng lượng (${oppStats.energyName}): ${oppStats.mp}/${oppStats.maxMp}
Chỉ số: Công ${oppStats.stats.atk}, Thủ ${oppStats.stats.def}, Tốc ${oppStats.stats.spd}, Thức ${oppStats.stats.spp}
Vũ Khí/Skill: ${oppStats.skills} | Khí vận: ${oppStats.traits}

Lịch sử giao tranh: ${match.logs.slice(-5).map(l => l.text).join(' | ')}
Bên A tung chiêu: "${actionText}"

NHIỆM VỤ (TRẢ VỀ JSON):
1. Tính toán sát thương chân thực dựa trên Công vs Thủ, Tốc vs Tốc (né tránh), Thức vs Thức (phản xạ/đọc vị).
2. Trừ MP/HP của Bên A nếu chiêu thức hao tổn.
3. Trừ HP của Bên B nếu trúng đòn.
4. "log": 1 đoạn văn tiên hiệp gay cấn mô tả kết quả đòn đánh này.${pvpKamuiR}
JSON: {"log": "...", "pA_HpChange": 0, "pA_MpChange": -50, "pB_HpChange": -1200}`;

      const payload = { contents: [{ parts: [{ text: "Xử lý đòn đánh!" }] }], systemInstruction: { parts: [{ text: sysPrompt }] }, generationConfig: { responseMimeType: "application/json" } };
              
              try {
                  const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
                  const result = await response.json();
          let parts = result.candidates?.[0]?.content?.parts || [];
          let textResult = parts.filter(p => !p.thought).map(p => p.text).join('\n') || parts[0]?.text || '{}';
          
          // Trích xuất JSON thông minh (Cân bằng ngoặc)
          const cbMatch = textResult.match(/```(?:json)?\s*(\{[\s\S]*?\})\s*```/i);
          if (cbMatch) {
              textResult = cbMatch[1];
          } else {
              const first = textResult.indexOf('{');
              if (first !== -1) {
                  let count = 0;
                  for (let i = first; i < textResult.length; i++) {
                      if (textResult[i] === '{') count++;
                      else if (textResult[i] === '}') {
                          count--;
                          if (count === 0) {
                              textResult = textResult.substring(first, i + 1);
                              break;
                          }
                      }
                  }
              }
          }
          // V8.06 FIX: Cạo sạch chữ Hán ở Võ Đài Giả Tưởng
          textResult = textResult.replace(/[\u3040-\u30ff\u3400-\u4dbf\u4e00-\u9fff\uf900-\ufaff\uff66-\uff9f]/g, '');
          const data = JSON.parse(filterRestrictedNames(textResult.replace(/```json/gi, '').replace(/```/g, '').trim()));

          let newP1 = { ...match.p1 }; let newP2 = { ...match.p2 };
          if (isP1) {
              newP1.hp = Math.max(0, newP1.hp + (Number(data.pA_HpChange)||0)); newP1.mp = Math.max(0, newP1.mp + (Number(data.pA_MpChange)||0));
              newP2.hp = Math.max(0, newP2.hp + (Number(data.pB_HpChange)||0));
          } else {
              newP2.hp = Math.max(0, newP2.hp + (Number(data.pA_HpChange)||0)); newP2.mp = Math.max(0, newP2.mp + (Number(data.pA_MpChange)||0));
              newP1.hp = Math.max(0, newP1.hp + (Number(data.pB_HpChange)||0));
          }

          let newLogs = [...match.logs, intentLog, { type: 'story', text: data.log }];
          let newStatus = 'playing';
          let nextTurn = isP1 ? match.p2.uid : match.p1.uid;

          if (newP1.hp <= 0 || newP2.hp <= 0) {
              newStatus = 'finished';
              nextTurn = null;
              newLogs.push({ type: 'system', text: `💀 TRẬN CHIẾN KẾT THÚC! Kẻ thắng cuộc: ${newP1.hp > 0 ? newP1.name : newP2.hp > 0 ? newP2.name : 'Đồng quy vu tận'}. Ảo cảnh vỡ nát, thần hồn trở về nhục thân.` });
          }

          await setDoc(roomRef, { pvpMatch: { p1: newP1, p2: newP2, logs: newLogs, currentTurnUid: nextTurn, status: newStatus, lastTurnTime: Date.now() } }, { merge: true });
          setTimeout(() => { if(pvpLogScrollRef.current) pvpLogScrollRef.current.scrollTop = pvpLogScrollRef.current.scrollHeight; }, 200);
      } catch(e) { console.error(e); }
      setIsPvpResolving(false);
  };
