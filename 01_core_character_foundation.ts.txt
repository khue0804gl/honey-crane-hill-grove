const pruneStatHistories = (state: any, maxKeep = 10) => {
    if (!state || typeof state !== 'object') return state;
    for (const key of STAT_HISTORY_KEYS) {
        if (Array.isArray(state[key]) && state[key].length > maxKeep) {
            state[key] = state[key].slice(0, maxKeep);
        }
    }
    // V2.99: Khóa trần tóm tắt cốt truyện 20 chương gần nhất theo lệnh Chủ toạ Dark
    if (Array.isArray(state.summaries) && state.summaries.length > 20) {
        state.summaries = state.summaries.slice(0, 20);
    }
    return state;
};

const capActiveEffects = (effects: any[], maxCount = 12): any[] => {
    if (!Array.isArray(effects)) return [];
    if (effects.length <= maxCount) return effects.filter(Boolean);
    
    const validEffects = effects.filter(Boolean);
    const priority1: any[] = [];
    const debuffs: any[] = [];
    const normalBuffs: any[] = [];

    validEffects.forEach(e => {
        if (e.type === 'core_method' || (e.durationMinutes || 0) > 50000000 || e.id === 'eff_chan_than_boc_phat' || e.id === 'eff_nhan_the_han_che') {
            priority1.push(e);
        } else if (e.type === 'debuff') {
            debuffs.push(e);
        } else {
            normalBuffs.push(e);
        }
    });

    const result: any[] = [...priority1];
    for (const d of debuffs) {
        if (result.length < maxCount) result.push(d);
    }
    normalBuffs.sort((a, b) => (b.durationMinutes || 0) - (a.durationMinutes || 0));
    for (const b of normalBuffs) {
        if (result.length < maxCount) result.push(b);
    }
    return result;
};

// V2.77 FIX (BÁO LỖI HOÀNG ĐẠT): Nhận diện thực thể Yêu Tộc / Thần Thú toàn diện (Background, Chức nghiệp, Cảnh giới, Khí vận)
const isYeuTocEntity = (stateOrBg: any, path?: string, level?: string, traits?: any[]) => {
    let bgId = '';
    let cPath = '';
    let cLevel = '';
    let tList: any[] = [];
    if (stateOrBg && typeof stateOrBg === 'object') {
        bgId = String(stateOrBg.backgroundId || stateOrBg.id || '');
        cPath = String(stateOrBg.cultivationPath || '');
        cLevel = String(stateOrBg.cultivationLevel || '');
        tList = Array.isArray(stateOrBg.traits) ? stateOrBg.traits : [];
    } else {
        bgId = String(stateOrBg || '');
        cPath = String(path || '');
        cLevel = String(level || '');
        tList = Array.isArray(traits) ? traits : [];
    }
    const tNames = tList.map((t: any) => (typeof t === 'string' ? t : String(t?.name || '')));
    const isYeuBg = isPureYeuTocBg(bgId) || ['bg_10', 'bg_25', 'bg_taotie', 'bg_long_toc', 'bg_con_bang', 'bg_ho_toc', 'bg_vien_co_cu_long'].includes(bgId);
    const isYeuPath = cPath === 'yeu_tu';
    const isYeuLevel = /(thông trí|khai quang|yêu đan|hóa hình|yêu tướng|yêu vương|yêu hoàng|yêu đế|thái cổ yêu tổ)/i.test(cLevel);
    const isYeuTrait = tNames.some((n: string) => /tiên thiên đại yêu|thái cổ ngự yêu sư|chân long huyết mạch|nhục thân cự long|viễn cổ cự long|yêu tộc/i.test(n));
    return isYeuBg || isYeuPath || isYeuLevel || isYeuTrait;
};

const getTurnYeuHumanMult = (stateOrBg: any, path?: string, level?: string, traits?: any[]): number => {
    const isY = isYeuTocEntity(stateOrBg, path, level, traits);
    let bgId = '';
    let isHuman: boolean | undefined = undefined;
    if (stateOrBg && typeof stateOrBg === 'object') {
        bgId = String(stateOrBg.backgroundId || stateOrBg.id || '');
        isHuman = stateOrBg.isHumanForm;
    } else {
        bgId = String(stateOrBg || '');
    }
    const isBeast = isPureYeuTocBg(bgId);
    const isHumanActive = isY && (
        isHuman === true ||
        (isHuman === undefined && !isBeast)
    );
    return isHumanActive ? 0.5 : 1.0;
};

const parseCharacterHeight = (rawHeight: any, gender: string = 'Nam'): { meters: number, display: string } => {
    let str = String(rawHeight || '').trim();
    if (!str) {
        const defM = gender === 'Nữ' ? 1.65 : 1.78;
        return { meters: defM, display: `${defM.toFixed(2)} m` };
    }
    let m = 0;
    const mMatch = str.match(/^([0-2])\s*[mM]\s*(\d{1,2})$/);
    if (mMatch) {
        m = parseFloat(mMatch[1]) + parseFloat(mMatch[2]) / 100;
    } else {
        const numMatch = str.match(/([0-9]+(?:\.[0-9]+)?)/);
        if (numMatch) {
            let val = parseFloat(numMatch[1]);
            if (val >= 100 && val <= 300) {
                m = val / 100;
            } else if (val >= 0.5 && val <= 3.5) {
                m = val;
            }
        }
    }
    if (!m || isNaN(m)) {
        m = gender === 'Nữ' ? 1.65 : 1.78;
    }
    m = Math.min(2.5, Math.max(1.2, Math.round(m * 100) / 100));
    return { meters: m, display: `${m.toFixed(2)} m` };
};

                              tNames.includes('Bán Ma (Huyết Mạch Lai)') || 
                              tNames.includes('Hợp Hoan Mị Cốt') || 
                              tNames.includes('Thiên Ma Thể') || 
                              tNames.includes('Ma Tôn Chuyển Thế') || 
                              path === 'ma_tu';

    if (isMaTocOrSuccubus) {
        const isBanMa = tNames.includes('Bán Mị Ma (Huyết Mạch Lai)') || tNames.includes('Bán Ma (Huyết Mạch Lai)');
        return {
            category: 'ma_toc',
            categoryLabel: 'Ma Giới Chân Ma (Mị Ma / Cực Đạo Ma Thể / Hư Không Ma Tộc)',
            name: (tNames.includes('Bán Mị Ma (Huyết Mạch Lai)') || bgId === 'bg_succubus') ? 'Mị Ma Chân Thân' : 'Ma Thần Bản Nguyên Chân Thân',
            purity: isBanMa ? 0.5 : 1.0,
            purityLabel: isBanMa ? 'Bán Ma Huyết Mạch (50% Ma Khí)' : 'Ma Tộc Thuần Chủng 100%',
            age,
            matureAge: 20,
            naturalBaseMeters: 1.75,
            cultivationMult: 1.0,
            totalMeters: 1.75,
            totalTruong: 0.53,
            displayStr: '0.53 trượng (~1.75 mét)',
            scaleComparison: 'Thể trạng Ma Tộc tinh xảo mị hoặc hoặc uy nghiêm ma đạo',
            combatFormDesc: 'Bản thể Ma Tộc ngưng tụ ma khí hộ thể, đồng tử ma nhãn hoặc mị nhãn câu hồn, vận chuyển ma lực xé rách đạo pháp.',
            humanFormMeters,
            humanFormDisplay,
            customHeightStr: parsedH.display,
            hasPhapTuong: true,
            phapTuongDesc: 'Chân Ma Pháp Tướng ngập tràn ma khí tựa Ma Thần viễn cổ!'
        };
    }

    // Nhánh Thảo Mộc / Dị Thạch Thành Tinh
    const isThaoThach = ['bg_21', 'bg_22', 'bg_23'].includes(bgId) || 
                        tNames.includes('Thạch Thần Bất Diệt') || 
                        tNames.includes('Bảo Vật Hóa Hình');
    if (isThaoThach) {
        const matureAge = 30;
        const ageRatio = Math.min(1.0, Math.max(0.1, age / matureAge));
        const naturalBaseMeters = 0.5 + (3.0 - 0.5) * ageRatio;
        const ttMultMap: Record<number, number> = { 1: 1.0, 4: 10.0, 7: 100.0, 10: 500.0, 18: 1500.0, 20: 2000.0 };
        const mult = ttMultMap[stageId] || (stageId >= 7 ? 100.0 : 5.0);
        const totalMeters = naturalBaseMeters * mult;
        const totalTruong = Math.round((totalMeters / 3.3) * 10) / 10;
        const kmStr = totalMeters >= 1000 ? ` (~${(totalMeters / 1000).toFixed(1)} km)` : ` (~${Math.round(totalMeters)} m)`;

        return {
            category: 'thao_thach',
            categoryLabel: 'Dị Thảo & Thạch Tinh Thành Tinh',
            name: bgId === 'bg_21' ? 'Thạch Thần Bản Nguyên Chân Thân' : 'Cổ Thụ Thần Mộc Chân Thân',
            purity: 1.0,
            purityLabel: 'Linh Tinh Khởi Sinh 100%',
            age,
            matureAge,
            naturalBaseMeters,
            cultivationMult: mult,
            totalMeters,
            totalTruong,
            displayStr: `${totalTruong.toLocaleString()} trượng${kmStr}`,
            scaleComparison: stageId >= 7 ? 'Cổ thụ nghìn trượng cắm rễ vào đại địa, cự thạch chọc thủng tầng mây' : 'Linh mộc cự thạch nguyên bản',
            combatFormDesc: `Khi bộc phát bản nguyên, hiển lộ chân thân thảo mộc/dị thạch khổng lồ ${totalTruong.toLocaleString()} trượng${kmStr} đè ép hư không!`,
            humanFormMeters,
            humanFormDisplay,
            customHeightStr: parsedH.display,
            hasPhapTuong: false
        };
    }

    // Nhánh Nhân Tộc & Các Hệ Tu Khác (Pháp Tu, Kiếm Tu, Thể Tu Nhân Tộc...)
    const hasPttdSkill = (state.journal?.skills || []).some((s: any) => s && /pháp thiên tượng địa/i.test(s.name || ''));
    const canSummonPhapTuong = stageId >= 18 || hasPttdSkill;
    const phapTuongTruong = canSummonPhapTuong ? Math.min(3300, Math.round(stageId * 165)) : 0;

    const nhanTocTruong = Math.round((humanFormMeters / 3.3) * 100) / 100;
    return {
        category: 'nhan_toc',
        categoryLabel: canSummonPhapTuong ? 'Nhân Tộc Đại Năng (Ngưng Tụ Pháp Tướng Hộ Thể)' : 'Nhân Tộc Phàm Thân',
        name: canSummonPhapTuong ? 'Đại Đạo Pháp Tướng Hộ Thể' : 'Nhân Tộc Thể Phách',
        purity: 1.0,
        purityLabel: 'Nhân Tộc Thuần Chủng 100%',
        age,
        matureAge: 18,
        naturalBaseMeters: baseH,
        cultivationMult: humanMult,
        totalMeters: humanFormMeters,
        totalTruong: nhanTocTruong,
        displayStr: canSummonPhapTuong 
            ? `Nhục thân ${nhanTocTruong} trượng (~${humanFormDisplay}) | Pháp Tướng: ${phapTuongTruong.toLocaleString()} trượng (~${Math.round(phapTuongTruong * 3.3 / 1000)} km)`
            : `${nhanTocTruong} trượng (~${humanFormDisplay})`,
        scaleComparison: canSummonPhapTuong 
            ? `Quy mô Pháp Tướng tối đa bằng 1/3 Chân Thân Thần Thú Yêu Tộc cùng cấp (${phapTuongTruong.toLocaleString()} trượng)`
            : `Thể trạng nhân tộc ${humanFormDisplay}, vóc dáng thon gọn linh hoạt điều khiển vạn pháp`,
        combatFormDesc: canSummonPhapTuong 
            ? `Chiến đấu với thể trạng nhân tộc linh hoạt ${humanFormDisplay}, khi lâm trận có thể ngưng tụ Đại Đạo Pháp Tướng khổng lồ cao ${phapTuongTruong.toLocaleString()} trượng hộ thể phía sau!`
            : `Nhục thân con người ${humanFormDisplay}, thi triển kiếm khí đạo pháp thanh thoát.`,
        hasPhapTuong: canSummonPhapTuong,
        phapTuongDesc: canSummonPhapTuong ? `Pháp Tướng Hộ Thể cao ${phapTuongTruong.toLocaleString()} trượng (khoảng 1/3 Chân Thân Yêu Thần).` : undefined,
        humanFormMeters,
        humanFormDisplay,
        customHeightStr: parsedH.display
    };
};
