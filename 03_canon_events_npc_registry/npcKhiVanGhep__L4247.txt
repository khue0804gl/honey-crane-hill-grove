const npcKhiVanGhep = (npc) => {
  const ts = Array.isArray(npc?.traits) ? npc.traits.map(t => typeof t === 'object' ? (t.name || '') : String(t)) : [];
  let chiSo = 1, tho = 1, tran = 2, coTien = false, tranHieu = 99, dacBiet = [];
  for (const t of ts) {
    const h = NPC_KHI_VAN_HIEU_UNG[t];
    if (!h) continue;
    chiSo *= h.chiSo; tho *= h.tho;
    if (h.tran === 0) coTien = true;            // Thien Menh: KHONG tran — chenep moi thu
    else if (h.tran < tranHieu) tranHieu = h.tran; // trait han che keo tran xuong
    if (h.mo) dacBiet.push(h.mo);
  }
  if (coTien) tran = 0;
  else if (tranHieu < 99) tran = tranHieu;      // loi keo<2, loc cao>2 — trait cuoi cung thang
  return { chiSo: Math.max(0.3, Math.min(3.0, chiSo)), tho, tran, dacBiet };
};
