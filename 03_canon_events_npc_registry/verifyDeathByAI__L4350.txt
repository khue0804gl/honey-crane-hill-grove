const verifyDeathByAI = async (npcsToVerify, contextStory) => {
    if (!npcsToVerify || npcsToVerify.length === 0) return [];
    
    const sysPrompt = `BẠN LÀ THẨM PHÁN SINH TỬ (DEATH & RESURRECTION INTERCEPTOR).
Nhiệm vụ của bạn là đọc NGỮ CẢNH (Cốt truyện, Diễn biến, Trạng thái, Ký ức) và PHÁN XÉT CỰC KỲ KHÁCH QUAN xem các nhân vật sau đây:
1. ĐÃ THỰC SỰ CHẾT (Mất mạng, tắt thở, xác chết, hồn phi phách tán, bị giết, tử vong)?
2. HAY CHƯA CHẾT / ĐÃ SỐNG LẠI (Chỉ bị thương, chạy thoát, được cứu sống, giả chết, nói ẩn dụ/thành ngữ, HOẶC tàn hồn ngưng tụ, có công nghệ khoa học/kén hồi phục tái tạo nhục thân, tái sinh, chuyển sinh, phục sinh, phân thân đoạt xá, HOẶC BỊ HÓA THÀNH THI MA / CƯƠNG THI / VONG LINH)?

[NGỮ CẢNH TỔNG THỂ ĐỂ ĐỐI CHIẾU]:
"${contextStory || "Không có cốt truyện chung."}"

[DANH SÁCH NHÂN VẬT CẦN PHÁN XÉT]:
${npcsToVerify.map((n, i) => `${i+1}. Tên nhân vật: ${n.name}\n   - Trạng thái hiện hữu: ${n.state || 'Không rõ'}\n   - Ký ức tóm tắt: ${String(n.historySummary || 'Không rõ').slice(-300)}\n   - Nhận định sơ bộ của hệ thống: ${n.isDead ? 'Một cái xác đã chết' : 'Chưa rõ/Có dấu hiệu báo tử'}`).join('\n\n')}

LỆNH TỐI CAO: 
- Khái niệm Tử Vong nghĩa là: Không còn sự sống vật lý tồn tại, hoặc linh hồn đã hồn phi phách tán.
- Nhiều khi từ "Chết" chỉ là lời đe dọa (VD: "Muốn chết à"), thành ngữ (VD: "Chết lặng", "Khóc chết", "Chết tiệt"), hoặc đã được người khác cứu (VD: "Suýt chết", "Giữ lại một mạng"). Đôi khi chỉ là giả chết.
- ĐẶC BIỆT LƯU Ý BẢO HỘ TÂN THỦ VÀ CHẠY TRỐN: Nếu NGUYÊN BẢN CỐT TRUYỆN nhắc tới việc nhân vật này được cứu, chạy thoát thân, hoặc chỉ trọng thương, BẠN BẮT BUỘC ĐÁNH GIÁ LÀ CHƯA CHẾT!
- ĐẶC BIỆT LƯU Ý KHOA HỌC / HUYỄN THUẬT / ĐOẠT XÁ / LUYỆN THI (V12.20): Nhân vật CÓ THỂ ĐÃ BỊ HỆ THỐNG GHI NHẬN CHẾT TỪ CHƯƠNG TRƯỚC, nhưng nếu chương này miêu tả họ đang thức tỉnh, tái tạo nhục thân, phân thân tẩu thoát thành công, trùng sinh, đoạt xá, HOẶC BỊ LUYỆN THÀNH THI MA / CƯƠNG THI tiếp tục tồn tại và chiến đấu -> BẠN BẮT BUỘC ĐÁNH GIÁ LÀ CHƯA CHẾT (isActuallyDead: false) VÀ GIẢI THÍCH RÕ LÝ DO HỌ TỒN TẠI!

HÃY DÙNG TRÍ TUỆ NHÂN TẠO CỦA BẠN ĐỂ SUY LUẬN TOÀN DIỆN CHỨ KHÔNG ĐƯỢC CHIA RA TỪ KHÓA MỘT CÁCH MÁY MÓC!
- Trả về JSON MẢNG CHÍNH XÁC (không thêm gì ngoài JSON). Array có cấu trúc:
[
  { "name": "Tên nhân vật", "isActuallyDead": true, "reason": "Bị chém đứt đầu trong trận chiến" },
  { "name": "Tên nhân vật", "isActuallyDead": false, "reason": "Mặc dù thân xác cũ bị hủy, nhưng đã dùng kén hồi phục Med-Pod để tái tạo nhục thân, ý thức vẫn linh hoạt." }
]`;

    try {
        const payload = { contents: [{ parts: [{ text: "Hãy phán xét sinh tử!" }] }], systemInstruction: { parts: [{ text: sysPrompt }] }, generationConfig: { responseMimeType: "application/json" } };
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
        const result = await response.json();
        let parts = result.candidates?.[0]?.content?.parts || [];
        let textResult = parts.filter(p => !p.thought).map(p => p.text).join('\n') || parts[0]?.text || '[]';
        let data = [];
        const cbMatch = textResult.match(/```(?:json)?\s*(\[[\s\S]*?\])\s*```/i);
        if (cbMatch) {
            textResult = cbMatch[1];
        } else {
            const first = textResult.indexOf('[');
            if (first !== -1) {
                let count = 0;
                for (let i = first; i < textResult.length; i++) {
                    if (textResult[i] === '[') count++;
                    else if (textResult[i] === ']') {
                        count--;
                        if (count === 0) {
                            textResult = textResult.substring(first, i + 1);
                            break;
                        }
                    }
                }
            }
        }
        try {
            data = JSON.parse(textResult.replace(/```json/gi, '').replace(/```/g, '').trim());
        } catch (e) {
            console.error("Lỗi parse JSON Tử Thần:", e);
        }
        return Array.isArray(data) ? data : [];
    } catch (e) {
        console.error("Lưỡi hái tử thần AI lỗi:", e);
        return []; 
    }
};
