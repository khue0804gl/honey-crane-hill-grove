      {/* V5.41: GIAO DIỆN TỈ THÍ (DANH SÁCH & LỜI MỜI) */}
      {isPvpListOpen && currentCoopRoom && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center bg-[var(--bg-overlay)] p-4 animate-in fade-in duration-200">
          <div className="bg-[var(--bg-panel)] w-full max-w-md rounded-2xl flex flex-col border border-[var(--border-main)] shadow-2xl relative">
            <div className="p-4 border-b border-[var(--border-main)] flex justify-between items-center bg-[var(--bg-surface)] rounded-t-2xl">
               <h3 className="text-lg font-bold text-red-500 flex items-center gap-2"><Swords size={20}/> Thách Đấu Liên Giới</h3>
               <button onClick={() => setIsPvpListOpen(false)} className="p-1 hover:bg-[var(--bg-hover)] rounded text-[var(--text-muted)]"><X size={20}/></button>
            </div>
            <div className="p-5 space-y-4 max-h-[60vh] overflow-y-auto custom-scrollbar">
                {/* Lời mời đến mình */}
                {(currentCoopRoom.pvpRequests || []).filter(r => r.toUid === user?.uid && Date.now() - r.timestamp < 180000).map((req, i) => (
                    <div key={`req-${i}`} className="bg-red-900/20 p-4 rounded-xl border border-red-500/50 animate-pulse shadow-sm">
                        <p className="text-sm font-bold text-red-400 mb-2">Chiến thư từ: {req.fromName}</p>
                        <div className="flex gap-2">
                            <button onClick={() => acceptPvpInvite(req)} className="flex-1 bg-red-600 hover:bg-red-500 text-white font-bold py-2 rounded-lg text-sm">Chấp Nhận</button>
                            <button onClick={() => declinePvpInvite(req)} className="flex-1 bg-[var(--bg-body)] text-[var(--text-muted)] hover:text-white border border-[var(--border-main)] font-bold py-2 rounded-lg text-sm">Từ Chối</button>
                        </div>
                    </div>
                ))}
                
                {/* Danh sách người chơi để thách đấu */}
                <h4 className="text-xs font-bold text-[var(--text-muted)] uppercase tracking-widest border-b border-[var(--border-main)] pb-2 mb-3">Đạo Hữu Trong Phòng</h4>
                {(currentCoopRoom.players || []).filter(p => p.uid !== user?.uid).map(p => {
                    const hasSent = (currentCoopRoom.pvpRequests || []).some(r => r.fromUid === user?.uid && r.toUid === p.uid && Date.now() - r.timestamp < 180000);
                    return (
                        <div key={p.uid} className="flex justify-between items-center bg-[var(--bg-body)] p-3 rounded-xl border border-[var(--border-main)]">
                            <div className="flex items-center gap-3">
                                <div className={`p-2 rounded-full border ${p.isOnline ? 'bg-teal-900/30 border-teal-500/50 text-teal-500' : 'bg-gray-800 border-gray-600 text-gray-500'}`}><User size={16}/></div>
                                <div><div className="font-bold text-[var(--text-main)] text-sm">{p.characterName || p.name}</div></div>
                            </div>
                            <button onClick={() => sendPvpInvite(p)} disabled={!p.isOnline || hasSent} className="bg-red-900/20 hover:bg-red-600 disabled:opacity-50 text-red-500 hover:text-white px-3 py-1.5 rounded-lg border border-red-900/50 text-xs font-bold transition-colors">
                                {hasSent ? 'Đang gửi...' : 'Tỉ Thí'}
                            </button>
                        </div>
                    );
                })}
            </div>
          </div>
        </div>
      )}

      {/* V5.41: VÕ ĐÀI GIẢ TƯỞNG (PVP ARENA MÀN HÌNH CHÍNH) */}
      {currentCoopRoom?.pvpMatch && (
          <div className="fixed inset-0 z-[100] bg-[var(--bg-body)] flex flex-col animate-in fade-in duration-500">
              <div className="p-3 md:p-4 border-b border-red-900/50 bg-gradient-to-b from-red-950/40 to-transparent flex justify-between items-center shrink-0">
                  <div className="flex items-center gap-3 text-red-500 font-bold uppercase tracking-widest"><Swords size={24} className="animate-pulse"/> Võ Đài Giả Tưởng</div>
                  {currentCoopRoom.pvpMatch.status === 'finished' && <button onClick={endPvpMatch} className="px-4 py-2 bg-red-600 text-white rounded-lg text-sm font-bold">Thoát Võ Đài</button>}
              </div>

              {/* HUD Máu/Năng Lượng 2 Bên */}
              <div className="grid grid-cols-2 gap-4 p-4 shrink-0 bg-[var(--bg-surface)] border-b border-[var(--border-main)] shadow-md">
                  {[currentCoopRoom.pvpMatch.p1, currentCoopRoom.pvpMatch.p2].filter(Boolean).map((p, i) => (
                      <div key={i} className={`p-4 rounded-xl border ${p.uid === user?.uid ? 'bg-blue-900/10 border-blue-500/50' : 'bg-red-900/10 border-red-500/50'} relative overflow-hidden`}>
                          <div className={`absolute top-0 ${i === 0 ? 'left-0' : 'right-0'} w-2 h-full ${p.uid === user?.uid ? 'bg-blue-500' : 'bg-red-500'}`}></div>
                          <div className="pl-4">
                              <h4 className={`font-bold text-lg mb-2 ${p.uid === user?.uid ? 'text-blue-400' : 'text-red-400'}`}>{p.name} <span className="text-[10px] text-[var(--text-muted)] font-normal">{p.level}</span></h4>
                              <div className="space-y-2">
                                  <div>
                                      <div className="flex justify-between text-[10px] font-bold text-red-300 mb-1"><span>HP</span><span>{Math.floor(p.hp)}/{p.maxHp}</span></div>
                                      <div className="h-1.5 w-full bg-red-950 rounded-full"><div className="h-full bg-red-500 transition-all duration-300" style={{width: `${Math.max(0,(p.hp/p.maxHp)*100)}%`}}></div></div>
                                  </div>
                                  <div>
                                      <div className="flex justify-between text-[10px] font-bold text-teal-300 mb-1"><span className="uppercase">{p.energyName}</span><span>{Math.floor(p.mp)}/{p.maxMp}</span></div>
                                      <div className="h-1 w-full bg-teal-950 rounded-full"><div className="h-full bg-teal-500 transition-all duration-300" style={{width: `${Math.max(0,(p.mp/p.maxMp)*100)}%`}}></div></div>
                                  </div>
                              </div>
                          </div>
                      </div>
                  ))}
              </div>

              {/* Diễn Biến Log */}
              <div ref={pvpLogScrollRef} className="flex-1 overflow-y-auto p-4 md:p-6 custom-scrollbar space-y-4">
                  {(currentCoopRoom.pvpMatch.logs || []).map((l, i) => (
                      <div key={i} className={`animate-in slide-in-from-bottom-2 ${l.type==='system' ? 'text-center' : l.type==='action' ? (l.uid===user?.uid ? 'text-right' : 'text-left') : 'text-left'}`}>
                          {l.type === 'system' ? <span className="inline-block px-4 py-2 bg-gray-900 border border-gray-700 rounded-full text-xs text-[var(--text-muted)]">{l.text}</span> :
                           l.type === 'action' ? <span className={`inline-block px-4 py-2 rounded-2xl text-sm font-bold shadow-sm ${l.uid===user?.uid ? 'bg-blue-600 text-white rounded-tr-sm' : 'bg-red-900/50 text-red-400 border border-red-500/30 rounded-tl-sm'}`}>{l.uid!==user?.uid && <span className="block text-[9px] text-red-300 mb-0.5">{l.name}</span>}{l.text}</span> :
                           <div className="p-4 bg-[var(--bg-surface)] border border-[var(--border-main)] rounded-2xl text-[15px] leading-relaxed text-[var(--text-main)]">{l.text}</div>}
                      </div>
                  ))}
                  {isPvpResolving && <div className="text-center text-red-500 font-bold italic animate-pulse py-4">Thiên Đạo đang phân định sát thương...</div>}
              </div>

              {/* Ô Nhập Lệnh */}
              {currentCoopRoom.pvpMatch.status === 'playing' && (
                  <div className="p-4 bg-[var(--bg-surface)] border-t border-[var(--border-main)] shrink-0 flex flex-col gap-3 relative">
                      {currentCoopRoom.pvpMatch.currentTurnUid === user?.uid ? (
                          <div className="absolute -top-12 left-1/2 -translate-x-1/2 flex flex-col items-center animate-in fade-in slide-in-from-bottom-2">
                              <span className="text-xs font-bold text-yellow-400 bg-yellow-900/50 px-3 py-1 rounded-full border border-yellow-500/50 shadow-lg">ĐẾN LƯỢT CỦA BẠN</span>
                              <span className={`text-2xl font-black ${pvpTurnTimer <= 10 ? 'text-red-500 animate-pulse scale-125' : 'text-yellow-400'} drop-shadow-md transition-all`}>{pvpTurnTimer}s</span>
                          </div>
                      ) : (
                           <div className="absolute -top-8 left-1/2 -translate-x-1/2">
                              <span className="text-xs font-bold text-[var(--text-muted)] bg-[var(--bg-body)] px-3 py-1 rounded-full border border-[var(--border-main)]">Đang chờ đối thủ ra chiêu...</span>
                          </div>
                      )}
                      <form onSubmit={(e) => { e.preventDefault(); handlePvpRealtimeAction(pvpInput); }} className="relative flex gap-2">
                          <textarea 
                              maxLength={200} 
                              value={pvpInput} 
                              onChange={e => {
                                  setPvpInput(e.target.value);
                                  e.target.style.height = 'auto';
                                  e.target.style.height = (e.target.scrollHeight) + 'px';
                              }} 
                              onKeyDown={(e) => {
                                  if (e.key === 'Enter' && !e.shiftKey) {
                                      e.preventDefault();
                                      if (pvpInput.trim() && currentCoopRoom.pvpMatch.currentTurnUid === user?.uid && !isPvpResolving) {
                                          handlePvpRealtimeAction(pvpInput);
                                          e.target.style.height = 'auto';
                                      }
                                  }
                              }}
                              rows={1}
                              disabled={currentCoopRoom.pvpMatch.currentTurnUid !== user?.uid || isPvpResolving} 
                              placeholder={currentCoopRoom.pvpMatch.currentTurnUid === user?.uid ? "Nhập hành động của bạn (Tung chiêu, phòng thủ...)" : "Đợi đối thủ ra tay..."} 
                              className="flex-1 bg-[var(--bg-body)] border border-[var(--border-main)] focus:border-red-500 focus:outline-none rounded-xl px-4 py-3 text-[var(--text-main)] resize-none overflow-hidden" 
                              style={{ minHeight: '48px', maxHeight: '150px' }}
                          />
                          <button type="submit" disabled={!pvpInput.trim() || currentCoopRoom.pvpMatch.currentTurnUid !== user?.uid || isPvpResolving} className="bg-red-600 hover:bg-red-500 disabled:opacity-50 text-white px-5 rounded-xl font-bold transition-colors" title="Tung chiêu"><Send size={20}/></button>
                          
                          {/* V5.43: Nút Thoát Khỏi (Đầu hàng nhanh) */}
                          <button type="button" onClick={surrenderArena} disabled={isPvpResolving} className="bg-gray-800 hover:bg-red-900/80 text-gray-400 hover:text-red-400 border border-gray-600 hover:border-red-500/50 disabled:opacity-50 px-4 rounded-xl font-bold transition-colors flex items-center justify-center shrink-0" title="Bỏ cuộc & Thoát nhanh">
                              Thoát
                          </button>
                      </form>
                  </div>
              )}
          </div>
      )}

      {/* CHAT TRUYỀN ÂM MODAL (V5.30) */}
      {isChatOpen && currentCoopRoom && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center bg-[var(--bg-overlay)] p-2 md:p-4 animate-in fade-in duration-200">
          <div className="bg-[var(--bg-panel)] w-full max-w-5xl h-[85vh] rounded-2xl flex flex-col border border-[var(--border-main)] shadow-2xl relative overflow-hidden">
            <div className="p-4 border-b border-[var(--border-main)] flex justify-between items-center bg-[var(--bg-surface)] shrink-0">
               <div className="flex items-center gap-3">
                   <div className="bg-blue-900/30 p-2 rounded-full border border-blue-500/50"><Mail className="text-blue-500" size={20} /></div>
                   <div>
                       <h3 className="text-lg md:text-xl font-bold text-blue-400 uppercase tracking-widest">Truyền Âm (Nhắn Tin)</h3>
                       <p className="text-[10px] md:text-xs text-[var(--text-muted)]">Liên lạc bí mật hoặc đàm đạo toàn vi diện</p>
                   </div>
               </div>
               <button onClick={() => setIsChatOpen(false)} className="p-2 bg-[var(--bg-body)] hover:bg-[var(--bg-hover)] border border-[var(--border-main)] rounded text-[var(--text-muted)]"><X size={20}/></button>
            </div>
            
            <div className="flex-1 flex overflow-hidden">
                {/* CỘT TRÁI: Sidebar Người Dùng */}
                <div className="w-1/3 md:w-64 border-r border-[var(--border-main)] bg-[var(--bg-body)] flex flex-col">
                    <div className="p-3 border-b border-[var(--border-main)] font-bold text-[10px] text-[var(--text-muted)] uppercase tracking-widest bg-[var(--bg-surface)] shadow-sm">Danh Sách Liên Lạc</div>
                    <div className="flex-1 overflow-y-auto custom-scrollbar p-2 space-y-1">
                        <div onClick={() => setActiveChatTab('global')} className={`cursor-pointer p-3 rounded-xl flex items-center gap-3 transition-colors ${activeChatTab === 'global' ? 'bg-blue-900/30 border border-blue-500/50' : 'hover:bg-[var(--bg-hover)] border border-transparent'}`}>
                            <div className="w-10 h-10 rounded-full bg-blue-500/20 flex items-center justify-center shrink-0"><Globe size={20} className="text-blue-400"/></div>
                            <div className="flex-1 overflow-hidden">
                                <div className="font-bold text-sm text-[var(--text-bold)] truncate">Kênh Thế Giới</div>
                                <div className="text-[10px] text-[var(--text-muted)] truncate">Tất cả đạo hữu</div>
                            </div>
                        </div>
                        {(currentCoopRoom.players || []).filter(p => p.uid !== user?.uid).map(p => (
                            <div key={p.uid} onClick={() => setActiveChatTab(p.uid)} className={`cursor-pointer p-3 rounded-xl flex items-center gap-3 transition-colors ${activeChatTab === p.uid ? 'bg-teal-900/30 border border-teal-500/50' : 'hover:bg-[var(--bg-hover)] border border-transparent'}`}>
                                <div className="w-10 h-10 rounded-full bg-teal-500/20 flex items-center justify-center shrink-0 relative">
                                    <User size={20} className="text-teal-400"/>
                                    {p.isOnline && <div className="absolute bottom-0 right-0 w-3 h-3 bg-emerald-500 border-2 border-[#090c10] rounded-full"></div>}
                                </div>
                                <div className="flex-1 overflow-hidden">
                                    <div className="font-bold text-sm text-[var(--text-bold)] truncate">{p.characterName || p.name}</div>
                                    <div className="text-[10px] text-[var(--text-muted)] truncate">{p.isOnline ? 'Đang Online' : 'Offline'}</div>
                                </div>
                            </div>
                        ))}
                    </div>
                </div>

                {/* CỘT PHẢI: Khung Nhắn Tin */}
                <div className="flex-1 flex flex-col bg-[var(--bg-body)]">
                    <div className="p-3 border-b border-[var(--border-main)] bg-[var(--bg-surface)] font-bold text-sm text-blue-400 shadow-sm z-10 flex items-center gap-2">
                        {activeChatTab === 'global' ? <><Globe size={16}/> Kênh Thế Giới (Công Khai)</> : <><MessageCircle size={16}/> Trò chuyện Riêng tư: {(currentCoopRoom.players || []).find(p => p.uid === activeChatTab)?.characterName || 'Đạo hữu'}</>}
                    </div>
                    <div className="flex-1 p-4 overflow-y-auto custom-scrollbar space-y-4" ref={chatScrollRef}>
                        {chatMessages.filter(m => activeChatTab === 'global' ? m.targetUid === 'global' : ((m.targetUid === activeChatTab && m.uid === user?.uid) || (m.targetUid === user?.uid && m.uid === activeChatTab))).length === 0 ? (
                            <div className="text-center text-[var(--text-muted)] italic mt-10">Chưa có thư từ truyền âm nào. Gửi một tin để chào hỏi đi!</div>
                        ) : (
                            chatMessages.filter(m => activeChatTab === 'global' ? m.targetUid === 'global' : ((m.targetUid === activeChatTab && m.uid === user?.uid) || (m.targetUid === user?.uid && m.uid === activeChatTab))).map((msg, i) => {
                                const isMe = msg.uid === user?.uid;
                                return (
                                    <div key={i} className={`flex flex-col ${isMe ? 'items-end' : 'items-start'} animate-in fade-in slide-in-from-bottom-2`}>
                                        {!isMe && activeChatTab === 'global' && <span className="text-[10px] font-bold text-[var(--text-muted)] mb-1 ml-1">{msg.name}</span>}
                                        <div className={`max-w-[85%] md:max-w-[75%] p-3 text-[15px] leading-relaxed ${isMe ? 'bg-blue-600 text-white rounded-2xl rounded-tr-sm shadow-md' : 'bg-[var(--bg-surface)] border border-[var(--border-main)] text-[var(--text-main)] rounded-2xl rounded-tl-sm shadow-sm'}`}>
                                            {msg.text}
                                        </div>
                                        <span className="text-[9px] font-mono text-[var(--text-muted)] mt-1 mx-1">{new Date(msg.timestamp).toLocaleTimeString('vi-VN')}</span>
                                    </div>
                                );
                            })
                        )}
                    </div>
                    <form onSubmit={handleSendChatMessage} className="p-3 bg-[var(--bg-surface)] border-t border-[var(--border-main)] flex gap-2 shrink-0">
                        <textarea 
                            maxLength={500} 
                            value={chatInput} 
                            onChange={e => {
                                setChatInput(e.target.value);
                                e.target.style.height = 'auto';
                                e.target.style.height = (e.target.scrollHeight) + 'px';
                            }} 
                            onKeyDown={(e) => {
                                if (e.key === 'Enter' && !e.shiftKey) {
                                    e.preventDefault();
                                    if (chatInput.trim()) {
                                        handleSendChatMessage(e);
                                        e.target.style.height = 'auto';
                                    }
                                }
                            }}
                            rows={1}
                            placeholder="Nhập mật ngữ truyền âm..." 
                            className="flex-1 bg-[var(--bg-body)] border border-[var(--border-main)] rounded-xl px-4 py-3 text-sm text-[var(--text-main)] focus:outline-none focus:border-blue-500 shadow-inner resize-none overflow-hidden" 
                            style={{ minHeight: '46px', maxHeight: '150px' }}
                        />
                        <button type="submit" disabled={!chatInput.trim()} className="bg-blue-600 hover:bg-blue-500 disabled:opacity-50 text-white px-6 rounded-xl transition-colors shadow-md flex items-center justify-center"><Send size={18}/></button>
                    </form>
                </div>
            </div>
          </div>
        </div>
      )}

      {/* MAP & LIVING WORLD SIMULATION MODAL (V3.0 THỰC TẾ TU TIÊN) */}
      {isMapOpen && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center bg-[var(--bg-panel)] animate-in fade-in duration-200">
          <div className="w-full h-full flex flex-col relative overflow-hidden bg-[var(--bg-panel)]">
            {/* Header Thanh Bản Đồ */}
            <div className="p-3 md:p-4 border-b border-[var(--border-main)] flex flex-wrap justify-between items-center bg-[var(--bg-surface)] shrink-0 shadow-lg z-30 gap-3">
               <div className="flex items-center gap-3">
                   <div className="p-2.5 bg-emerald-950/50 border border-emerald-500/40 rounded-xl text-emerald-400 shadow-[0_0_15px_rgba(16,185,129,0.2)]">
                       <Map size={24} />
                   </div>
                   <div>
                       <div className="flex items-center gap-2 flex-wrap">
                           <h3 className="text-base md:text-lg font-black text-emerald-400 uppercase tracking-widest">
                               Sơn Hà Địa Đồ - {gameState.worldName || 'Cửu Châu Tiên Giới'}
                           </h3>
                           <span className="text-[11px] bg-purple-900/30 px-2.5 py-0.5 rounded border border-purple-500/40 font-mono text-purple-300 font-bold">
                               {gameState.timelineTitle || 'Thanh Vân Khí Vận Kỷ'}
                           </span>
                           <span className="text-[11px] bg-[var(--bg-body)] px-2.5 py-0.5 rounded border border-[var(--border-main)] font-mono text-teal-300 font-bold">
                               Tọa Độ: ({gameState.coordinates?.x ?? 0}, {gameState.coordinates?.y ?? 0}, z={gameState.coordinates?.z ?? currentPlaneZ})
                           </span>
                       </div>
                       <p className="text-xs text-[var(--text-muted)] mt-0.5 flex items-center gap-2 flex-wrap">
                           <span>Vị trí: <strong className="text-emerald-400">{gameState.currentLocation || 'Khởi Nguyên Chi Địa'}</strong></span>
                           <span>| Tốc độ: <strong className="text-orange-400">{formatLargeNumber(totalSpd)}</strong> dặm/h</span>
                           <span>| Thể lực: <strong className="text-yellow-400">{gameState.stamina}/100</strong></span>
                           <span>| Năng lượng: <strong className="text-teal-400">{formatLargeNumber(gameState.spiritualEnergy)}</strong></span>
                       </p>
                   </div>
               </div>

               {/* Tabs Chuyển Đổi & Nút Đóng */}
               <div className="flex items-center gap-2 flex-wrap">
                 {/* 3 Tab chính (Canon & Sử Ký chỉ Admin mới thấy) */}
                 <div className="flex bg-[var(--bg-body)] p-1 rounded-xl border border-[var(--border-main)]">
                    <button 
                       onClick={() => setMapActiveTab('world')} 
                       className={`px-3 py-1.5 rounded-lg text-xs font-bold transition-all flex items-center gap-1.5 ${mapActiveTab === 'world' ? 'bg-emerald-600 text-white shadow-md' : 'text-[var(--text-muted)] hover:text-white'}`}
                    >
                       <Castle size={14}/> {gameState.worldName || 'Cửu Châu Giang Hồ'}
                    </button>
                    <button 
                       onClick={() => setMapActiveTab('grid')} 
                       className={`px-3 py-1.5 rounded-lg text-xs font-bold transition-all flex items-center gap-1.5 ${mapActiveTab === 'grid' ? 'bg-teal-600 text-white shadow-md' : 'text-[var(--text-muted)] hover:text-white'}`}
                    >
                       <Crosshair size={14}/> Thám Hiểm 2D
                    </button>
                    {isAdminLoggedIn && (
                       <button 
                          onClick={() => setMapActiveTab('canon')} 
                          className={`px-3 py-1.5 rounded-lg text-xs font-bold transition-all flex items-center gap-1.5 ${mapActiveTab === 'canon' ? 'bg-purple-600 text-white shadow-md' : 'text-[var(--text-muted)] hover:text-white'}`}
                       >
                          <Clock size={14}/> Canon & Sử Ký (Admin)
                       </button>
                    )}
                 </div>

                 {/* Chuyển Tầng z */}
                 <div className="flex bg-[var(--bg-body)] p-1 rounded-xl border border-[var(--border-main)]" title="Chuyển tầng không gian (Tam Giới)">
                    <button 
                       onClick={() => setCurrentPlaneZ(1)} 
                       className={`px-2.5 py-1 rounded-lg text-[10px] font-bold transition-all flex items-center gap-1 ${currentPlaneZ === 1 ? 'bg-purple-600 text-white shadow-sm' : 'text-purple-400/60 hover:text-purple-300'}`}
                    >
                       <Sparkles size={12}/> Thượng Giới
                    </button>
                    <button 
                       onClick={() => setCurrentPlaneZ(0)} 
                       className={`px-2.5 py-1 rounded-lg text-[10px] font-bold transition-all flex items-center gap-1 ${currentPlaneZ === 0 ? 'bg-emerald-600 text-white shadow-sm' : 'text-emerald-400/60 hover:text-emerald-300'}`}
                    >
                       <Home size={12}/> Mặt Đất
                    </button>
                    <button 
                       onClick={() => setCurrentPlaneZ(-1)} 
                       className={`px-2.5 py-1 rounded-lg text-[10px] font-bold transition-all flex items-center gap-1 ${currentPlaneZ === -1 ? 'bg-red-600 text-white shadow-sm' : 'text-red-400/60 hover:text-red-300'}`}
                    >
                       <Skull size={12}/> Cửu U
                    </button>

                                             </div>
                                          </>
                                       )}
                                    </div>
                                 )}

                     </div>
                  </div>

               {mapActiveTab === 'grid' && (
                  <div className="flex-1 flex flex-col lg:flex-row overflow-hidden h-full">

'bg-emerald-600 text-white border-emerald-400 shadow-md scale-105' : 'bg-[var(--bg-surface)] text-gray-400 border-[var(--border-main)] hover:text-white'}`}

                                       : `Đã quan sát được địa hình thực tế (${selectedGridCell.terrainName}). Chưa biết tên gọi chính thức vì chưa hỏi thăm dân địa phương hoặc chưa đọc biển hiệu.`}
                                 </p>
                              </div>

                              <div className="space-y-2 bg-[var(--bg-surface)] p-3 rounded-xl border border-[var(--border-main)] text-xs">
                                 <div className="flex justify-between"><span className="text-teal-400 font-bold">Linh Mạch:</span><span className="font-mono text-teal-300 font-bold">{formatLargeNumber(selectedGridCell.aura)} điểm</span></div>
                                 <div className="flex justify-between"><span className="text-blue-400 font-bold">Độ Bền:</span><span className="font-mono text-blue-300 font-bold">{formatLargeNumber(selectedGridCell.integrity)}/5000</span></div>
                                 <div className="flex justify-between"><span className="text-purple-400 font-bold">Tạp Khí:</span><span className="font-mono text-purple-300 font-bold">{selectedGridCell.pollution} điểm</span></div>
                                 <div className="flex justify-between pt-1 border-t border-gray-800">
                                    <span className="text-gray-400">Thế lực:</span>
                                    <span className="text-yellow-400 font-bold">{isNameKnown ? selectedGridCell.rulingFaction : 'Chưa xác thực'}</span>
                                 </div>
                                 {selectedGridCell.resources && selectedGridCell.resources.length > 0 && (
                                    <div className="flex justify-between pt-1 border-t border-gray-800"><span className="text-gray-400">Sản vật:</span><span className="text-emerald-400 font-medium">{selectedGridCell.resources.join(', ')}</span></div>
                                 )}
                              </div>

                              {selectedGridCell.isRiftPortal && (
                                 <button 
                                    onClick={() => {
                                       if (selectedGridCell.targetRiftZ !== undefined) {
                                          setCurrentPlaneZ(selectedGridCell.targetRiftZ);
                                          handleMoveGridStep(0, 0, selectedGridCell.targetRiftZ);
                                       }
                                    }}
                                    className="w-full py-2.5 bg-fuchsia-600 hover:bg-fuchsia-500 text-white rounded-xl text-xs font-bold shadow-md flex items-center justify-center gap-1.5 transition-transform hover:scale-105"
                                 >
                                    <Sparkles size={14}/> Bước Vào Cổng Hư Không (Sang tầng z={selectedGridCell.targetRiftZ})
                                 </button>
                              )}

                              {(() => {
                                 const curX = gameState.coordinates?.x ?? 0;
                                 const curY = gameState.coordinates?.y ?? 0;
                                 const curZ = gameState.coordinates?.z ?? currentPlaneZ;
                                 const isHere = curX === selectedGridCell.x && curY === selectedGridCell.y && curZ === selectedGridCell.z;
                                 const distLi = getGridDistanceLi({ x: curX, y: curY, z: curZ }, selectedGridCell);

                                 if (isHere) {
                                    return <div className="p-3 bg-emerald-950/40 border border-emerald-500/50 rounded-xl text-center text-xs font-bold text-emerald-400">✓ Bạn đang đứng tại ô này</div>;
                                 }

                                 return (
                                    <button
                                       onClick={() => handleMoveGridStep(selectedGridCell.x - curX, selectedGridCell.y - curY, selectedGridCell.z)}
                                       className="w-full py-3 bg-emerald-600 hover:bg-emerald-500 text-white rounded-xl text-xs font-bold uppercase tracking-wider shadow-lg transition-transform hover:scale-105"
                                    >
                                       Bước Đến Ô Này ({distLi} dặm)
                                    </button>
                                 );
                              })()}
                           </div>
                        );
                     })() : (
                           <div className="flex flex-col items-center justify-center h-full text-center text-gray-500 italic text-xs">
                              <Crosshair size={36} className="mb-2 opacity-30 text-teal-400"/>
                              Nhấp vào một ô trên bản đồ để xem chi tiết tình báo hoặc di chuyển bước chân.
                           </div>
                        )}
                     </div>
                  </div>
               )}

               {/* TAB 3: CANON & SỔ SỬ KÝ THẾ GIỚI (CHỈ ADMIN MỚI ĐƯỢC XEM) */}
               {mapActiveTab === 'canon' && isAdminLoggedIn && (
                  <div className="flex-1 flex flex-col lg:flex-row overflow-hidden h-full p-4 md:p-6 gap-6 bg-[var(--bg-body)]">
                     {/* Cột Trái: Dòng Thời Gian Nguyên Tác (Canon) */}
                     <div className="flex-1 bg-[var(--bg-surface)] p-4 md:p-5 rounded-2xl border border-[var(--border-main)] flex flex-col overflow-hidden">
                        <h4 className="text-sm font-bold text-purple-400 uppercase tracking-widest mb-3 flex items-center gap-2 border-b border-gray-800 pb-2">
                           <Clock size={16}/> Dòng Thời Gian Nguyên Tác (Canon Events)
                        </h4>
                        <div className="flex-1 overflow-y-auto custom-scrollbar space-y-3 pr-2">
                           {(gameState.canonTimeline || DEFAULT_CANON_TIMELINE).map(evt => (
                              <div key={evt.id} className={`p-3.5 rounded-xl border transition-all ${evt.hasDiverged ? 'bg-orange-950/20 border-orange-500/60' : 'bg-[var(--bg-body)] border-gray-800'}`}>
                                 <div className="flex justify-between items-start mb-1.5 flex-wrap gap-1">
                                    <span className="text-xs font-bold text-white">{evt.title}</span>
                                    <span className={`text-[11px] leading-5 font-bold px-1.5 py-0.5 rounded border uppercase ${evt.hasDiverged ? 'bg-orange-900/40 text-orange-300 border-orange-500/50' : 'bg-emerald-950/40 text-emerald-400 border-emerald-600/40'}`} style={{ letterSpacing: '0.03em' }}>
                                       {evt.hasDiverged ? '⚡ Đã Phân Kỳ' : '🟢 Định Sẵn'}
                                    </span>
                                 </div>
                                 <div className="text-[10px] text-gray-400 font-mono mb-2">
                                    Năm thứ {evt.triggerYear}, Tháng {evt.triggerMonth} • {FIXED_WORLD_GRAPH[evt.locationId]?.name || evt.locationId}
                                 </div>
                                 {(evt.divergenceLevel || evt.chainId || evt.disasterTag) && (
                                    <div className="text-[10px] font-mono mb-2 text-orange-300/80">
                                       {evt.divergenceLevel ? `Mức phân kỳ: ${evt.divergenceLevel}/3` : ''}
                                       {evt.chainId ? ` • Chuỗi: ${evt.chainId}` : ''}
                                       {evt.disasterTag ? ` • Kiếp: ${evt.disasterTag}` : ''}
                                    </div>
                                 )}
                                 <p className="text-xs text-gray-300 leading-relaxed italic">
                                    "{evt.canonDescription}"
                                 </p>
                                 {evt.hasDiverged && evt.divergenceLog && (
                                    <p className="text-xs text-orange-400 font-medium mt-2 pt-2 border-t border-orange-900/40">
                                       Biến số can thiệp: {evt.divergenceLog}
                                    </p>
                                 )}
                              </div>
                           ))}
                        </div>
                     </div>

                     {/* Cột Phải: Sổ Sử Ký Thế Giới (Ledger) */}
                     <div className="flex-1 bg-[var(--bg-surface)] p-4 md:p-5 rounded-2xl border border-[var(--border-main)] flex flex-col overflow-hidden">
                        <h4 className="text-sm font-bold text-orange-400 uppercase tracking-widest mb-3 flex items-center gap-2 border-b border-gray-800 pb-2">
                           <BookOpen size={16}/> Sổ Sử Ký Thế Giới (Phân Kỳ Nhân Quả)
                        </h4>
                        <div className="flex-1 overflow-y-auto custom-scrollbar space-y-3 pr-2">
                           {(!gameState.worldHistoryLedger || gameState.worldHistoryLedger.length === 0) ? (
                              <div className="text-center py-16 text-gray-500 italic text-xs">
                                 Chưa có biến cố nào bị người chơi can thiệp bẻ cong.<br/>Cửu Châu Giới vẫn vận hành đúng theo kịch bản nguyên tác!
                              </div>
                           ) : (
                              gameState.worldHistoryLedger.map((ledger, idx) => (
                                 <div key={idx} className="p-3.5 bg-[var(--bg-body)] rounded-xl border border-orange-900/40 hover:border-orange-500/50 transition-colors">
                                    <div className="flex justify-between items-center mb-1">
                                       <span className="text-xs font-bold text-orange-300">{ledger.title}</span>
                                       <span className="text-[9px] font-mono text-gray-500">Năm {ledger.year}, Thg {ledger.month}</span>
                                    </div>
                                    <div className="text-[10px] text-gray-400 mb-1.5">{ledger.locationName}</div>
                                    {(ledger.divergenceLevel !== undefined || ledger.aftermathType) && (
                                       <div className="text-[10px] font-mono text-orange-300/80 mb-1.5">
                                          {ledger.divergenceLevel !== undefined ? `Mức phân kỳ: ${ledger.divergenceLevel}/3` : ''}
                                          {ledger.aftermathType ? ` • Aftermath: ${ledger.aftermathType}` : ''}
                                       </div>
                                    )}
                                    <p className="text-xs text-gray-300 leading-relaxed">
                                       {ledger.description}
                                    </p>
                                 </div>
                              ))
                           )}
                        </div>
                     </div>
                  </div>
               )}

            </div>
          </div>
        </div>
      )}
