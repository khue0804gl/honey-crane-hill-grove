          let hasChanges = false;

              if (!q.target || q.target <= 0) return q;
              const allItems = [...(gameState.inventory || [])];
              (gameState.companions || []).forEach(c => { if (c.inventory) allItems.push(...c.inventory); });
              let totalCount = 0;
              const qText = `${q.name} ${q.desc}`.toLowerCase();
              
              allItems.forEach(item => {
                  const iName = item.name.toLowerCase();
                  if (iName.length < 3) return;
                  const cleanItemName = iName.replace(/(hư hỏng|ôi thiu|mốc meo|bị phong ấn)/g, '').trim();
                  
                  // V8.58 FIX: Bảo vệ pháp bảo xịn, cấm quét nhầm tên

                  const qTargetName = qText.replace(/(thu thập|tìm|hái|săn)/g, '').trim();
                  
                  let isMatch = false;
                  if (cleanItemName === qTargetName || cleanItemName === q.name.toLowerCase()) {
                      isMatch = true; 
                  } else if (!isHighRarity) {
                      if (qText.includes(cleanItemName) || cleanItemName.includes(qTargetName)) {
                          isMatch = true;
                      }
                  }

                  if (isMatch) {
                      totalCount += (Number(item.quantity) || 0);
                  }
              });
              if (totalCount > (q.progress || 0)) {
                  hasChanges = true;
                  return { ...q, progress: Math.min(q.target, totalCount) };
              }
              return q;
          });
          if (hasChanges) setGameState(prev => ({ ...prev, quests: syncedQuests }));
      };
