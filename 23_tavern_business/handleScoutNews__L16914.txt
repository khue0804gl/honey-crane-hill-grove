      time: `Vừa xong`,

      title: picked.title,

      type: picked.type as any
    };

      return {

        tavern: {
          ...curTv,
          vault: Math.max(0, (Number(curTv.vault) || 0) - 20),
          news: [newEntry, ...(curTv.news || [])].slice(0, 20),
