          else if (newMyHp <= 0) { endStatusText = `💀 BẠN ĐÃ BỊ ĐÁNH BẠI! Giáng cấp bảng xếp hạng.`; resolveElo = true; amIWinner = false; }
          else if (newOppHp <= 0) { endStatusText = `🏆 CHIẾN THẮNG! Đoạt lấy Điểm Thiên Đạo của đối thủ!`; resolveElo = true; amIWinner = true; }

          setArenaState(prev => {
              const newLogs = [...prev.logs, { type: 'story', text: data.battleLog }];
              if (data.playerHpChange !== 0 || data.opponentHpChange !== 0) {
                 newLogs.push({ type: 'system', text: `Biến động sinh lực: Phe Ta ${data.playerHpChange||0} HP. | Địch Nhân ${data.opponentHpChange||0} HP.` });
              }
              if (endStatusText) newLogs.push({ type: 'system', text: endStatusText });

              return {
                  ...prev,
                  myHP: newMyHp, myMP: newMyMp, myStamina: newMySta,
                  oppHP: newOppHp, oppMP: newOppMp, oppStamina: newOppSta,
                  logs: newLogs,
                  options: (newMyHp > 0 && newOppHp > 0) ? (data.nextOptions || prev.options) : [],
                  isGenerating: false
              };
          });
          
          setTimeout(() => { if (arenaScrollRef.current) arenaScrollRef.current.scrollTop = arenaScrollRef.current.scrollHeight; }, 100);

          // V6.2: Cập Nhật Cướp Ngôi, Thắng Thua và Lịch Sử
          if (resolveElo && user && arenaState.opponentId) {
              try {
                  const myDocId = `${user?.uid}_${arenaState.myProfileId}`;
                  const myRef = doc(db, 'artifacts', ARENA_APP_ID, 'public', 'data', 'ranked_arena', myDocId);
                  const oppRef = doc(db, 'artifacts', ARENA_APP_ID, 'public', 'data', 'ranked_arena', arenaState.opponentId);
                  
                  const mySnap = await getDoc(myRef);
                  let myDataToSave = mySnap.exists() ? mySnap.data() : { rankPosition: 999999, wins: 0, losses: 0, matchHistory: [] };
                  let myCurrentRank = myDataToSave.rankPosition || 999999;
                  let oppRank = arenaState.opponentScore || 999999; 

                  // V6.5: Ghi lại nguyên nhân thắng thua do AI đúc kết
                  const historyLogMe = { time: Date.now(), result: amIWinner ? 'win' : 'loss', opp: oppData.name, summary: data.matchSummary || "Quyết chiến sinh tử." };
                  const historyLogOpp = { time: Date.now(), result: !amIWinner ? 'win' : 'loss', opp: myData.name, summary: data.matchSummary || "Quyết chiến sinh tử." };

                  // Cập nhật Thắng/Thua và Lịch sử cho Mình
                  myDataToSave.wins = (myDataToSave.wins || 0) + (amIWinner ? 1 : 0);
                  myDataToSave.losses = (myDataToSave.losses || 0) + (!amIWinner ? 1 : 0);
                  myDataToSave.matchHistory = [historyLogMe, ...(myDataToSave.matchHistory || [])].slice(0, 10); // Giữ 10 trận gần nhất

                  // Logic Tranh Đoạt (Hoán Đổi Vị Trí)
                  if (amIWinner && oppRank < myCurrentRank) {
                      myDataToSave.rankPosition = oppRank;
                      oppRank = myCurrentRank; // Đẩy địch xuống
                  } else if (!amIWinner && oppRank > myCurrentRank) {
                      myDataToSave.rankPosition = oppRank;
                      oppRank = myCurrentRank; // Kéo mình xuống
                  }

                  await setDoc(myRef, myDataToSave, { merge: true });

                  // Cập nhật cho Đối thủ
                  const oppSnap = await getDoc(oppRef);
                  if (oppSnap.exists()) {
                      const oppDataToSave = oppSnap.data();
                      oppDataToSave.wins = (oppDataToSave.wins || 0) + (!amIWinner ? 1 : 0);
                      oppDataToSave.losses = (oppDataToSave.losses || 0) + (amIWinner ? 1 : 0);
                      oppDataToSave.matchHistory = [historyLogOpp, ...(oppDataToSave.matchHistory || [])].slice(0, 10);
                      oppDataToSave.rankPosition = oppRank;
                      await setDoc(oppRef, oppDataToSave, { merge: true });
                  }
              } catch(err) { console.error("Lỗi cập nhật xếp hạng:", err); }
          }

      } catch (e) {
          setArenaState(prev => ({ ...prev, isGenerating: false, logs: [...prev.logs, { type: 'system', text: 'Thiên Đạo nghẽn mạng, vui lòng thử lại...' }] }));
      }
  };
