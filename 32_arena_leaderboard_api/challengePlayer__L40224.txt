          const oppData = safeDecompress(oppDoc.pvpDataStr);
          if (!oppData) throw new Error("Dữ liệu đối thủ bị hỏng");
          
          const stConfig = getStageConfig(arenaState.mySnapshot.cultivationLevel);
          const myMaxHp = (100 * Math.pow(2, stConfig.id - 1)) + (Number(arenaState.mySnapshot.bonusMaxHp) || 0);

          setArenaState(prev => ({

              tab: 'battle',
              opponentSnapshot: oppData, opponentId: oppDoc.id, opponentScore: oppDoc.rankScore,
              myHP: myMaxHp, myMaxHP: myMaxHp, myMP: myMaxMp, myMaxMP: myMaxMp, myStamina: Number(prev.mySnapshot.stamina) || 100, myMaxStamina: 100,

          }));
