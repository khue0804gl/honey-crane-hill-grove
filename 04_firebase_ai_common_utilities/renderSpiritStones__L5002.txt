const renderSpiritStones = (amount) => {
  const val = Number(amount) || 0;
  if (val < 0) return <span className="text-red-500 font-bold bg-red-900/30 px-2 py-0.5 rounded border border-red-500/50 animate-pulse">NỢ {Math.abs(val).toLocaleString('vi-VN')} LT</span>;
  
  const { cuc, thuong, trung, ha, total } = getStonesBreakdown(amount);
  if (total === 0) return <span className="text-gray-400 font-bold">0 Hạ</span>;
  
  return (
      <div className="flex gap-2 flex-wrap justify-end font-mono text-[13px]">
          {cuc > 0 && <span className="text-red-500 font-bold bg-red-900/10 px-2 py-0.5 rounded border border-red-900/30">{cuc.toLocaleString('vi-VN')} Cực</span>}
          {thuong > 0 && <span className="text-purple-400 font-bold bg-purple-900/10 px-2 py-0.5 rounded border border-purple-900/30">{thuong.toLocaleString('vi-VN')} Thượng</span>}
          {trung > 0 && <span className="text-blue-400 font-bold bg-blue-900/10 px-2 py-0.5 rounded border border-blue-900/30">{trung.toLocaleString('vi-VN')} Trung</span>}
          {ha > 0 && <span className="text-gray-400 font-bold bg-gray-800/30 px-2 py-0.5 rounded border border-gray-700/50">{ha.toLocaleString('vi-VN')} Hạ</span>}
      </div>
  );
};
