const formatEffTime = (m) => {
    if (!m || m <= 0) return "0 phút";
    if (m > 50000000) return "Vĩnh viễn"; // V9.84: Nhận diện thời gian vô hạn của Tâm Pháp
    if (m < 60) return `${Math.floor(m)} phút`;
    if (m < 1440) return `${Math.floor(m/60)} giờ`;
    if (m < 43200) return `${Math.floor(m/1440)} ngày`;
    if (m < 518400) return `${Math.floor(m/43200)} tháng`;
    return `${Math.floor(m/518400)} năm`;
};
