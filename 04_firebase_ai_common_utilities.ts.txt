window.fetch = async (url, options) => {
    // Chỉ đánh chặn các API gọi đến Google Gemini
    if (typeof url === 'string' && url.includes('generativelanguage.googleapis.com')) {
        const now = Date.now();
        let lastError = null;
        let originalPayload = null;

        if (options && options.body) {
            try { originalPayload = JSON.parse(options.body); } catch(e){}
        }

        for (let i = 0; i < AI_MODELS.length; i++) {
            let model = AI_MODELS[i];
            
            // Model đang kiệt sức (bị giới hạn), bỏ qua để dùng Model thấp hơn
            if (now < model.cooldownUntil) continue; 

            const smartUrl = url.replace(/models\/[^:]+:/, `models/${model.name}:`);
            let currentOptions = options;

            if (originalPayload) {
                let currentPayload = { ...originalPayload };
                
                // Tắt bộ lọc an toàn để chống lỗi 403 (Vi phạm thiên lý) do bối cảnh Dark Fantasy máu me
                currentPayload.safetySettings = [
                    { category: "HARM_CATEGORY_HARASSMENT", threshold: "BLOCK_NONE" },
                    { category: "HARM_CATEGORY_HATE_SPEECH", threshold: "BLOCK_NONE" },
                    { category: "HARM_CATEGORY_SEXUALLY_EXPLICIT", threshold: "BLOCK_NONE" },
                    { category: "HARM_CATEGORY_DANGEROUS_CONTENT", threshold: "BLOCK_NONE" }
                ];

                // V1.29.5 & V2.85: Gỡ bỏ hoàn toàn thinkingConfig và hạ trần maxOutputTokens <= 8192 để tránh HTTP 400
                if (currentPayload.generationConfig) {
                    if (currentPayload.generationConfig.thinkingConfig) {
                        delete currentPayload.generationConfig.thinkingConfig;
                    }
                    if (currentPayload.generationConfig.maxOutputTokens && currentPayload.generationConfig.maxOutputTokens > 8192) {
                        currentPayload.generationConfig.maxOutputTokens = 8192;
                    }
                }

                currentOptions = { ...options, body: JSON.stringify(currentPayload) };
            }

            try {
                const response = await originalFetch(smartUrl, currentOptions);
                if (!response.ok) {
                    // V1.29.5: Mở rộng luân chuyển khi gặp 429, 5xx, 403, 404 và cả 400 (do model không hỗ trợ param/quota)
                    if (response.status === 429 || response.status >= 500 || response.status === 403 || response.status === 404 || response.status === 400) {
                        model.cooldownUntil = now + 60000; 
                        lastError = new Error(`HTTP ${response.status}: Model ${model.name} gặp sự cố.`);
                        console.warn(`[Đại Đạo Luân Chuyển] ${model.name} thất bại (Mã: ${response.status}). Chuyển xuống Model dự phòng tiếp theo...`);
                        continue; 
                    } else {
                        throw new Error(`Thiên Đạo từ chối (Mã ${response.status}): Lời nói của ngươi vi phạm thiên lý hoặc quá dài, hãy nói lại!`);
                    }
                }
                return response; // Thành công thì trả về luôn cho game
            } catch (error) {
                if (error.name === 'AbortError') throw error; // Không bắt lỗi người dùng chủ động ngắt
                lastError = error;
            }
        }
        throw lastError || new Error("Toàn bộ Đám Mây Đa Vũ Trụ đều sụp đổ!");
    }
    
    // Các lệnh fetch khác (như Firebase) chạy bình thường không can thiệp
    return originalFetch(url, options); 
};

const formatLargeNumber = (num) => {
  if (num === undefined || num === null || isNaN(num) || !isFinite(num)) return '0';
  const absNum = Math.abs(num);
  if (absNum < 1000) return Math.floor(num).toLocaleString('vi-VN');

  let formatted = '';
  if (absNum >= 1e18) {
    const tyTy = (absNum / 1e18).toFixed(2);
    formatted = `${tyTy} Tỷ Tỷ`;
  } else if (absNum >= 1e15) {
    const trTy = (absNum / 1e15).toFixed(2);
    formatted = `${trTy} Triệu Tỷ`;
  } else if (absNum >= 1e12) {
    const nghinTy = (absNum / 1e12).toFixed(2);
    formatted = `${nghinTy} Nghìn Tỷ`;
  } else if (absNum >= 1e9) {
    const ty = Math.floor(absNum / 1e9);
    const tr = Math.floor((absNum % 1e9) / 1e8);
    formatted = tr > 0 ? `${ty}Tỷ ${tr}` : `${ty}Tỷ`;
  } else if (absNum >= 1e6) {
    const tr = Math.floor(absNum / 1e6);
    const k = Math.floor((absNum % 1e6) / 1e5);
    formatted = k > 0 ? `${tr}Tr ${k}` : `${tr}Tr`;
  } else if (absNum >= 1e3) {
    const k = Math.floor(absNum / 1e3);
    const h = Math.floor((absNum % 1e3) / 1e2);
    formatted = h > 0 ? `${k}N ${h}` : `${k}N`;
  }
  return num < 0 ? `-${formatted}` : formatted;
};

    if (m < 43200) return `${Math.floor(m/1440)} ngày`;
    if (m < 518400) return `${Math.floor(m/43200)} tháng`;
    return `${Math.floor(m/518400)} năm`;
};

const applyWordAliases = (text, aliasesText, isEnabled = true) => {
    if (!text || typeof text !== 'string') return text;
    let processedText = filterRestrictedNames(text);
    if (!isEnabled || !aliasesText || typeof aliasesText !== 'string') return processedText;
    try {
        const lines = aliasesText.split('\n');
        lines.forEach(line => {
            const parts = line.split('-');
            if (parts.length >= 2) {
                const originalPart = parts[0].trim();
                const replacement = parts.slice(1).join('-').trim();
                if (originalPart && replacement) {
                    let wordsToReplace = [];
                    // Hỗ trợ cả định dạng (từ 1, từ 2) lẫn từ 1, từ 2 không cần ngoặc đơn
                    const cleanOriginal = (originalPart.startsWith('(') && originalPart.endsWith(')'))
                        ? originalPart.slice(1, -1)
                        : originalPart;
                    wordsToReplace = cleanOriginal.split(',').map(w => w.trim().replace(/^\(|\)$/g, '')).filter(w => w);
                    
                    // Sắp xếp từ dài nhất lên trước để tránh việc từ ngắn nuốt từ dài (VD: "cậu ta" trước "cậu")
                    wordsToReplace.sort((a, b) => b.length - a.length);

                    wordsToReplace.forEach(word => {
                        if (!word) return;
                        const escapeRegExp = (str) => str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
                        const regex = new RegExp(escapeRegExp(word), 'gi'); // Dùng 'gi' để tìm cả chữ hoa lẫn chữ thường
                        
                        processedText = processedText.replace(regex, (match) => {
                            const firstChar = match.charAt(0);
                            // Kiểm tra xem chữ cái đầu tiên của từ bị thay thế (trên nội dung) đang là viết hoa hay viết thường
                            if (firstChar === firstChar.toUpperCase() && firstChar !== firstChar.toLowerCase()) {
                                // Tự động viết hoa chữ cái đầu cho từ thay thế
                                return replacement.charAt(0).toUpperCase() + replacement.slice(1);
                            } else {
                                // Chuyển từ thay thế về chữ thường (nếu từ gốc đang là chữ thường)
                                return replacement.charAt(0).toLowerCase() + replacement.slice(1);
                            }
                        });
                    });
                }
            }
        });
        return processedText;
    } catch(e) {
        return text;
    }
};

    const descNoMark = removeAccents(item.desc || '');
    const nameStr = cleanName.toLowerCase();
    
    const isCurrency = (
        nameNoMark.includes('linh thach') || 
        nameNoMark.includes('kim te') || 
        nameNoMark.match(/^(ha|trung|thuong|cuc)\s*pham$/) ||
        (nameNoMark.includes('ha pham') && descNoMark.includes('linh thach')) ||
        (nameNoMark.includes('trung pham') && descNoMark.includes('linh thach'))
    ) && 
    !nameNoMark.includes('quang') && 
    !nameNoMark.includes('mo ') && 
    !nameNoMark.includes('ban do') && 
    !nameNoMark.includes('tinh tuy') &&
    !nameNoMark.includes('manh');
                   
    if (isCurrency) {
                  let multiplier = 1;
                  if (nameNoMark.includes('cuc')) multiplier = 1000000;
                  else if (nameNoMark.includes('thuong')) multiplier = 10000;
                  else if (nameNoMark.includes('trung')) multiplier = 100;
                  
                  const match = nameStr.match(/^(\d+)\s*/);
                  const qtyFromName = match ? parseInt(match[1], 10) : 0;
                  const finalQty = Math.max(finalItemQty, qtyFromName);
                  
                  newStones += finalQty * multiplier;
                } else {
                  if (finalItemQty > 0) {
                    const existIdx = newInventory.findIndex(i => i.name === cleanName && i.containerId === item.containerId && i.type === item.type && i.rarity === item.rarity && i.desc === item.desc && !i.durability && !i.attackBonus);
                    if (existIdx >= 0 && item.type !== 'equip') {
                        newInventory[existIdx].quantity += finalItemQty;
                    } else {
                        newInventory.push({ ...item, name: cleanName, quantity: finalItemQty });
                    }
                  }
                }
              });
              return { inventory: newInventory, spiritStones: newStones };
            };

const generateSpoiledName = (name) => {
    const n = String(name).toLowerCase();
    let adj = "hư hỏng";
    if (/(thịt|cá|huyết|xác|canh|cháo|nướng)/.test(n)) {
        const adjs = ["ôi thiu", "bốc mùi thối", "rữa nát", "đầy dòi bọ", "chảy nước vàng"];
        adj = adjs[Math.floor(Math.random() * adjs.length)];
    } else if (/(bánh|cơm|mì|lương khô|đậu|hạt)/.test(n)) {
        const adjs = ["mốc meo xanh rờn", "ôi chua", "cứng như đá", "nổi mốc trắng", "bị mọt ăn rỗng"];
        adj = adjs[Math.floor(Math.random() * adjs.length)];
    } else if (/(đan|dược|dịch|hoàn|tán|tửu|cao|ma dược|potion)/.test(n)) {
        const adjs = ["tiêu tán dược tính", "phế thải", "đóng cặn độc", "biến chất đục ngầu", "tỏa hắc khí"];
        adj = adjs[Math.floor(Math.random() * adjs.length)];
    } else if (/(thảo|quả|sâm|hoa|nấm|táo|linh chi|trái)/.test(n)) {
        const adjs = ["héo rũ", "thối rữa", "chảy nước đen", "mất hết linh khí", "nhũn nát"];
        adj = adjs[Math.floor(Math.random() * adjs.length)];
    } else {
        const adjs = ["biến chất", "mục nát", "đã hư hại", "nổi mốc meo"];
        adj = adjs[Math.floor(Math.random() * adjs.length)];
    }
    const cleanName = String(name).replace(/\s*\([^)]*\)\s*/g, '').trim();
    return `${cleanName} (${adj})`;
};

const getItemUnitWeight = (item) => {
    if (!item || typeof item !== 'object') return 0;
    const name = String(item.name || '').toLowerCase();
    const nNoMark = name.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    
    // V8.40 FIX: Linh thạch là tiền tệ, sẽ tự động hóa thành số dư nên không chiếm dung lượng vật lý trong lúc tính toán
    if ((nNoMark.includes('linh thach') || nNoMark.includes('kim te') || nNoMark.match(/^(ha|trung|thuong|cuc)\s*pham$/)) && !nNoMark.includes('quang') && !nNoMark.includes('mo ') && !nNoMark.includes('ban do') && !nNoMark.includes('tinh tuy') && !nNoMark.includes('manh')) {
        return 0; 
    }
    
    // Rất nhẹ (10g - 50g): Giấy tờ, bùa chú, đan dược, linh thảo
    if (name.match(/(đan|dịch|thuốc|bùa|phù|giấy|thư|ngọc giản|tơ|lông|chỉ)/)) return 0.05;
    if (name.match(/(cỏ|thảo|lá|hoa|quả|trái|sâm|nấm|tủy|phấn)/)) return 0.1;
    
    // Nhẹ (100g - 200g): Đồ ăn nhỏ, trang sức, hồn tinh
    if (name.match(/(bánh|thịt|cơm|nhẫn|túi|vòng|trâm|ngọc bội|hồn tinh|yêu đan)/)) return 0.2;

    if (item.type === 'equip') {
        // Vũ khí hạng nặng (15kg - 50kg+)
        if (name.match(/(trọng kiếm|búa|rìu|kích|chùy|cự|đỉnh|lò|chuông|bia|cột|trụ)/)) return 30.0; 
        // Phòng cụ nặng
        if (name.match(/(giáp|khiên|thuẫn|áo thiếc|xích)/)) return 8.0;
        // Vũ khí thường (Cầm tay)
        if (name.match(/(kiếm|đao|thương|cung|trượng|côn|roi|quạt)/)) return 3.0; 
        // Y phục nhẹ
        if (name.match(/(áo|quần|đạo bào|giày|mũ|hài|khố|choàng|khăn)/)) return 0.5; 
        return 2.0;
    }
    
    // Vật liệu thô (Đá, sắt, quặng, gỗ)
    if (name.match(/(đá|thạch|quặng|sắt|đồng|kim loại|thép|gỗ|thiết|khối|ngọc thạch)/)) return 15.0;

    if (item.type === 'demon_soul') return 0.1;
    if (['spatial', 'sealed_spatial', 'soul_spatial'].includes(item.type)) return 0.1;
    if (item.type === 'food') return 0.5;
    
    return 0.5; // Mặc định cho các thứ khác
};
