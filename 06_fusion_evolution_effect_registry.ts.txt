// V1.45: BẢNG ÁNH XẠ PHÔI DUNG HỢP CỨNG (FUSION_INGREDIENTS_MAP)
// Thu hồi triệt để khí vận phôi đỏ ban đầu khi đã kích hoạt Thần thể dung hợp (Theo Góp Ý HOENGGG)
// ============================================================================
export const getRecursiveFusionIngredients = (fusedName: string): string[] => {
    if (!fusedName || typeof MASTER_FUSION_MATRIX === 'undefined') return [];
    const fLow = String(fusedName).toLowerCase().trim();
    const directRecipe = MASTER_FUSION_MATRIX.find(r => r.name.toLowerCase() === fLow);
    if (!directRecipe) return [];
    const result = new Set<string>();
    const queue = [
        ...directRecipe.ingredients,
        ...(directRecipe.alternativeIngredients ? directRecipe.alternativeIngredients.flat() : [])
    ];
    while (queue.length > 0) {
        const ing = queue.shift()!;
        const ingLow = ing.toLowerCase().trim();
        if (!result.has(ingLow)) {
            result.add(ingLow);
            const subRecipe = MASTER_FUSION_MATRIX.find(r => r.name.toLowerCase() === ingLow);
            if (subRecipe) {
                queue.push(...subRecipe.ingredients);
                if (subRecipe.alternativeIngredients) {
                    queue.push(...subRecipe.alternativeIngredients.flat());
                }
            }
        }
    }
    return Array.from(result);
};

export const resolveCanonicalEffect = (inputNameOrId: string, currentStageId: number = 1, bgId: string = '') => {
  if (!inputNameOrId) return null;
  const raw = String(inputNameOrId).trim().toLowerCase();
  
  if (MASTER_EFFECTS_REGISTRY[raw]) {
    const def = MASTER_EFFECTS_REGISTRY[raw];
    const isImmune = def.tier === 1 && def.type !== 'buff' && (currentStageId >= 3 || ['bg_20', 'bg_21', 'bg_22', 'bg_ctan', 'bg_undead'].includes(bgId));
    return { effect: def, isImmune };
  }
  
  for (const k in MASTER_EFFECTS_REGISTRY) {
    const def = MASTER_EFFECTS_REGISTRY[k];
    const dName = def.name.toLowerCase();
    if (dName === raw || raw.includes(dName) || dName.includes(raw)) {
      const isImmune = def.tier === 1 && def.type !== 'buff' && (currentStageId >= 3 || ['bg_20', 'bg_21', 'bg_22', 'bg_ctan', 'bg_undead'].includes(bgId));
      return { effect: def, isImmune };
    }
  }
  
  return null;
};
