                            {(statHistoryModal.activeRecord.breakdown && statHistoryModal.activeRecord.breakdown.length > 0) ? (

'bg-emerald-900/30 text-emerald-400 border-emerald-500/50' : 'bg-red-900/30 text-red-400 border-red-500/50'}`}>

                    <div className="space-y-4 animate-in slide-in-from-left-4">
                        {statHistoryModal.introText && (

                                {statHistoryModal.introText}

                                Chưa có ghi chép lịch sử nào.

                            <div key={i} className="bg-[var(--bg-surface)] p-4 rounded-xl border border-[var(--border-main)] flex justify-between items-center group hover:border-teal-500/50 transition-all">
                                <div className="flex-1 pr-4">
                                    <div className="text-[10px] font-mono text-[var(--text-muted)] mb-1.5">{String(h.time || '')}</div>

                          const idx = newInv.findIndex(i => i.name === discardItemModal.name && i.containerId === discardItemModal.containerId);

                          } else {
                              const idx = newInv.findIndex(i => i.name === discardItemModal.name && i.containerId === discardItemModal.containerId);
                              if (idx >= 0) {
                                  const itemToDrop = { ...newInv[idx], quantity: discardAmount, containerId: 'ground' };
                                  newInv[idx].quantity -= discardAmount;
                                  if (newInv[idx].quantity <= 0) newInv.splice(idx, 1);

                          {gameState.gameMode !== 'mong_ao' && (

                  {/* THANH TÌM KIẾM VÀ SẮP XẾP KHÍ VẬN */}

                      {(() => {

                          if (traitSearch.trim() !== '') {
                              const query = traitSearch.toLowerCase().trim();
                              if (/^-?\d+$/.test(query)) {

                                  displayTraits = displayTraits.filter(t => Math.abs(t.cost || 0) === numQuery);
                              } else {
                                  const stripAccents = (s: string) => (s || "").normalize("NFD").replace(/[̀-ͯ]/g, "").replace(/đ/g, "d").replace(/Đ/g, "D").toLowerCase();
                                  const normVi = (s: string) => (s || "").toLowerCase().normalize("NFC").replace(/đoạ/g, "đọa").replace(/hoà/g, "hòa").replace(/uỷ/g, "ủy");
                                  const qNorm = normVi(query);
                                  const qStrip = stripAccents(query);

                                      const dNorm = normVi(t.desc);
                                      const nStrip = stripAccents(t.name);

                              return (

'bg-[var(--bg-body)] border-gray-700/50 opacity-60 cursor-not-allowed grayscale' : 'bg-[var(--bg-surface)] border-[var(--border-main)] hover:border-gray-500 cursor-pointer'}`}>

                                  {/* V7.87 FIX: Giao diện Xích Chéo & Ổ Khóa */}
                                  {isLocked && (
                                      <div className="absolute inset-0 z-20 bg-gray-950/75 flex items-center justify-center rounded-xl overflow-hidden">
                                          <div className="absolute w-[150%] h-1 bg-gradient-to-r from-transparent via-gray-400 to-transparent rotate-45 opacity-40 shadow-[0_0_5px_#000]"></div>
                                          <div className="absolute w-[150%] h-1 bg-gradient-to-r from-transparent via-gray-400 to-transparent -rotate-45 opacity-40 shadow-[0_0_5px_#000]"></div>
                                          <div className="bg-[var(--bg-body)] p-2.5 rounded-full border border-gray-600 shadow-[0_0_15px_rgba(0,0,0,0.8)] z-10" title="Xung khắc với Khí vận đã chọn">
                                              <Lock size={24} className="text-gray-400"/>
                                          </div>
                                      </div>
                                  )}

                                  <div className="flex justify-between items-start mb-2 relative z-10">
                                      <div className="flex items-center gap-1.5">
                                          <div className={`font-bold text-base ${RARITY_COLORS[trait.rarity]?.text || 'text-gray-400'}`}>{trait.name}</div>
                                          <button onClick={(e) => { e.stopPropagation(); showAlert(`CHI TIẾT KHÍ VẬN: ${trait.name}\n\nPhẩm chất: ${trait.rarity}\nTiêu hao: ${trait.cost > 0 ? '-' + trait.cost : '+' + Math.abs(trait.cost)} ${'Điểm Luân Hồi'}\n\nMiêu tả chi tiết:\n${trait.desc}`); }} className="text-[var(--text-muted)] hover:text-purple-400 transition-colors p-0.5" title="Xem chi tiết"><Info size={16}/></button>
                                      </div>
                                      <div className={`font-mono text-xs font-bold ${trait.cost > 0 ? 'text-red-500' : 'text-emerald-500'}`}>{trait.cost > 0 ? `-${trait.cost}` : `+${Math.abs(trait.cost)}`}</div>
                                  </div>
                                  <div className="relative z-10 mt-auto">
                                      <div className="text-[var(--text-muted)] text-sm line-clamp-3">{trait.desc}</div>
                                  </div>

                              );
                          });
                      })()}
                  </div>

      {/* SYSTEM CATEGORY MODAL */}
      {isSystemModalOpen && (
          <div className="fixed inset-0 z-50 flex items-center justify-center bg-[var(--bg-overlay)] p-4">
              <div className="bg-[var(--bg-panel)] w-full max-w-4xl max-h-[80vh] rounded-2xl flex flex-col border border-red-500/50 shadow-[0_0_30px_rgba(239,68,68,0.3)] relative">
                  <div className="p-5 border-b border-red-900/50 flex justify-between items-center bg-[var(--bg-body)] shrink-0">
                      <h3 className="text-xl font-bold flex items-center gap-2 text-red-500"><Settings className="text-red-500 animate-spin-slow"/> Lựa Chọn Hệ Thống Đồng Hành</h3>

                      Lưu ý: Bạn chỉ được phép kích hoạt DUY NHẤT 1 Hệ thống hỗ trợ cho mỗi kiếp luân hồi. {'Điểm Luân Hồi'} sẽ thay đổi tương ứng.
                  </div>

                      {TRAITS_POOL.filter(t => t.isSystemChild).map(sys => (
                          <div key={sys.id} onClick={() => selectSystemChild(sys.id)} className="p-5 rounded-xl border border-red-900/50 bg-[var(--bg-body)] hover:border-red-500 hover:bg-red-900/20 cursor-pointer transition-all shadow-md group">
                              <div className="flex justify-between items-start mb-3">
                                  <div className="font-bold text-lg text-red-400 group-hover:text-red-300">{sys.name}</div>

      {isWeaponModalOpen && (
          <div className="fixed inset-0 z-50 flex items-center justify-center bg-[var(--bg-overlay)] p-4">
              <div className="bg-[var(--bg-panel)] w-full max-w-4xl max-h-[80vh] rounded-2xl flex flex-col border border-yellow-500/50 shadow-[0_0_30px_rgba(234,179,8,0.3)] relative">
                  <div className="p-5 border-b border-yellow-900/50 flex justify-between items-center bg-[var(--bg-body)] shrink-0">
                      <h3 className="text-xl font-bold flex items-center gap-2 text-yellow-500"><Swords className="text-yellow-500 animate-spin-slow"/> Lựa Chọn Thiên Phú Binh Khí</h3>
                      <button onClick={() => setIsWeaponModalOpen(false)} className="p-2 hover:bg-[var(--bg-surface)] rounded text-[var(--text-muted)]"><X size={20}/></button>
                  </div>
                  <div className="p-4 text-center bg-yellow-900/10 border-b border-yellow-900/30 text-sm text-yellow-300 italic">
                      Lưu ý: Bạn chỉ được phép chọn DUY NHẤT 1 loại Binh Khí Bổn Mạng. {'Điểm Luân Hồi'} sẽ thay đổi tương ứng.
                  </div>
                  <div className="flex-1 overflow-y-auto p-5 custom-scrollbar grid grid-cols-1 md:grid-cols-2 gap-4 content-start bg-[var(--bg-body)]">
                      {TRAITS_POOL.filter(t => t.isWeaponChild).map(sys => (
                          <div key={sys.id} onClick={() => selectWeaponChild(sys.id)} className="p-5 rounded-xl border border-yellow-900/50 bg-[var(--bg-body)] hover:border-yellow-500 hover:bg-yellow-900/20 cursor-pointer transition-all shadow-md group">
                              <div className="flex justify-between items-start mb-3">
                                  <div className="font-bold text-lg text-yellow-400 group-hover:text-yellow-300">{sys.name}</div>
                                  <div className="font-mono text-sm font-bold text-yellow-500 bg-yellow-900/30 px-2 py-0.5 rounded border border-yellow-500/30">-{sys.cost} Điểm</div>
                              </div>
                              <div className="text-[var(--text-main)] text-sm leading-relaxed whitespace-pre-wrap">
                                  {sys.desc}
                              </div>
                          </div>
                      ))}
                  </div>
              </div>
          </div>
      )}

      {/* BACKGROUND MODAL */}
      {isBgSelectOpen && (
          <div className="fixed inset-0 z-50 flex items-center justify-center bg-[var(--bg-overlay)] p-4">
              <div className="bg-[var(--bg-panel)] w-full max-w-4xl max-h-[80vh] rounded-2xl flex flex-col border border-[var(--border-main)] shadow-2xl relative">
                  <div className="p-5 border-b border-[var(--border-main)] flex justify-between items-center shrink-0">
                      <h3 className="text-xl font-bold flex items-center gap-2 text-[var(--text-main)]"><Map className="text-blue-500"/> Chọn Nơi Sinh</h3>
                      <button onClick={() => setIsBgSelectOpen(false)} className="p-2 hover:bg-[var(--bg-surface)] rounded text-[var(--text-muted)]"><X size={20}/></button>
                  </div>

                  {/* THANH TÌM KIẾM VÀ SẮP XẾP NƠI SINH / XUẤT THÂN */}
                  <div className="p-4 border-b border-[var(--border-main)] bg-[var(--bg-body)] flex flex-col sm:flex-row gap-3 shrink-0">
                      <div className="relative flex-1">
                          <Search size={18} className="absolute left-3 top-1/2 -translate-y-1/2 text-[var(--text-muted)]" />
                          <input 
                              type="text" 
                              placeholder="Tìm tên nơi sinh, xuất thân, lộ trình..." 
                              value={bgSearch} 
                              onChange={e => setBgSearch(e.target.value)} 
                              className="w-full bg-[var(--bg-surface)] border border-[var(--border-main)] rounded-xl pl-10 pr-10 py-2.5 text-sm text-[var(--text-main)] focus:outline-none focus:border-blue-500 transition-colors" 
                          />
                          {bgSearch && (
                              <button onClick={() => setBgSearch('')} className="absolute right-3 top-1/2 -translate-y-1/2 text-[var(--text-muted)] hover:text-[var(--text-main)] p-1">
                                  <X size={16} />
                              </button>
                          )}
                      </div>
                      <select 
                          value={bgSort} 
                          onChange={e => setBgSort(e.target.value)} 
                          className="bg-[var(--bg-surface)] text-[var(--text-main)] border border-[var(--border-main)] rounded-xl px-3 py-2.5 text-sm font-bold focus:outline-none focus:border-blue-500 cursor-pointer shrink-0"
                      >
                          <option value="default">Sắp xếp: Mặc định</option>
                          <option value="cost_desc">Sắp xếp: Điểm giảm dần (Cao -{'>'} Thấp)</option>
                          <option value="cost_asc">Sắp xếp: Điểm tăng dần (Thấp -{'>'} Cao)</option>
                          <option value="name_asc">Sắp xếp: Tên A -{'>'} Z</option>
                      </select>
                  </div>

                  <div className="flex-1 overflow-y-auto p-5 custom-scrollbar grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4 content-start">
                      {(() => {
                          let displayBgs = [...BACKGROUNDS_POOL];
                          if (bgSearch.trim() !== '') {
                              const q = bgSearch.toLowerCase().trim();
                              if (/^-?\d+$/.test(q)) {
                                  const num = parseInt(q, 10);
                                  displayBgs = displayBgs.filter(b => b.cost === num || Math.abs(b.cost || 0) === Math.abs(num));
                              } else {
                                  displayBgs = displayBgs.filter(b => 
                                      String(b.name).toLowerCase().includes(q) ||
                                      String(b.pro || '').toLowerCase().includes(q) ||
                                      String(b.con || '').toLowerCase().includes(q) ||
                                      String(b.hidden || '').toLowerCase().includes(q)
                                  );
                              }
                          }
                          if (bgSort === 'cost_desc') {
                              displayBgs.sort((a, b) => (b.cost || 0) - (a.cost || 0));
                          } else if (bgSort === 'cost_asc') {
                              displayBgs.sort((a, b) => (a.cost || 0) - (b.cost || 0));
                          } else if (bgSort === 'name_asc') {
                              displayBgs.sort((a, b) => a.name.localeCompare(b.name, 'vi'));
                          }

                          if (displayBgs.length === 0) {
                              return <div className="col-span-full text-center py-10 text-[var(--text-muted)] italic">Không tìm thấy nơi sinh nào phù hợp.</div>;
                          }

                          return displayBgs.map(bg => {
                              const isSelected = selectedBgId === bg.id;
                              return (
                              <div key={bg.id} onClick={() => {
                                  setSelectedBgId(bg.id);
                                  setIsBgSelectOpen(false);
                                  if (bg.id === 'bg_thien_dao_khuoc_tu') {
                                      setSelectedTraitIds(prev => {
                                          const cleaned = prev.filter(tid => {
                                              const t = TRAITS_POOL.find(x => x.id === tid);
                                              return !isForbiddenThienDaoKhuocTuTrait(t);
                                          });
                                          if (cleaned.length < prev.length) {
                                              showAlert("Xuất thân Thiên Đạo Khước Từ đã tự động thanh lọc và loại bỏ các Khí Vận may mắn / hệ thống và debuff phế phẩm trùng lặp (Thiên Đạo Áp Chế, Thiên Sinh Phế Vật, Thiên Sinh Rác Rưởi)!");
                                          }
                                          return cleaned;
                                      });
                                  } else if (isPureYeuTocBg(bg.id)) {
                                      setSelectedTraitIds(prev => {
                                          const cleaned = prev.filter(tid => {
                                              const t = TRAITS_POOL.find(x => x.id === tid);
                                              return !(tid === 504 || tid === 605 || tid === 803 || tid === 701 || (t && (t.isSystemParent || t.isSystemChild)));
                                          });
                                          if (cleaned.length < prev.length) {
                                              showAlert("Xuất thân Yêu Tộc Thuần Chủng đã tự động loại bỏ các Khí Vận tương khắc (Kẻ Ăn Mày, Ngự Yêu, Hệ Thống)!");
                                          }
                                          return cleaned;
                                      });
                                  }
                              }} className={`p-4 rounded-xl cursor-pointer border transition-all relative flex flex-col ${isSelected ? 'bg-blue-900/20 border-blue-500 shadow-md' : 'bg-[var(--bg-surface)] border-[var(--border-main)] hover:border-blue-400/50'}`}>
                                  {isSelected && <div className="absolute top-0 right-0 bg-blue-500 text-white p-1 rounded-bl-lg"><Check size={14} strokeWidth={4}/></div>}
                                  <div className="flex justify-between items-start mb-2">
                                      <div className="flex items-center gap-1.5">
                                          <div className="font-bold text-base text-[var(--text-bold)]">{bg.name}</div>
                                          <button onClick={(e) => { e.stopPropagation(); showAlert(`CHI TIẾT XUẤT THÂN: ${bg.name}\n\nƯu điểm: ${bg.pro}\nNhược điểm: ${bg.con}\n\nHiệu ứng ngầm / Kịch bản:\n${bg.hidden || 'Không có'}`); }} className="text-[var(--text-muted)] hover:text-blue-400 transition-colors p-0.5" title="Xem chi tiết"><Info size={16}/></button>
                                      </div>
                                      <div className={`font-mono text-xs font-bold ${bg.cost > 0 ? 'text-red-500' : bg.cost < 0 ? 'text-emerald-500' : 'text-[var(--text-muted)]'}`}>{bg.cost > 0 ? `-${bg.cost}` : bg.cost < 0 ? `+${Math.abs(bg.cost)}` : '0'}</div>
                                  </div>
                                  <div className="text-[var(--text-muted)] text-sm mb-1"><span className="text-emerald-500">Ưu:</span> {bg.pro}</div>
                                  <div className="text-[var(--text-muted)] text-sm mb-1"><span className="text-red-500">Nhược:</span> {bg.con}</div>
                                  <div className="text-[var(--text-muted)] text-xs italic opacity-80 line-clamp-2 mt-2">{bg.hidden}</div>
                              </div>
                              );
                          });
                      })()}
                  </div>
              </div>
          </div>
      )}
