// V1.29.6: QUY LUẬT DUYÊN KHỞI & VÔ NGÃ — NĂNG LƯỢNG TƯƠNG SINH TƯƠNG KHẮC 7 HỆ PHÁI
const computePathEnergyInteraction = (cultivationPath, stageId, monthsPassed, actionText, currentAura, currentDemonicQi) => {
    const act = String(actionText || '').toLowerCase();
    const isDevour = /(thôn phệ|hút cạn|cướp đoạt|ma công|hóa công|huyết tế|thao thiết|vạn ác)/.test(act);
    const isPeaceful = /(bế quan|tu luyện|đả tọa|nghỉ ngơi|dưỡng thương|chữa trị|hồi phục|luyện đan|chế phù|vẽ bùa|đọc sách|uống trà|ngủ|tĩnh tâm)/.test(act);

    // Tiêu chuẩn tiêu hao theo cảnh giới (Luyện Khí 2, Trúc Cơ 8, Kim Đan 30, Nguyên Anh 120, Hóa Thần 500)
    let baseRate = 2;
    if (stageId === 1) baseRate = 2;
    else if (stageId === 2) baseRate = 8;
    else if (stageId === 3) baseRate = 30;
    else if (stageId === 4) baseRate = 120;
    else if (stageId === 5) baseRate = 500;
    else baseRate = Math.floor(1500 * Math.pow(2.2, Math.max(0, stageId - 6)));

    let auraChange = 0;
    let demonicQiChange = 0;
    let pollutionChange = 0;
    let pathLog = "";

    const path = cultivationPath || 'tu_tien';
    const effectiveMonths = Math.max(0.1, monthsPassed);

    if (path === 'the_tu') {
        // Thể Tu: Dĩ khí huyết nhục thân vi bản, chỉ tiêu hao vi lượng 20% linh khí thiên địa
        if (isPeaceful) {
            auraChange = -Math.ceil(baseRate * 0.2 * effectiveMonths);
        }
    } else if (path === 'ma_tu') {
        // Ma Tu: Ưu tiên cắn nuốt Ma Khí. Nếu thiếu Ma Khí, cắn nuốt Linh Khí và chuyển hóa thải ra Ma Khí/Trọc Khí
        const totalDemand = Math.ceil(baseRate * (isDevour ? 3 : 1) * effectiveMonths);
        if (currentDemonicQi >= totalDemand) {
            demonicQiChange = -totalDemand;
        } else {
            const auraDrain = Math.min(currentAura, totalDemand);
            auraChange = -auraDrain;
            // Duyên khởi: Ma công thải trọc — 25% linh khí bị biến chất thành ma khí phóng xuất ra môi trường
            demonicQiChange = Math.floor(auraDrain * 0.25);
            pollutionChange = Math.floor(auraDrain * 0.05);
            if (demonicQiChange >= 15) {
                pathLog = `[Ma Khí Tán Phát] Ma công uẩn chuyển đã làm biến chất một phần địa mạch, ma khí tăng thêm +${demonicQiChange}.`;
            }
        }
    } else if (path === 'quy_tu') {
        // Quỷ Tu: Tiêu hao âm khí, nếu ở nơi linh khí thuần dương cao (> 15000) bị áp chế nhẹ
        const demand = Math.ceil(baseRate * 0.8 * effectiveMonths);
        if (currentDemonicQi > 0) {
            demonicQiChange = -Math.min(currentDemonicQi, demand);
        } else {
            auraChange = -Math.ceil(demand * 0.5);
        }
    } else if (path === 'nho_tu') {
        // Nho Tu: Hạo Nhiên Chính Khí tự động tịnh hóa tạp khí và tiêu trừ ma khí xung quanh
        const demand = Math.ceil(baseRate * effectiveMonths);
        auraChange = -demand;
        if (currentDemonicQi > 0) {
            demonicQiChange = -Math.min(currentDemonicQi, Math.ceil(baseRate * 0.5 * effectiveMonths));
        }
    } else if (path === 'phat_tu') {
        // Phật Tu: Công đức tịnh hóa, siêu độ ma khí chuyển hóa thành linh khí lành tính trả về đất trời
        const demand = Math.ceil(baseRate * effectiveMonths);
        auraChange = -demand;
        if (currentDemonicQi > 0) {
            const purifyAmount = Math.min(currentDemonicQi, Math.ceil(baseRate * 0.8 * effectiveMonths));
            demonicQiChange = -purifyAmount;
            // Duyên khởi: Oán niệm tiêu trừ thì đất mẹ hồi sinh, 50% ma khí hóa thành linh khí
            auraChange += Math.floor(purifyAmount * 0.5);
            pathLog = `[Tịnh Hóa Đạo Trường] Phạm âm phổ độ chúng sinh, thanh tẩy -${purifyAmount} Ma Khí và chuyển hóa thành linh khí trả về cho sơn hà!`;
        }
    } else {
        // Tiên Tu / Yêu Tu: Hấp thu linh khí thanh tịnh bình thường
        if (isPeaceful) {
            auraChange = -Math.ceil(baseRate * effectiveMonths);
        }
    }

    return { auraChange, demonicQiChange, pollutionChange, pathLog };
};

// V3.97: TỌA ĐỘ VÀ KHOẢNG CÁCH DẶM THỰC TẾ DỰA TRÊN ĐỒ THỊ CỐ ĐỊNH (FIXED TOPOLOGY GRAPH)
const getLocationCoords = (nameOrId, timelineId) => {
    if (!nameOrId) return { x: 0, y: 0, z: 0 };
    const activeGraph = getActiveWorldGraph(timelineId);
    const node = activeGraph[nameOrId] || Object.values(activeGraph).find(n => n.name === nameOrId);
    if (node) return node.coordinates;
    return { x: 0, y: 0, z: 0 };
};

const getDistance = (loc1, loc2, timelineId) => {
    if (!loc1 || !loc2 || loc1 === loc2) return 0;
    const activeGraph = getActiveWorldGraph(timelineId);
    const node1 = activeGraph[loc1] || Object.values(activeGraph).find(n => n.name === loc1);
    const node2 = activeGraph[loc2] || Object.values(activeGraph).find(n => n.name === loc2);

    if (node1 && node2) {
        // Nếu có tuyến đường nối trực tiếp trong đồ thị, dùng đúng cự ly dặm định sẵn
        const directPath = (node1.connectedPaths || []).find(p => p.targetId === node2.id);
        if (directPath) return directPath.distanceLi;

        // Nếu không nối trực tiếp, tính cự ly không gian thực theo tọa độ
        const dx = node1.coordinates.x - node2.coordinates.x;
        const dy = node1.coordinates.y - node2.coordinates.y;
        return Math.floor(Math.sqrt(dx * dx + dy * dy));
    }
    return 100;
};

const formatTravelTime = (distance, speed) => {
    if (distance === 0) return "Đang ở đây";
    if (speed <= 0) return "Không thể di chuyển";
    const hours = distance / speed;
    if (hours < 0.016) return "Thuấn di (Nháy mắt)";
    if (hours < 1) return `${Math.ceil(hours * 60)} phút`;
    if (hours < 24) return `${Math.floor(hours)} giờ ${Math.ceil((hours%1)*60)} phút`;
    const days = Math.floor(hours / 24);
    const remHours = Math.floor(hours % 24);
    if (days > 365) return `${Math.floor(days/365)} năm ${days%365} ngày`;
    return `${days} ngày ${remHours > 0 ? remHours + ' giờ' : ''}`;
};

      "Thiên Linh Căn (Hỏa)", "Thiên Linh Căn (Thổ)", "Dị Linh Căn (Lôi)",
      "Dị Linh Căn (Băng)", "Dị Linh Căn (Phong)", "Song Linh Căn (Kim, Thủy)",
      "Song Linh Căn (Mộc, Hỏa)", "Thiên Linh Căn Cực Phẩm"
    ];
    let hash = 0;
    const nameStr = String(npc.name || 'NPC');
    for (let i = 0; i < nameStr.length; i++) { hash = (hash << 5) - hash + nameStr.charCodeAt(i); hash |= 0; }
    return highPool[Math.abs(hash) % highPool.length];
  }

  // Ưu tiên 3: Trung cấp (Kim Đan / Nguyên Anh)
  const isMidLevel = /(nguyên anh|kim đan)/i.test(levelStr) || stageId === 4 || stageId === 5;
  if (isMidLevel) {
    let hash = 0;
    const nameStr = String(npc.name || 'NPC');
    for (let i = 0; i < nameStr.length; i++) { hash = (hash << 5) - hash + nameStr.charCodeAt(i); hash |= 0; }
    const roll = Math.abs(hash) % 6;
    const midPool = ["Song Linh Căn (Kim, Thủy)", "Song Linh Căn (Mộc, Hỏa)", "Song Linh Căn (Thổ, Mộc)",
                     "Tam Linh Căn (Kim, Mộc, Thổ)", "Tứ Linh Căn", "Ngũ Linh Căn"];
    return midPool[roll];
  }

  // Logic phân phối: phàm nhân / NPC cấp thấp = Phàm Thể Vô Căn hoặc Tạp Căn
  let hash = 0;
  const nameStr = String(npc.name || 'NPC');
  for (let i = 0; i < nameStr.length; i++) { hash = (hash << 5) - hash + nameStr.charCodeAt(i); hash |= 0; }
  const roll = Math.abs(hash) % 10;
  if (roll < 6) return "Phàm Thể Vô Căn";
  if (roll < 9) return "Tạp Căn (Ngũ Hành Lộn Xộn)";
  return "Ngũ Linh Căn";
};

const extractNpcSpiritualRootFromStory = (npcName: string, textToScan: string, currentNpc: any): string | null => {
  if (!npcName || !textToScan) return null;
  const cleanName = String(npcName).replace(/\s*[\(\[].*?[\)\]]\s*/g, '').trim();
  if (!cleanName || cleanName.length < 2) return null;

  // V2.55 CHỐT CHẶN: PHẠM VI HẸP THEO ĐOẠN (Tránh lây nhiễm toàn bộ Hồng Trần Lục thành Thiên Linh Căn - Báo lỗi Ori)
  const escName = cleanName.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const nameIdx = textToScan.indexOf(cleanName);
  if (nameIdx === -1) return null;

  // Chỉ quét trong cửa sổ cục bộ 200 ký tự xung quanh tên NPC
  const startIdx = Math.max(0, nameIdx - 200);
  const endIdx = Math.min(textToScan.length, nameIdx + cleanName.length + 200);
  const localSnippet = textToScan.slice(startIdx, endIdx);

  // Phải có hành động đúc / tạo / tẩy tủy / ban linh căn TRỰC TIẾP trong phạm vi cục bộ của NPC
  const localActionRegex = new RegExp(`(?:đúc|tạo|đúc tạo|ngưng tụ|khai mở|tẩy tủy|thăng cấp|thăng tiến|tiến hóa|thức tỉnh|chuyển hóa|ban|khôi phục|tái tạo|đúc lại|biến dị|đột biến|thăng hoa|tinh thuần|hóa thành|chuyển biến)\\s+(?:cho|thành|được|lại)?\\s*.*?(?:linh căn|căn cơ)`, 'i');
  if (!localActionRegex.test(localSnippet)) return null;

  // 2. Tìm kiếm tên linh căn cụ thể trong phạm vi cục bộ
  const rootSpecificPatterns = [
    new RegExp(`\\[?${escName}\\]?.*?(?:thành|lên|được|có|ngưng tụ thành|đúc thành)\\s+([A-Za-zÀ-ỹ\\s\\(\\),]+?Linh Căn(?:\\s*\\([A-Za-zÀ-ỹ\\s,]+\\))?)`, 'i'),
    new RegExp(`(?:đúc|tạo|khai mở|ngưng tụ|ban cho|ban)\\s+([A-Za-zÀ-ỹ\\s\\(\\),]+?Linh Căn(?:\\s*\\([A-Za-zÀ-ỹ\\s,]+\\))?)\\s+(?:cho|vào|lên)?\\s*\\[?${escName}\\]?`, 'i'),
    new RegExp(`(?:thành|lên|đạt được|sang|là|ngưng tụ thành|đúc thành)\\s+((?:Thiên|Dị|Song|Tam|Tứ|Ngũ|Cực Phẩm|Hỗn Độn|Thuần Dương|Thái Âm|Tiên Thiên|Huyền Thiên|Bát Hoang|Đơn)?\\s*[A-Za-zÀ-ỹ\\s\\(\\),]+?\\s*Linh Căn(?:\\s*\\([A-Za-zÀ-ỹ\\s,]+\\))?)`, 'i'),
    new RegExp(`((?:Thiên|Dị|Song|Tam|Tứ|Ngũ|Cực Phẩm|Hỗn Độn|Thuần Dương|Thái Âm|Tiên Thiên|Huyền Thiên|Bát Hoang)?\\s*(?:Kim|Mộc|Thủy|Hỏa|Thổ|Lôi|Băng|Phong|Quang|Ám|Độc|Kiếm|Hỗn Độn|Âm|Dương)\\s*Linh Căn(?:\\s*\\([A-Za-zÀ-ỹ\\s,]+\\))?)`, 'i')
  ];

  for (const pat of rootSpecificPatterns) {
    const m = localSnippet.match(pat);
    if (m && m[1]) {
      let extracted = m[1].trim().replace(/\\s+/g, ' ');
      extracted = extracted.replace(/^(?:được|thành|lên|sang|là|ra|cho|một|chiếc|đạo)\\s+/i, '').trim();
      if (/linh căn/i.test(extracted) && extracted.length >= 8 && extracted.length <= 40) {
        return extracted.split(' ').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ');
      }
    }
  }

  // 3. Nếu không bắt được tên chi tiết trong phạm vi cục bộ, dò hệ nguyên tố trong snippet cục bộ
  const elementMatches = [
    { re: /(?:hỏa|lửa|viêm|chúc dung)/i, name: "Thiên Linh Căn (Hỏa)" },
    { re: /(?:lôi|sấm|sét|thiên lôi)/i, name: "Dị Linh Căn (Lôi)" },
    { re: /(?:băng|hàn|tuyết)/i, name: "Dị Linh Căn (Băng)" },
    { re: /(?:thủy|nước|hải|đại dương)/i, name: "Thiên Linh Căn (Thủy)" },
    { re: /(?:mộc|cây|rừng|thảo)/i, name: "Thiên Linh Căn (Mộc)" },
    { re: /(?:kim|kiếm|sắt|vàng)/i, name: "Thiên Linh Căn (Kim)" },
    { re: /(?:thổ|đất|địa|nham)/i, name: "Thiên Linh Căn (Thổ)" },
    { re: /(?:phong|gió)/i, name: "Dị Linh Căn (Phong)" },
    { re: /(?:quang|ánh sáng)/i, name: "Dị Linh Căn (Quang)" },
    { re: /(?:ám|bóng tối|ma)/i, name: "Dị Linh Căn (Ám)" }
  ];

  for (const elem of elementMatches) {
    if (elem.re.test(localSnippet)) {
      return elem.name;
    }
  }

  return null;
};

// Bóc tách tên công pháp gốc và phẩm giai chuẩn xác, chống cắt xén mất tên (Báo lỗi Ethan195)
const parseSkillNameAndTier = (rawName: string, fallbackTier?: string): { baseName: string; tier: string; fullName: string } => {
    let s = String(rawName || '').trim();
    s = s.replace(/nguỵ/gi, 'Ngụy');
    
    // Dạng tiền tố: "Ngụy - Hư Không Nhất Niệm" hoặc "Ngụy: Hư Không Nhất Niệm"
    const prefixMatch = s.match(/^(Ngụy|Phỏng|Phàm|Ngân|Hoàng|Địa|Thiên|Tiên|Thần|Đạo)\s*[\-:]\s*(.+)$/i);
    if (prefixMatch) {
        let t = prefixMatch[1].charAt(0).toUpperCase() + prefixMatch[1].slice(1).toLowerCase();
        if (t.toLowerCase() === 'nguỵ') t = 'Ngụy';
        const b = prefixMatch[2].replace(/\s*[\(\[].*?[\)\]]\s*/g, '').trim();
        return { baseName: b, tier: t, fullName: `${b} - ${t}` };
    }
    
    // Dạng hậu tố: "Hư Không Nhất Niệm - Ngụy" hoặc "Hư Không Nhất Niệm (Ngụy)"
    const suffixMatch = s.match(/^(.+?)\s*[\-:\(\[]\s*(Ngụy|Nguỵ|Phỏng|Phàm|Ngân|Hoàng|Địa|Thiên|Tiên|Thần|Đạo)[\)\]]?$/i);
    if (suffixMatch) {
        const b = suffixMatch[1].replace(/\s*[\(\[].*?[\)\]]\s*/g, '').trim();
        let t = suffixMatch[2].trim();
        if (t.toLowerCase() === 'nguỵ') t = 'Ngụy';
        else t = t.charAt(0).toUpperCase() + t.slice(1).toLowerCase();
        return { baseName: b, tier: t, fullName: `${b} - ${t}` };
    }
    
    // Nếu tên bị rỗng hoặc chỉ còn duy nhất 1 chữ trong 10 tier (như save lỗi của Ethan195)
    if (/^(Ngụy|Nguỵ|Phỏng|Phàm|Ngân|Hoàng|Địa|Thiên|Tiên|Thần|Đạo)$/i.test(s)) {
        let t = s.toLowerCase() === 'nguỵ' ? 'Ngụy' : (s.charAt(0).toUpperCase() + s.slice(1).toLowerCase());
        return { baseName: '', tier: t, fullName: t };
    }
    
    // Tên thông thường không kèm hậu tố tier
    const cleanBase = s.replace(/\s*[\(\[].*?[\)\]]\s*/g, '').trim();
    const t = normalizeSkillTier(fallbackTier);
    return { baseName: cleanBase, tier: t, fullName: `${cleanBase} - ${t}` };
};

    const fullText = `${nameL} ${typeL} ${descL}`;

    // 1. Thần thông / Trận pháp / Chú thuật / Huyễn ảo / Tâm trí / Tinh thần / Hồn
    const isMental = /(thần thông|huyễn|mộng|ảo|chú|nguyền|hồn|đọc tâm|tinh thần|trận pháp|tâm ma|niệm lực|phù chú)/.test(fullText);
    
    // 2. Thân pháp / Tốc độ / Độn thuật / Bộ pháp / Khinh công / Tránh né / Lôi độn / Phong độn
    const isSpeed = /(thân pháp|độn thuật|bộ pháp|khinh công|phi hành|tốc độ|chớp nhoáng|quang độn|lôi độn|phong độn|huyết độn|hư không độn|di hình|hoán vị|quy ảnh|du long)/.test(fullText);
    
    // 3. Hộ thể / Phòng ngự / Ngạnh công / Giáp / Khiên / Kim chung / Bất diệt
    const isDefense = /(hộ thể|phòng ngự|ngạnh công|kim chung|quy giáp|bất diệt|toái ngọc|huyền giáp|kiên bích|hộ thân|hóa kình|chắn|khiên)/.test(fullText);
    
    // 4. Sát phạt / Đao kiếm / Quyền cước / Thương kích / Trảm kích / Công kích
    const isOffensive = /(kiếm|đao|thương|kích|quyền|cước|chưởng|chỉ|trảm|sát|công|phá|liệt|bạo|huyết sát|thôn phệ)/.test(fullText);
    
    // 5. Cân bằng / Toàn diện / Hỗn độn / Căn cơ / Cửu chuyển / Vô cực
    const isAllAround = /(quân bình|toàn diện|hỗn độn|căn cơ|cửu chuyển|vô cực|thái cực|bàn cổ|đại đạo)/.test(fullText);

    let weights = { atk: 1.0, def: 0, spd: 0, spp: 0 };
    let scalingText = '1.0 Công';
    let primaryStat = 'atk';
    let primaryName = 'Công';

    if (isMental) {
        if (isSpeed) {
            weights = { atk: 0.3, def: 0, spd: 0.8, spp: 1.5 };
            scalingText = '1.5 Thức, 0.8 Tốc';
            primaryStat = 'spp';
            primaryName = 'Thần Thức & Tốc';
        } else {
            weights = { atk: 0.5, def: 0, spd: 0, spp: 1.5 };
            scalingText = '1.5 Thức, 0.5 Công';
            primaryStat = 'spp';
            primaryName = 'Thần Thức';
        }
    } else if (isSpeed) {
        if (/(hư không|không gian|thuấn di|thần)/.test(fullText)) {
            weights = { atk: 0.4, def: 0, spd: 1.5, spp: 0.5 };
            scalingText = '1.5 Tốc, 0.5 Thức';
            primaryStat = 'spd';
            primaryName = 'Tốc Độ & Thần Thức';
        } else {
            weights = { atk: 0.5, def: 0, spd: 1.5, spp: 0 };
            scalingText = '1.5 Tốc, 0.5 Công';
            primaryStat = 'spd';
            primaryName = 'Tốc Độ';
        }
    } else if (isDefense) {
        weights = { atk: 0.5, def: 1.5, spd: 0, spp: 0 };
        scalingText = '1.5 Thủ, 0.5 Công';
        primaryStat = 'def';
        primaryName = 'Phòng Thủ';
    } else if (isAllAround && cultivationPath === 'tu_tien') {
        weights = { atk: 1.0, def: 1.0, spd: 1.0, spp: 1.0 };
        scalingText = '1.0 Công, 1.0 Thủ, 1.0 Tốc, 1.0 Thức';
        primaryStat = 'all';
        primaryName = 'Bình Quân Căn Cơ';
    } else if (isOffensive) {
        if (/(kiếm ý|đao ý|ý cảnh|tâm kiếm|ngự kiếm)/.test(fullText)) {
            weights = { atk: 1.4, def: 0, spd: 0.3, spp: 0.6 };
            scalingText = '1.4 Công, 0.6 Thức';
            primaryStat = 'atk';
            primaryName = 'Công & Kiếm Ý';
        } else {
            weights = { atk: 1.5, def: 0, spd: 0.5, spp: 0 };
            scalingText = '1.5 Công, 0.5 Tốc';
            primaryStat = 'atk';
            primaryName = 'Tấn Công';
        }
    } else {
        if (cultivationPath === 'tu_tien') {
            weights = { atk: 1.2, def: 0.4, spd: 0.4, spp: 0.4 };
            scalingText = '1.2 Công, 0.4 Căn Cơ';
            primaryStat = 'atk';
            primaryName = 'Công';
        } else {
            weights = { atk: 1.5, def: 0, spd: 0.5, spp: 0 };
            scalingText = '1.5 Công, 0.5 Tốc';
            primaryStat = 'atk';
            primaryName = 'Tấn Công';
        }
    }

    const sumWeights = weights.atk + weights.def + weights.spd + weights.spp;
    return {
        scalingText,
        weights,
        sumWeights: sumWeights > 0 ? sumWeights : 1.0,
        primaryStat,
        primaryName
    };
};
