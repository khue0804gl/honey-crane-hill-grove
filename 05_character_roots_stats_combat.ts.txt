            }
        }
    }

    const mortalBgs = ['bg_17', 'bg_tieu_nhi', 'bg_lang_y', 'bg_ngu_phu', 'bg_tieu_thuong', 'bg_ca_ki', 'bg_linh_quen', 'bg_no_le', 'bg_tu_tu', 'bg_kiem_no', 'bg_dao_tac', 'bg_vo_su_pham'];
    const isMortalBg = mortalBgs.includes(bgId) || tNames.includes('Kẻ Ăn Mày') || tNames.includes('Thường Dân');
    const isGodBg = ['bg_8', 'bg_9', 'bg_10', 'bg_19', 'bg_20', 'bg_21', 'bg_22', 'bg_23', 'bg_24', 'bg_25', 'bg_long_toc', 'bg_con_bang', 'bg_ho_toc', 'bg_vu_toc', 'bg_chan_thien_menh', 'bg_elf', 'bg_angel', 'bg_blood_angel', 'bg_succubus', 'bg_nightmare', 'bg_undead', 'bg_titan', 'bg_behemoth', 'bg_taotie', 'bg_ctan', 'bg_vien_co_cu_long'].includes(bgId);
    
    if (isMortalBg && Math.random() < 0.90) {
        return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
    }
    if (!isGodBg && !isMortalBg && Math.random() < 0.40) {
        return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
    }

    const countRoll = Math.random();
    const count = countRoll > 0.8 ? 2 : countRoll > 0.4 ? 3 : countRoll > 0.1 ? 4 : 5;
    
    if (count === 2 && Math.random() > 0.8) {
        const el = rares[Math.floor(Math.random()*rares.length)];
        return { name: `Dị Linh Căn (${el})`, elements: [el], purity: 'Biến Dị', color: 'text-purple-400 font-bold' };
    }
    
    const shuffled = [...els].sort(() => 0.5 - Math.random());
    const myEls = shuffled.slice(0, count);
    const prefix = count === 2 ? 'Song' : count === 3 ? 'Tam' : count === 4 ? 'Tứ' : 'Ngũ (Tạp)';
    
    let color = 'text-[var(--text-main)]';
    if (count === 2) color = 'text-blue-400 font-bold';
    else if (count === 3) color = 'text-emerald-500 font-bold';
    else if (count >= 4) color = 'text-[var(--text-muted)] font-medium';

    const calcPurity = count === 2 ? 'Thượng Phẩm' : count === 3 ? 'Trung Phẩm' : 'Kém';
    return { name: `${prefix} Linh Căn (${myEls.join('-')})`, elements: myEls, purity: calcPurity, color };
};

        if (name === 'Trận Sư Chính Thức' || name === 'Bố Trận Sơ Cấp') { pSpp += 10; }

        // Cấp Cam / Tím (Bảo Thể / Linh Thể)
        if (name === 'Phong Ảnh Thể') { pSpd += 40; }
        if (name === 'Vạn Pháp Bất Xâm') { pDef += 40; pSpp += 20; }
        if (name === 'Quỷ Đạo Truyền Nhân') { pSpp += 50; pAtk += 15; pDef -= 10; }
        if (name === 'Tiên Phong Đạo Cốt') { pSpp += 20; pDef += 10; }
        if (name === 'Học Bá Tu Tiên') { pSpp += 15; }
        if (name === 'Vạn Thú Thân Hòa') { pSpp += 10; }
        if (name === 'Thiên Sinh Khí Sư') { pAtk += 10; pDef += 10; }
        if (name === 'Dược Vương Hóa Thân') { pSpp += 20; }
        if (name === 'Hắc Khách Thiên Cơ') { pSpp += 60; } 
        if (name === 'Thánh Nữ Mạng') { pSpp += 40; } 

        // Thể Chất Dị Biệt & Bất Lợi
        if (name === 'Tuyệt Dục Tàn Khu') { pSpd += 60; pAtk += 20; pDef -= 30; }
        if (name === 'Đoạn Kinh Mạch') { pAtk -= 20; pSpd -= 20; }
        if (name === 'Thiên Đạo Áp Chế') { pAtk -= 20; pDef -= 20; pSpd -= 20; pSpp -= 20; pMult -= 0.3; } // V9.11 NERF: Trừ thẳng 30% bạo kích để chống "Buff Chồng"
    });

    // V9.11 FIX & NERF: KHÓA "LỖ HỔNG HUYỄN MỘNG" (Hack Thần Thức vô hạn)
    const soulAgeGap = Math.max(0, Math.floor(Number(soulAge)) - Math.floor(Number(physicalAge)));
    if (soulAgeGap >= 100) {
        // V11.54 FIX: Giới hạn max Thần Thức nhận được từ Tuổi Linh Hồn dựa trên Cảnh giới (Max = stageId * 50)
        const maxBonusSpp = stageId * 50;
        const bonusSpp = Math.min(maxBonusSpp, Math.floor(soulAgeGap / 100));
        pSpp += bonusSpp;
        
        // Hình phạt "Tải Trọng Linh Hồn": Nếu hồn quá già (chênh > 3000 năm) mà xác phàm yếu, nhục thân sẽ bị ép nát.
        if (soulAgeGap >= 3000) {
            pDef -= 20;
            pAtk -= 10;
        }
    }

    if (runeCount > 0) {
        pMult += (runeCount * 0.1);
    }
    // Đảm bảo hệ số bạo kích không bị âm quá đáng
    pMult = Math.max(0.1, pMult);

    // V1.28: Tách bạch rõ ràng Chỉ số từ Thể Chất Thuần Túy để dùng cho Đột Phá, cấm Công Pháp hack stack
    traitAtk = Math.max(0, pAtk - coreAtk);
    traitDef = Math.max(0, pDef - coreDef);
    traitSpd = Math.max(0, pSpd - coreSpd);
    traitSpp = Math.max(0, pSpp - coreSpp);

    return { 
        atk: pAtk, def: pDef, spd: pSpd, spp: pSpp, mult: pMult,
        traitAtk, traitDef, traitSpd, traitSpp,
        coreAtk, coreDef, coreSpd, coreSpp
    };
};

// V1.28: TÍNH LỰC CHIẾN THỰC TẾ (CHỈ SỐ LÀ CHÂN LÝ - XÓA BỎ BỘI SỐ CẢNH GIỚI NHÂN TẠO)
const getCombatPower = (totalAtk, totalDef, totalSpd, totalSpp, stageId = 1, activeEffects = []) => {
    let effectBonus = 0;
    if (Array.isArray(activeEffects)) {
        activeEffects.forEach(e => {
            if (e && e.statMods) {
                effectBonus += (Number(e.statMods.attack) || 0) + (Number(e.statMods.defense) || 0) + (Number(e.statMods.speed) || 0) + (Number(e.statMods.spiritPower) || 0);
            }
        });
    }
    // Lực chiến phản ánh trung thực năng lực chiến đấu thực tế sau mọi buff Đan Dược / Trận Pháp / Khí Vận
    const realPower = Math.max(1, Math.floor(totalAtk + totalDef + totalSpd + totalSpp + effectBonus));
    return realPower;
};

const resolveItemRarity = (item: any): string => {
  if (!item) return 'Trắng';
  const rawRarity = String(item.rarity || '').trim();
  if (rawRarity && rawRarity !== 'Trắng' && RARITY_COLORS[rawRarity]) return rawRarity;

  const tier = String(item.tier || '').toLowerCase();
  const name = String(item.name || '').toLowerCase();

  if (tier.includes('hỗn độn') || tier.includes('thần khí') || tier.includes('sáng thế') || name.includes('hỗn độn') || name.includes('chén thánh') || name.includes('behelit') || name.includes('tụ bảo bồn')) {
    return 'Đỏ';
  }
  if (tier.includes('tiên khí') || tier.includes('thông thiên linh bảo') || tier.includes('thánh khí') || tier.includes('thần giai') || name.includes('tiên khí')) {
    return 'Cam';
  }
  if (tier.includes('pháp bảo') || tier.includes('cổ bảo') || tier.includes('chân khí') || tier.includes('tiên giai') || tier.includes('yêu hoàng')) {
    return 'Vàng';
  }
  if (tier.includes('bảo khí') || tier.includes('linh khí') || tier.includes('địa giai') || tier.includes('huyền giai') || tier.includes('linh đan') || tier.includes('yêu vương')) {
    return 'Tím';
  }
  if (tier.includes('pháp khí') || tier.includes('yêu khí') || tier.includes('linh giai') || tier.includes('yêu thú')) {
    return 'Xanh lam';
  }
  if (tier.includes('tinh lương') || tier.includes('hạ phẩm') || tier.includes('tài nguyên') || name.includes('linh thảo') || name.includes('thảo dược')) {
    return 'Xanh lá';
  }
  if (/(rách|mẻ|gãy|cùn|rỉ sét|phế phẩm|hỏng|vỡ)/.test(name) || tier.includes('phế phẩm')) {
    return 'Xám';
  }
  return rawRarity || 'Trắng';
};

const getItemColor = (item: any) => {
  const r = resolveItemRarity(item);
  return RARITY_COLORS[r] || RARITY_COLORS['Trắng'];
};

const getItemTypeIcon = (item: any, size = 14, className = '') => {
  if (!item) return <Package size={size} className={className}/>;
  if (item.type === 'sealed_spatial') return <Lock size={size} className="text-orange-400 shrink-0"/>;
  if (['spatial', 'soul_spatial'].includes(item.type)) return <Package size={size} className={className}/>;
  if (item.type === 'equip') {
    const n = String(item.name || '').toLowerCase();
    if (/(giáp|y|bào|khải|sam|đai|mũ|nón|ủng|hài|ngoa)/.test(n)) return <Shield size={size} className={className}/>;
    return <Swords size={size} className={className}/>;
  }
  if (item.type === 'food') return <FlaskRound size={size} className={className}/>;
  if (item.type === 'demon_soul') return <Flame size={size} className={className}/>;
  if (item.type === 'mount') return <PawPrint size={size} className={className}/>;
  if (item.type === 'misc') {
    const n = String(item.name || '').toLowerCase();
    if (/(ngọc giản|bí kíp|tàn quyển|thư|phù)/.test(n)) return <ScrollText size={size} className={className}/>;
    if (/(quặng|khoáng|ngọc|tinh thạch|châu|thạch)/.test(n)) return <Gem size={size} className={className}/>;
    return <Sparkles size={size} className={className}/>;
  }
  return <Package size={size} className={className}/>;
};
