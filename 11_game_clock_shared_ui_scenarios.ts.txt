const getInGameDate = (age, startAge = 18) => {
  const safeAge = Number(age) || 18;
  const safeStart = Number(startAge) || 18;
  const yearsPassed = Math.max(0, safeAge - safeStart);
  
  const totalAbsoluteMinutes = Math.round(yearsPassed * 360 * 24 * 60);
  
  let year = Math.floor(totalAbsoluteMinutes / (360 * 24 * 60)) + 1;
  
  const remainingMinutesAfterYear = totalAbsoluteMinutes % (360 * 24 * 60);
  
  const totalDays = Math.floor(remainingMinutesAfterYear / (24 * 60));
  const remainingMinutesAfterDay = remainingMinutesAfterYear % (24 * 60);
  
  const month = Math.floor(totalDays / 30) + 1;
  const day = (totalDays % 30) + 1;
  
  const hour = Math.floor(remainingMinutesAfterDay / 60);
  const minute = remainingMinutesAfterDay % 60;
  
  const formattedHour = hour.toString().padStart(2, '0');
  const formattedMinute = minute.toString().padStart(2, '0');
  
  let timeDesc = "";
  if (hour >= 3 && hour < 5) timeDesc = "Rạng sáng";
  else if (hour >= 5 && hour < 8) timeDesc = "Sáng sớm";
  else if (hour >= 8 && hour < 11) timeDesc = "Buổi sáng";
  else if (hour >= 11 && hour < 13) timeDesc = "Giữa trưa";
  else if (hour >= 13 && hour < 15) timeDesc = "Đầu chiều";
  else if (hour >= 15 && hour < 18) timeDesc = "Chiều muộn";
  else if (hour >= 18 && hour < 19) timeDesc = "Chập tối";
  else if (hour >= 19 && hour < 22) timeDesc = "Tối";
  else if (hour >= 22 && hour < 24) timeDesc = "Khuya";
  else timeDesc = "Nửa đêm"; // Từ 00:00 đến 02:59
  
  return `${timeDesc} (${formattedHour}:${formattedMinute}) - Ngày ${day} Tháng ${month} Năm ${year}`;
};

const getClockTime = (age, startAge = 18) => {
  const safeAge = Number(age) || 18;
  const safeStart = Number(startAge) || 18;
  const yearsPassed = Math.max(0, safeAge - safeStart);
  const totalAbsoluteMinutes = Math.round(yearsPassed * 360 * 24 * 60);
  const remainingMinutesAfterDay = totalAbsoluteMinutes % (24 * 60);
  const hour = Math.floor(remainingMinutesAfterDay / 60);
  const minute = remainingMinutesAfterDay % 60;
  return `${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}`;
};

const getRealmTimeStr = (realmName, knownRealms) => {
    const realm = knownRealms[realmName] || { accumulated: 0 };
    return getInGameDate(__GlobalStartAge + realm.accumulated, __GlobalStartAge);
};
const getRealmClockTime = (realmName, knownRealms) => {
    const realm = knownRealms[realmName] || { accumulated: 0 };
    return getClockTime(__GlobalStartAge + realm.accumulated, __GlobalStartAge);
};

const StaticPanel = ({ title, icon: Icon, children, titleRightNode, collapsible = false, fullHeight = true, defaultCollapsed = false }) => {
  const [isCollapsed, setIsCollapsed] = useState(defaultCollapsed);
  useEffect(() => {
    if (defaultCollapsed !== undefined) setIsCollapsed(defaultCollapsed);
  }, [defaultCollapsed]);
  return (
    <div className={`flex flex-col bg-[var(--bg-panel)] rounded-lg shadow-xl border border-[var(--border-main)] ${isCollapsed ? 'h-auto' : (fullHeight ? 'h-full' : 'h-auto')}`}>
      <div className={`h-12 shrink-0 bg-[var(--bg-surface)] ${isCollapsed ? '' : 'border-b'} border-[var(--border-main)] flex items-center px-4 justify-between sticky top-0 z-10 ${isCollapsed ? 'rounded-lg' : 'rounded-t-lg'}`}>
        <div className={`flex items-center gap-2 overflow-hidden ${collapsible ? 'cursor-pointer select-none' : ''}`} onClick={() => collapsible && setIsCollapsed(!isCollapsed)}>
          <Icon size={16} className="text-teal-500" />
          <span className="text-sm font-bold uppercase truncate text-[var(--text-bold)]" style={{ letterSpacing: '0.02em', lineHeight: 1.5 }}>{title}</span>
        </div>
        <div className="flex items-center gap-2">
          {titleRightNode}
          {collapsible && (
            <button onClick={() => setIsCollapsed(!isCollapsed)} className="text-[var(--text-muted)] hover:text-teal-500 transition-colors p-1 rounded hover:bg-[var(--bg-hover)]" title={isCollapsed ? "Mở rộng" : "Thu gọn"}>
              <ChevronDown size={18} className={`transition-transform duration-300 ${isCollapsed ? '' : 'rotate-180'}`} />
            </button>
          )}
        </div>
      </div>
      {!isCollapsed && (
        <div className="flex-1 flex flex-col min-h-0 overflow-hidden">
          {children}
        </div>
      )}
    </div>
  );
};

const StatBar = ({ icon: Icon, label, value, max, colorClass, showPercent = false, isImmersion = false, tooltip, onClick }) => {
  const safeVal = Number(value) || 0;
  const safeMax = Number(max) || 1;
  const percent = Math.min(100, (safeVal / safeMax) * 100);
  
  // V5.47 FIX: Nhận diện thọ nguyên vô cực (Biên độ trên 1 triệu năm)
  const isInfinite = safeMax >= 999999 && label === 'Tuổi Tác Sinh Học';
  
  let displayValue = formatLargeNumber(safeVal);
  if (max) {
      if (isInfinite) {
          displayValue += ' / ∞';
      } else {
          displayValue += '/' + formatLargeNumber(safeMax);
      }
  }
  if (showPercent && max && !isInfinite) displayValue += ' (' + Math.floor(percent) + '%)';
  
  return (
  <div className={"mb-4 " + (onClick ? "cursor-pointer group" : "")} onClick={onClick} title={tooltip}>
    <div className="flex justify-between text-sm mb-1.5 text-[var(--text-muted)] font-bold tracking-wider">
      <span className={"flex items-center gap-1 uppercase " + (onClick ? "group-hover:text-teal-400 transition-colors" : "")}>
        <Icon size={14} /> {label}
      </span>
      <span className={isImmersion ? "" : "border-b border-dashed border-[var(--text-muted)] hover:text-[var(--text-bold)] transition-colors"}>
        {isImmersion ? '???' : displayValue}
      </span>
    </div>
    {!isImmersion && (
      <div className="h-2.5 w-full bg-[var(--border-main)] rounded-full overflow-hidden">
         <div className={"h-full transition-all duration-500 " + colorClass} style={{ width: percent + '%' }} />
      </div>
    )}
  </div>
)};
