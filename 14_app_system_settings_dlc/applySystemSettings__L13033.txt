  const applySystemSettings = (s: any) => {
    if (!s || typeof s !== 'object') return;
    try {
      const getVal = (k1: string, k2: string) => s[k1] !== undefined ? s[k1] : s[k2];

      const showCrowd = getVal('settingShowCrowd', 'tuTienShowCrowd');
      if (showCrowd !== undefined && showCrowd !== null) {
        const val = String(showCrowd) !== 'false';
        setSettingShowCrowd(val);
        localStorage.setItem('tuTienShowCrowd', String(val));
      }
      const nsfw = getVal('settingNsfwMode', 'tuTienNsfwMode');
      if (nsfw !== undefined && nsfw !== null) {
        const val = String(nsfw) === 'true';
        setSettingNsfwMode(val);
        localStorage.setItem('tuTienNsfwMode', String(val));
      }
      const humor = getVal('settingHumorousCrowd', 'tuTienHumorCrowd');
      if (humor !== undefined && humor !== null) {
        const val = String(humor) === 'true';
        setSettingHumorousCrowd(val);
        localStorage.setItem('tuTienHumorCrowd', String(val));
      }
      const hideOpts = getVal('settingHideOptions', 'tuTienHideOptions');
      if (hideOpts !== undefined && hideOpts !== null) {
        const val = String(hideOpts) === 'true';
        setSettingHideOptions(val);
        localStorage.setItem('tuTienHideOptions', String(val));
      }
      const hideTags = getVal('settingHideOptionTags', 'tuTienHideOptionTags');
      if (hideTags !== undefined && hideTags !== null) {
        const val = String(hideTags) === 'true';
        setSettingHideOptionTags(val);
        localStorage.setItem('tuTienHideOptionTags', String(val));
      }
      const epic = getVal('settingEpicCombat', 'tuTienEpicCombat');
      if (epic !== undefined && epic !== null) {
        const val = String(epic) === 'true';
        setSettingEpicCombat(val);
        localStorage.setItem('tuTienEpicCombat', String(val));
      }
      const deep = getVal('settingDeepReasoning', 'tuTienDeepReasoning');
      if (deep !== undefined && deep !== null) {
        const val = String(deep) === 'true';
        setSettingDeepReasoning(val);
        localStorage.setItem('tuTienDeepReasoning', String(val));
      }
      const noPraise = getVal('settingNoPraise', 'tuTienNoPraise');
      if (noPraise !== undefined && noPraise !== null) {
        const val = String(noPraise) !== 'false';
        setSettingNoPraise(val);
        localStorage.setItem('tuTienNoPraise', String(val));
      }
      const noMutter = getVal('settingNoMuttering', 'tuTienNoMuttering');
      if (noMutter !== undefined && noMutter !== null) {
        const val = String(noMutter) === 'true';
        setSettingNoMuttering(val);
        localStorage.setItem('tuTienNoMuttering', String(val));
      }
      const optsCount = getVal('settingOptionsCount', 'tuTienOptionsCount');
      if (optsCount !== undefined && optsCount !== null) {
        const count = Math.max(4, Math.min(10, Number(optsCount) || 4));
        setSettingOptionsCount(count);
        localStorage.setItem('tuTienOptionsCount', String(count));
      }
      const invMode = getVal('invSearchMode', 'tuTienInvMode');
      if (invMode !== undefined && invMode !== null) {
        setInvSearchMode(String(invMode));
        localStorage.setItem('tuTienInvMode', String(invMode));
      }
      const alias = getVal('isAliasEnabled', 'tuTienAlias');
      if (alias !== undefined && alias !== null) {
        const val = String(alias) !== 'false';
        setIsAliasEnabled(val);
        localStorage.setItem('tuTienAlias', String(val));
      }
      if (s.aliasInput !== undefined && s.aliasInput !== null && typeof setAliasInput === 'function') {
        setAliasInput(String(s.aliasInput));
      }
      const genZAll = getVal('settingGenZAll', 'tuTienGenZAll');
      if (genZAll !== undefined && genZAll !== null) {
        const val = String(genZAll) === 'true';
        setSettingGenZAll(val);
        localStorage.setItem('tuTienGenZAll', String(val));
      }
      const genZRom = getVal('settingGenZRomance', 'tuTienGenZRomance');
      if (genZRom !== undefined && genZRom !== null) {
        const val = String(genZRom) === 'true';
        setSettingGenZRomance(val);
        localStorage.setItem('tuTienGenZRomance', String(val));
      }
      const genZCom = getVal('settingGenZCombat', 'tuTienGenZCombat');
      if (genZCom !== undefined && genZCom !== null) {
        const val = String(genZCom) === 'true';

        const val = String(genZCu) === 'true';
        setSettingGenZCultivate(val);
        localStorage.setItem('tuTienGenZCultivate', String(val));
      }
      const subStats = getVal('visibleSubStats', 'tuTienSubStats');
      if (subStats !== undefined && subStats !== null) {
        try {
          const arr = Array.isArray(subStats) ? subStats : JSON.parse(subStats);

        setChapterLengthLevel(val);
        localStorage.setItem('tuTienChapLength', String(val));

      }
      const autoExp = getVal('isAutoWorldExpansion', 'tuTienAutoWorldExp');
      if (autoExp !== undefined && autoExp !== null) {
        const val = String(autoExp) === 'true';
        setIsAutoWorldExpansion(val);
        localStorage.setItem('tuTienAutoWorldExp', String(val));
      }
    } catch(err) {
      console.error("Lỗi khi áp dụng Cài Đặt Hệ Thống:", err);
    }
  };
