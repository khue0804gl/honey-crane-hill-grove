  if (coTien) tran = 0;

  return { chiSo: Math.max(0.3, Math.min(3.0, chiSo)), tho, tran, dacBiet };
};

export const getEventNpcsForEvent = (evt: CanonEvent, curYear: number): EventNpc[] => {
  if (!evt) return [];

  Object.values(EVENT_NPCS).forEach(n => { if (n && n.attachEventId === evt.id) ids.add(n.id); });
  return Array.from(ids)

    .filter(Boolean)
    .filter(n => curYear >= (n.activeWindow?.fromYear ?? 1) && curYear <= (n.activeWindow?.toYear ?? 9999));
};

// Sinh khối [CHUỖI NHÂN QUẢ] cho sự kiện có sự kiện tiền đề
export const buildChainContext = (evt: CanonEvent, list: CanonEvent[]): string => {
  if (!evt || !evt.parentEventId || !Array.isArray(list)) return '';
  const parent = list.find(e => e.id === evt.parentEventId);
  if (!parent) return '';
  const cons = (parent.consequences && parent.consequences[0]) || null;
  const expected = cons ? (parent.hasDiverged ? cons.ifDiverged : cons.ifIntact) : null;
  let out = `\n     + [CHUỖI NHÂN QUẢ] Sự kiện tiền đề: "${parent.title}" — Trạng thái: ${parent.hasDiverged ? 'ĐÃ BỊ PHÂN KỲ BỞI NVC' : 'NGUYÊN TÁC CHƯA BỊ BẺ CONG'}`;
  if (expected) out += `\n     → Hệ quả kỳ vọng phải phản ánh vào chương này: ${expected}`;
  if (parent.hasDiverged) out += `\n     → CẢNH BÁO: sự kiện tiền đề đã bị bẻ cong, BẠN BỊ CẤM chạy lại kịch bản gốc của sự kiện này!`;
  return out;
};

// Sinh Aftermath Event khi Canon bị phân kỳ (tutien2.txt PHẦN IV.4 / VII.3)
export const buildAftermathEvent = (evt: CanonEvent | undefined, survivor?: string, blocked?: string): { type: string; target?: string; description: string } => {
  const evtTitle = evt?.title || 'sự kiện nguyên tác';
  if (survivor) {
    return {
      type: 'debt_of_gratitude',
      target: survivor,
      description: `Nạn nhân ${survivor} của ${evtTitle} đã sống sót nhờ NVC, sinh ra Món Nợ Ân Tình: ân nhân sẽ theo đuổi trả ơn hoặc trả thù tùy duyên, kéo theo hệ quả về sau.`

    return {
      type: 'vengeful_enemy',
      target: blocked,
      description: `Kẻ đáng lẽ được lợi ${blocked} trong ${evtTitle} đã bị chặn, sinh ra Kẻ Thù Oán Hận: hắn sẽ truy xét nguyên nhân thất bại và tìm tới NVC.`

  }
  return {

  };
};

// Chuẩn hóa requiredConditions thành token máy đọc được (tutien2.txt PHẦN VI.1)
export const checkCanonConditions = (evt: CanonEvent, ctx: { curYear: number; curLocationId: string; activeState: any }): boolean => {
  const conds = evt?.requiredConditions || [];
  if (conds.length === 0) return true;
  return conds.every((raw) => {
    const token = String(raw).trim();
    if (!token) return true;
    const low = token.toLowerCase();
    try {
      if (low.startsWith('year>=')) return ctx.curYear >= Number(token.slice(6));
      if (low.startsWith('year<=')) return ctx.curYear <= Number(token.slice(6));
      if (low.startsWith('loc=')) return ctx.curLocationId.includes(token.slice(4).trim());
      if (low.startsWith('z=')) return String(ctx.activeState?.coordinates?.z ?? 0) === token.slice(2).trim();
      if (low.startsWith('dlc:')) return (ctx.activeState?.activeDLCs || []).includes(token.slice(4).trim());
      if (low.startsWith('trait:')) {
        const t = token.slice(6).trim();
        return (ctx.activeState?.traits || []).some((x: any) => String(x?.name || x) === t);
      }
      if (low.startsWith('hasitem:')) {
        const it = token.slice(8).trim();
        return (ctx.activeState?.inventory || []).some((x: any) => String(x?.name || '').includes(it));
      }
    } catch (err) {
      return true;
    }
    return true; // Token chưa nhận dạng được -> bỏ qua để không phá dữ liệu save cũ
  });
};
