                                            <div className="flex justify-between border-b border-[var(--border-main)] pb-1.5 text-[11px]">

                if (isGlobalCloneSystem) {

                    safeContainers = [{ id: 'shared_clone_inv', name: 'Không Gian Vô Hạn', capacity: 9999999 }];

                            <p className="text-xl font-bold uppercase tracking-widest">Hành Trang Bị Phong Ấn</p>

                    safeContainers = comp.containers || [];
                    safeInv = comp.inventory || [];
                    compName = comp.name;
                }
                
                let displayInv = safeInv.filter(i => i.containerId === activeCompContainerId);
                if (compInvFilter !== 'all') {

                                                        setStoryLog(prev => {

'bg-orange-600 border-orange-500 text-white' : 'bg-[var(--bg-surface)] border-[var(--border-main)] text-[var(--text-muted)] hover:text-[var(--text-bold)]'}`}>Tất cả</button>

'bg-orange-600 border-orange-500 text-white' : 'bg-[var(--bg-surface)] border-[var(--border-main)] text-[var(--text-muted)] hover:text-orange-400'}`}><Axe size={12}/> Trang bị</button>

'bg-orange-600 border-orange-500 text-white' : 'bg-[var(--bg-surface)] border-[var(--border-main)] text-[var(--text-muted)] hover:text-orange-400'}`}><FlaskRound size={12}/> Đan/Thực</button>
                              <button onClick={() => setCompInvFilter('misc')} className={`px-3 py-1.5 rounded-lg text-xs font-bold whitespace-nowrap border transition-colors flex items-center gap-1 ${compInvFilter === 'misc' ? 'bg-orange-600 border-orange-500 text-white' : 'bg-[var(--bg-surface)] border-[var(--border-main)] text-[var(--text-muted)] hover:text-orange-400'}`}><Package size={12}/> Khác</button>
                           </div>
                        </div>

                        {/* Danh Sách Vật Phẩm Tùy Tùng */}
                        <div className="flex-1 bg-[var(--bg-body)] border border-[var(--border-main)] rounded-xl p-3 overflow-y-auto custom-scrollbar mb-4 min-h-[160px] max-h-[250px]">
                            {displayInv.length === 0 ? (
                               <div className="h-full flex flex-col items-center justify-center text-[var(--text-muted)] opacity-50 py-10">
                                   <Package size={48} className="mb-3"/>
                                   <p className="text-base italic">Không gian này trống rỗng.</p>
                               </div>
                            ) : (
                               <div className="flex flex-col gap-1">
                                  {displayInv.map((item, idx) => {
                                     const isSelected = selectedCompInvItem?.name === item.name;
                                     const cVisual = getItemColor(item);
                                     const cRarity = resolveItemRarity(item);
                                     return (
                                     <div key={idx} onClick={() => { setSelectedCompInvItem(item); setCompMoveAmount(item.quantity); }} className={`cursor-pointer px-3 py-2 flex items-center transition-all rounded select-none ${isSelected ? 'bg-orange-900/20 border-l-2 border-orange-500 shadow-sm' : `hover:bg-[var(--bg-hover)] border-l-2 ${cVisual.border?.split(' ')[0] || 'border-transparent'}`}`}>
                                         <div className="flex-1 flex justify-between items-center w-full overflow-hidden">
                                             <div className="flex items-center gap-2 overflow-hidden w-full">
                                                 <div className={`w-5 h-5 rounded flex items-center justify-center shrink-0 ${cVisual.bg || 'bg-gray-800/30'}`}>
                                                    {getItemTypeIcon(item, 13, cVisual.iconColor || 'text-gray-300')}
                                                 </div>
                                                 <span className={`text-sm font-bold truncate leading-6 ${cVisual.text || 'text-gray-200'}`}>{item.name}</span>
                                                 <span className={`text-[10px] px-1.5 py-0.2 rounded border font-mono uppercase tracking-wider shrink-0 ${cVisual.bg || 'bg-gray-800/30'} ${cVisual.border || 'border-gray-600/40'} ${cVisual.text || 'text-gray-300'}`}>
                                                    {cRarity}
                                                 </span>
                                             </div>
                                             <span className="text-xs font-mono font-bold text-[var(--text-muted)] shrink-0 pl-3">x{item.quantity}</span>
                                         </div>
                                     </div>
                                  )})}
                               </div>
                            )}
                        </div>

                        {/* Bảng Chi Tiết Vật Phẩm Của Tùy Tùng */}
                        {selectedCompInvItem ? (
                           <div className="bg-[var(--bg-surface)] border border-[var(--border-main)] rounded-xl p-4 relative overflow-hidden animate-in fade-in zoom-in-95 duration-200">
                              <div className={`absolute top-0 right-0 w-32 h-full bg-gradient-to-l opacity-20 pointer-events-none ${RARITY_COLORS[selectedCompInvItem.rarity]?.bg?.replace('/10', '/100') || 'from-gray-500'}`}></div>
                              
                              <div className="flex justify-between items-start mb-2 relative z-10">
                                 <div>
                                    <h4 className={`text-lg font-bold ${RARITY_COLORS[selectedCompInvItem.rarity]?.text || 'text-[var(--text-main)]'}`}>{selectedCompInvItem.name}</h4>
                                    <div className="text-xs text-[var(--text-muted)] mt-1 flex items-center gap-2">
                                       <span className="border border-[var(--border-main)] px-1.5 py-0.5 rounded bg-[var(--bg-body)]">{selectedCompInvItem.tier || 'Phàm Khí'}</span>
                                       <span className="border border-[var(--border-main)] px-1.5 py-0.5 rounded bg-[var(--bg-body)]">{selectedCompInvItem.rarity || 'Trắng'}</span>
                                       {selectedCompInvItem.subLevel && <span className="border border-red-900/50 px-1.5 py-0.5 rounded bg-red-900/20 text-red-400 font-bold">Cấp {selectedCompInvItem.subLevel}</span>}
                                       <span className="font-mono text-orange-400">⚖️ {getItemTotalWeight(selectedCompInvItem).toFixed(2)} kg</span>
                                    </div>
                                 </div>
                                 <button onClick={() => setSelectedCompInvItem(null)} className="p-1 hover:bg-[var(--bg-body)] rounded text-[var(--text-muted)]"><X size={16}/></button>
                              </div>

                              <div className="text-sm text-[var(--text-main)] italic leading-relaxed mb-4 relative z-10">"{selectedCompInvItem.desc}"</div>
                              
                              {selectedCompInvItem.perks && selectedCompInvItem.perks.length > 0 && (
                                 <div className="mb-4 relative z-10">
                                    <span className="text-[10px] uppercase font-bold text-[var(--text-muted)] mb-1 block">Đặc tính:</span>
                                    <div className="flex flex-wrap gap-1.5">
                                       {selectedCompInvItem.perks.map((p, i) => <span key={i} className="bg-[var(--bg-body)] border border-[var(--border-main)] px-2 py-1 rounded text-xs font-bold text-[var(--text-main)] shadow-sm">{String(p)}</span>)}
                                    </div>
                                 </div>
                              )}

                              {selectedCompInvItem.quantity > 1 && (
                                  <div className="flex items-center gap-2 mb-3 bg-[var(--bg-body)] p-2 rounded-lg border border-[var(--border-main)] relative z-10">
                                      <span className="text-[10px] uppercase font-bold text-[var(--text-muted)] whitespace-nowrap">Số lượng:</span>
                                      <input type="range" min="1" max={selectedCompInvItem.quantity} value={compMoveAmount} onChange={e => setCompMoveAmount(Number(e.target.value))} className="flex-1 accent-orange-500 h-1.5 bg-gray-700 rounded-lg appearance-none cursor-pointer" />
                                      <input type="number" min="1" max={selectedCompInvItem.quantity} value={compMoveAmount} onChange={e => setCompMoveAmount(Math.min(selectedCompInvItem.quantity, Math.max(1, parseInt(e.target.value)||1)))} className="w-14 bg-[var(--bg-surface)] text-[var(--text-main)] border border-[var(--border-main)] rounded px-1.5 py-1 text-xs font-bold text-center focus:outline-none focus:border-orange-500" />
                                  </div>
                              )}

                              <div className="flex gap-2 relative z-10 border-t border-[var(--border-main)] pt-3">
                                 {!isGlobalCloneSystem && selectedCompInvItem.type === 'equip' && (
                                     <button onClick={() => {
                                         const getEquipSlotCustom = (item) => {
                                             if (item.equipSlot && ['weapon', 'armor', 'head', 'boots', 'accessory', 'artifact'].includes(item.equipSlot)) return item.equipSlot;
                                             const textToMatch = String(item.name + " " + (item.desc || "")).toLowerCase();
                                             if (/(kiếm|đao|thương|kích|côn|búa|rìu|cung|trượng|xích|roi|chùy|súng|pháo|ám khí)/.test(textToMatch)) return 'weapon';
                                             if (/(áo|giáp|bào|pháp\s*y|y\s*phục|tơ|khiên|thuẫn|quần|(?:^|\s)y(?:\s|$))/.test(textToMatch)) return 'armor';
                                             if (/(mũ|quán|trâm|nón|miện|khăn|cài|vương miện|scouter|máy đo)/.test(textToMatch)) return 'head';
                                             if (/(giày|hài|ngoa|ủng)/.test(textToMatch)) return 'boots';
                                             if (/(nhẫn|ngọc|bội|liên|vòng|phù|chuyền|xuyến|lệnh bài)/.test(textToMatch) && !textToMatch.includes('hỏa phù')) return 'accessory';
                                             return 'artifact';
                                         };
                                         const slot = getEquipSlotCustom(selectedCompInvItem);
                                         
                                         setGameState(prev => {
                                             const newComps = [...prev.companions];
                                             const cIdx = newComps.findIndex(x => x.name === compName);
                                             if (cIdx < 0) return prev;
                                             
                                             if (!newComps[cIdx].equipped) newComps[cIdx].equipped = { weapon: null, armor: null, head: null, boots: null, artifact: null, accessory: null };
                                             
                                             const itemIdx = newComps[cIdx].inventory.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                             if (itemIdx >= 0) {
                                                 newComps[cIdx].inventory[itemIdx].quantity -= 1;
                                                 if (newComps[cIdx].inventory[itemIdx].quantity <= 0) newComps[cIdx].inventory.splice(itemIdx, 1);
                                             }
                                             
                                             if (newComps[cIdx].equipped[slot]) {
                                                 const oldItem = newComps[cIdx].equipped[slot];
                                                 const existIdx = newComps[cIdx].inventory.findIndex(i => i.name === oldItem.name && i.containerId === (selectedCompInvItem.containerId || 'default') && !i.attackBonus);
                                                 if (existIdx >= 0) newComps[cIdx].inventory[existIdx].quantity += 1;
                                                 else newComps[cIdx].inventory.push({ ...oldItem, quantity: 1, containerId: selectedCompInvItem.containerId || 'default' });
                                             }
                                             
                                             let enrichedItem = { ...selectedCompInvItem };
                                             enrichedItem.equipSlot = slot;
                                             if (enrichedItem.durability !== undefined && enrichedItem.maxDurability) {
                                                 const duraCost = Math.max(1, Math.floor(enrichedItem.maxDurability * 0.01));
                                                 enrichedItem.durability = Math.max(0, enrichedItem.durability - duraCost);
                                             }
                                             
                                             newComps[cIdx].equipped[slot] = enrichedItem;
                                             return { ...prev, companions: newComps };
                                         });
                                         setStoryLog(prev => {
                                             const newLog = [...prev, { type: 'system', text: `[Hệ Thống] Phân thân ${compName} đã trang bị [${selectedCompInvItem.name}].`, chapter: gameState.chapterCount }];
                                             return newLog.length > 50 ? newLog.slice(-50) : newLog;
                                         });
                                         setSelectedCompInvItem(null);
                                     }} className="flex-1 bg-teal-600 hover:bg-teal-500 text-white rounded-lg font-bold text-xs py-2 shadow-lg transition-transform hover:scale-105">
                                         Đeo Bộ Phận
                                     </button>
                                 )}
                                 {/* Nút Lấy Về NVC */}
                                 <button onClick={() => {
                                     setGameState(prev => {
                                         if (isGlobalCloneSystem) {
                                            const newCloneInv = [...(prev.cloneInventory || [])];
                                            const itemIdx = newCloneInv.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                            if (itemIdx >= 0) {
                                                const itemToTake = { ...newCloneInv[itemIdx], quantity: compMoveAmount };
                                                newCloneInv[itemIdx].quantity -= compMoveAmount;
                                                if (newCloneInv[itemIdx].quantity <= 0) newCloneInv.splice(itemIdx, 1);
                                                
                                                const newInv = [...prev.inventory];
                                                const targetCid = prev.containers[0].id; 
                                                const exist = newInv.find(x => x.name === itemToTake.name && x.containerId === targetCid);
                                                if (exist) exist.quantity += compMoveAmount;
                                                else newInv.push({...itemToTake, containerId: targetCid});
                                                
                                                return {...prev, cloneInventory: newCloneInv, inventory: newInv};
                                            }
                                            return prev;
                                         }

                                         const newComps = [...prev.companions];
                                         const cIdx = newComps.findIndex(x => x.name === compName);
                                         const itemIdx = newComps[cIdx].inventory.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                         
                                         if (itemIdx >= 0) {
                                             const itemToTake = { ...newComps[cIdx].inventory[itemIdx], quantity: compMoveAmount };
                                             newComps[cIdx].inventory[itemIdx].quantity -= compMoveAmount;
                                             if (newComps[cIdx].inventory[itemIdx].quantity <= 0) newComps[cIdx].inventory.splice(itemIdx, 1);
                                             
                                             const newInv = [...prev.inventory];
                                             const targetCid = prev.containers[0].id; 
                                             const exist = newInv.find(x => x.name === itemToTake.name && x.containerId === targetCid);
                                             if (exist) exist.quantity += compMoveAmount;
                                             else newInv.push({...itemToTake, containerId: targetCid});
                                             
                                             return {...prev, companions: newComps, inventory: newInv};
                                         }
                                         return prev;
                                     });
                                     setStoryLog(prev => {
                                         const newLog = [...prev, { type: 'system', text: `[Hệ Thống] Đã lấy lại [${selectedCompInvItem.name} x${compMoveAmount}] từ ${compName}.`, chapter: gameState.chapterCount }];
                                         return newLog.length > 50 ? newLog.slice(-50) : newLog;
                                     });
                                     setSelectedCompInvItem(null);
                                 }} className="flex-1 bg-orange-600 hover:bg-orange-500 text-white rounded-lg font-bold text-xs py-2 shadow-lg transition-transform hover:scale-105">
                                     Chuyển về Bản Thể
                                 </button>
                                 
                                 {/* V4.73: Nút Chuyển Túi Nội Bộ Của Tùy Tùng */}
                                 {safeContainers.length > 1 && selectedCompInvItem.type !== 'spatial' && selectedCompInvItem.type !== 'sealed_spatial' && (
                                     <select 
                                         className="flex-1 bg-[var(--bg-body)] border border-[var(--border-main)] rounded-lg text-xs px-1 font-bold text-orange-400 focus:outline-none focus:border-orange-500 cursor-pointer"
                                         value=""
                                         onChange={(e) => {
                                             const targetCid = e.target.value;
                                             if (!targetCid) return;
                                             setGameState(prev => {
                                                 const newComps = [...prev.companions];
                                                 const cIdx = newComps.findIndex(x => x.name === compName);
                                                 const itemIdx = newComps[cIdx].inventory.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                                 
                                                 if (itemIdx >= 0) {
                                                     const targetBag = safeContainers.find(c => c.id === targetCid);
                                                     const currWeight = newComps[cIdx].inventory.filter(i => i.containerId === targetCid).reduce((sum, i) => sum + getItemTotalWeight(i), 0);
                                                     if (currWeight + (getItemUnitWeight(selectedCompInvItem) * compMoveAmount) > (targetBag.capacity || 30)) {
                                                         showAlert(`Túi [${targetBag.name}] đã đầy, không thể chứa thêm!`);
                                                         return prev;
                                                     }
                                                     
                                                     const itemToMove = { ...newComps[cIdx].inventory[itemIdx], quantity: compMoveAmount, containerId: targetCid };
                                                     newComps[cIdx].inventory[itemIdx].quantity -= compMoveAmount;
                                                     if (newComps[cIdx].inventory[itemIdx].quantity <= 0) newComps[cIdx].inventory.splice(itemIdx, 1);
                                                     
                                                     const existDest = newComps[cIdx].inventory.find(x => x.name === itemToMove.name && x.containerId === targetCid);
                                                     if (existDest) existDest.quantity += compMoveAmount;
                                                     else newComps[cIdx].inventory.push(itemToMove);
                                                 }
                                                 return {...prev, companions: newComps};
                                             });
                                             setSelectedCompInvItem(null);
                                         }}
                                     >
                                         <option value="" disabled>Đổi túi...</option>
                                         {safeContainers.filter(c => c.id !== selectedCompInvItem.containerId).map(c => (
                                             <option key={c.id} value={c.id}>Sang: {c.name}</option>
                                         ))}
                                     </select>
                                 )}

                                 <button onClick={() => {
                                     setGameState(prev => {
                                         if (isGlobalCloneSystem) {
                                            const newCloneInv = [...(prev.cloneInventory || [])];
                                            const itemIdx = newCloneInv.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                            if (itemIdx >= 0) {
                                                newCloneInv[itemIdx].quantity -= compMoveAmount;
                                                if (newCloneInv[itemIdx].quantity <= 0) newCloneInv.splice(itemIdx, 1);
                                            }
                                            return {...prev, cloneInventory: newCloneInv};
                                         }

                                         const newComps = [...prev.companions];
                                         const cIdx = newComps.findIndex(x => x.name === compName);
                                         const itemIdx = newComps[cIdx].inventory.findIndex(x => x.name === selectedCompInvItem.name && x.containerId === selectedCompInvItem.containerId);
                                         if (itemIdx >= 0) {
                                             newComps[cIdx].inventory[itemIdx].quantity -= compMoveAmount;
                                             if (newComps[cIdx].inventory[itemIdx].quantity <= 0) newComps[cIdx].inventory.splice(itemIdx, 1);
                                         }
                                         return {...prev, companions: newComps};
                                     });
                                     setStoryLog(prev => {
                                         const newLog = [...prev, { type: 'system', text: `[Hệ Thống] Đã vứt bỏ [${selectedCompInvItem.name} x${compMoveAmount}] từ túi của ${compName}.`, chapter: gameState.chapterCount }];
                                         return newLog.length > 50 ? newLog.slice(-50) : newLog;
                                     });
                                     setSelectedCompInvItem(null);
                                 }} className="px-4 py-2 bg-red-900/20 text-red-500 hover:bg-red-600 hover:text-white rounded-lg font-bold text-xs flex items-center justify-center gap-1 transition-colors border border-red-900/30 shrink-0">
                                    <Trash2 size={14}/> Vứt
                                 </button>
                              </div>
                           </div>
                        ) : (
                           <div className="bg-[var(--bg-surface)] border border-[var(--border-main)] border-dashed rounded-xl p-6 flex flex-col items-center justify-center text-center opacity-50">
                              <MousePointerClick size={24} className="mb-2 text-[var(--text-muted)]"/>
                              <p className="text-xs text-[var(--text-muted)]">Chọn một vật phẩm ở trên<br/>để xem chi tiết và thu hồi.</p>
                           </div>
                        )}
                    </div>
                );
            })()}
          </div>
        </div>
      )}
