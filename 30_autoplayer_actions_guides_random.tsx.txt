      const isMongAo = gameState.gameMode === 'mong_ao';
      const validBuilds = DESTINY_BUILDS.filter(b => {
          const bgObj = BACKGROUNDS_POOL.find(x => x.id === b.bg);
          if (!bgObj) return false;
          
          const traitsValid = b.traits.every(tid => TRAITS_POOL.some(t => t.id === tid));
          if (!traitsValid) return false;

          if (isMongAo) return true;

          const bgCost = bgObj.cost || 0;
          const traitsCost = b.traits.reduce((sum, tid) => {
              const tr = TRAITS_POOL.find(t => t.id === tid);
              return sum + (tr ? (tr.cost || 0) : 0);
          }, 0);

          return (bgCost + traitsCost) <= 10;
      });
      
      if (validBuilds.length === 0) {
          return showAlert("Thiên cơ hỗn loạn, chưa tìm thấy kịch bản định mệnh phù hợp với thiên đạo này!");

      const chosenBuild = validBuilds[Math.floor(Math.random() * validBuilds.length)];
      
      setSelectedBgId(chosenBuild.bg);
      setSelectedTraitIds(chosenBuild.traits);

      // Cập nhật Tôn Danh, Giới Tính, Đạo Tâm, Văn Phong
      setGameState(prev => {
          const personality = PERSONALITIES_POOL.find(p => p.name === chosenBuild.daoHeart);
          const newDaoHeart = personality ? `${personality.name} - ${personality.desc}` : prev.daoHeart;
          return {
              ...prev,
              name: prev.name && prev.name.trim() !== '' ? prev.name : (chosenBuild.suggestedName || prev.name),
              gender: chosenBuild.gender || prev.gender || 'Nam',
              daoHeart: newDaoHeart,

          };
      });
      
      // Đồng bộ Khí vận lên Giao diện lưới 6 ô để người chơi dễ đọc
      const chosenTraitObjects = TRAITS_POOL.filter(t => chosenBuild.traits.includes(t.id));
      let newRolled = [...chosenTraitObjects];
      const selectableTraits = TRAITS_POOL.filter(t => {
          if (t.isDynamic || chosenBuild.traits.includes(t.id)) return false;
          return t.id < 10000;
      });
      const shuffled = [...selectableTraits].sort(() => 0.5 - Math.random());
      
      while(newRolled.length < 6 && shuffled.length > 0) {
          newRolled.push(shuffled.pop());
      }
      setRolledTraits(newRolled);

      const styleName = chosenBuild.narrativeStyle === 'ha_nhan' ? 'Thiết Huyết Tu Tiên' : chosenBuild.narrativeStyle === 'toi_gian' ? 'Tối Giản' : chosenBuild.narrativeStyle === 'hai_huoc' ? 'Kịch Bản / Hài Hước' : 'Cổ Phong Uy Nghiêm';

      showAlert(`🌟 THIÊN CƠ AN BÀI: [${chosenBuild.name}]\n\n` +
                `👤 Tôn danh đề xuất: ${chosenBuild.suggestedName || 'Tán Tu'} (${chosenBuild.gender || 'Nam'})\n` +
                `📜 Quá khứ: ${chosenBuild.desc}\n` +
                `🎭 Đạo tâm: ${chosenBuild.daoHeart || 'Tùy Duyên'}\n` +
                `✍️ Văn phong: ${styleName}\n\n` +
                `Hệ thống đã tự động phối hợp Tôn Danh, Xuất Thân, Khí Vận, Đạo Tâm và Văn Phong một cách hoàn mỹ cho đạo hữu!`);
  };
