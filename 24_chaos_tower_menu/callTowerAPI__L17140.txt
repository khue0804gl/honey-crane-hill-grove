              endLog = { type: 'system', text: '🏆 CHIẾN THẮNG! Đối thủ đã tan biến. Kỷ lục của bạn đã được cập nhật thành công!' };
          }

          setChaosTowerState(prev => {
              const newLogs = [...prev.logs, { type: 'story', text: data.battleLog }];
              if (data.playerHpChange !== 0 || data.opponentHpChange !== 0) newLogs.push({ type: 'system', text: `Sinh lực: Ta ${data.playerHpChange||0} HP. | Địch ${data.opponentHpChange||0} HP.` });
              if (endLog) newLogs.push(endLog);
              return { ...prev, myHP: newMyHp, myMP: newMyMp, oppHP: newOppHp, oppMP: newOppMp, logs: newLogs, options: newStatus === 'battle' ? (data.nextOptions || prev.options) : [], isGenerating: false, status: newStatus };
          });
          setTimeout(() => { if (towerScrollRef.current) towerScrollRef.current.scrollTop = towerScrollRef.current.scrollHeight; }, 100);
      } catch (e) {
          setChaosTowerState(prev => ({ ...prev, isGenerating: false, logs: [...prev.logs, { type: 'system', text: 'Thiên Đạo nghẽn mạng, vui lòng thử lại...' }] }));
      }
  };
