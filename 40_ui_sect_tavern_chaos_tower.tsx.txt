                           <span className={`text-xs font-bold uppercase tracking-widest mb-4 flex items-center gap-2 border-b pb-2 ${gameState.sect.isDynasty ? 'text-amber-500 border-amber-900/30' : 'text-blue-400 border-blue-900/30'}`}><Home size={16}/> {gameState.sect.isDynasty ? 'Cơ Quan / Phủ Đệ' : 'Phân Khu Kiến Trúc'}</span>

                   </div>
               )}

               {/* TAB CƠ SỞ HẠ TẦNG */}
               {sectModalTab === 'infra' && (
                   <div className="max-w-5xl mx-auto space-y-6 animate-in slide-in-from-bottom-4">
                       <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                           {(gameState.sect.infrastructure || []).map((inf, idx) => {
                               const upgradeCost = inf.level * 1000;
                               const repairCost = Math.floor(((inf.maxDurability - inf.durability) / inf.maxDurability) * 500);
                               
                               return (
                                   <div key={inf.id} className={`p-5 rounded-2xl border shadow-sm relative ${'bg-[var(--bg-surface)] border-purple-900/30'} flex flex-col`}>
                                       <div className="flex justify-between items-start mb-3">
                                           <div>
                                               <h4 className={`text-lg font-bold ${gameState.sect.isDynasty ? 'text-yellow-500' : 'text-purple-400'} uppercase tracking-widest`}>{inf.name} <span className="text-sm text-gray-400 lowercase">(Cấp {inf.level})</span></h4>
                                               <p className="text-xs text-gray-400 mt-1 italic">{inf.desc}</p>
                                           </div>
                                           <div className={`p-2 rounded-lg border ${gameState.sect.isDynasty ? 'bg-yellow-900/20 border-yellow-500/30 text-yellow-500' : 'bg-purple-900/20 border-purple-500/30 text-purple-400'}`}>
                                               {inf.id === 'dong_phu' ? <Home size={20}/> : inf.id === 'luyen_dan' ? <FlaskRound size={20}/> : inf.id === 'tang_kinh' ? <BookOpen size={20}/> : <Sparkles size={20}/>}
                                           </div>

                                       <div className="space-y-3 mb-4 flex-1">
                                           <div>
                                               <div className="flex justify-between text-[10px] font-bold text-blue-400 mb-1"><span>ĐỘ BỀN</span><span>{Math.floor(inf.durability)}/{inf.maxDurability}</span></div>
                                               <div className="h-1.5 w-full bg-blue-950 rounded-full overflow-hidden"><div className="h-full bg-blue-500 transition-all" style={{width: `${Math.max(0, (inf.durability/inf.maxDurability)*100)}%`}}></div></div>
                                           </div>
                                           <div className="grid grid-cols-2 gap-2 text-xs font-mono">
                                               <div className="bg-[var(--bg-body)] p-2 rounded border border-gray-800">
                                                   <span className="text-gray-500 block mb-0.5">{inf.id === 'dong_phu' ? 'Sức Chứa:' : inf.id === 'linh_dien' ? 'Mức Sản Lượng:' : 'Hiệu Suất:'}</span>
                                                   {(() => {
                                                       const ratio = inf.maxDurability > 0 ? (inf.durability / inf.maxDurability) : 1;
                                                       const effMult = Math.max(0.1, ratio);
                                                       if (inf.id === 'dong_phu') {
                                                           const realCap = Math.floor(inf.capacity * effMult);
                                                           return <span className="text-emerald-400 font-bold">{formatLargeNumber(realCap)} Người {effMult < 1 ? <span className="text-red-500 text-[10px] line-through ml-1" title="Sức chứa bị sụt giảm do hỏng hóc">{formatLargeNumber(inf.capacity)}</span> : ''}</span>;
                                                       }
                                                       const realEff = Math.floor(inf.efficiency * effMult);
                                                       return <span className="text-emerald-400 font-bold">{realEff}% {effMult < 1 ? <span className="text-red-500 text-[10px] line-through ml-1" title="Hiệu suất bị sụt giảm do hỏng hóc">{inf.efficiency}%</span> : ''}</span>;
                                                   })()}
                                               </div>
                                               <div className="bg-[var(--bg-body)] p-2 rounded border border-gray-800">
                                                   <span className="text-gray-500 block mb-0.5">Bảo Trì / Tháng:</span>
                                                   <span className="text-red-400 font-bold">-{inf.maintenance} LT</span>

                                           <button onClick={() => {

                                               showConfirm(`Nâng cấp [${inf.name}] lên Cấp ${inf.level + 1} sẽ tốn ${formatLargeNumber(upgradeCost)} Linh Thạch từ Quỹ. Xác nhận?`, () => {

                                                   setGameState(prev => {
                                                       const newInfra = [...(prev.sect.infrastructure || [])];
                                                       const idx = newInfra.findIndex(i => i.id === inf.id);
                                                       if (idx >= 0) {

                                                           // V13.9 FIX: Mở rộng Không Gian theo Hệ số nhân (Exponential) thay vì cộng dồn tuyến tính
                                                           newInfra[idx].maxDurability = Math.floor(newInfra[idx].maxDurability * 1.2) + 50;
                                                           newInfra[idx].durability = newInfra[idx].maxDurability;
                                                           newInfra[idx].maintenance = Math.floor(newInfra[idx].maintenance * 1.5) + 10;

                                               {inf.level >= 30 ? 'Tối Đa (Max)' : `Nâng Cấp (${formatLargeNumber(upgradeCost)})`}
                                           </button>
                                           <button onClick={() => {
                                               if (inf.durability >= inf.maxDurability) return showAlert("Công trình đang ở trạng thái hoàn hảo, không cần bảo trì!");
                                               showConfirm(`Bảo trì [${inf.name}] phục hồi Độ Bền sẽ tốn ${formatLargeNumber(repairCost)} Linh Thạch từ Quỹ. Xác nhận?`, () => {
                                                   if ((gameState.sect.wealth || 0) < repairCost) return showAlert("Quỹ không đủ Linh Thạch để bảo trì!");
                                                   setGameState(prev => {
                                                       const newInfra = [...(prev.sect.infrastructure || [])];
                                                       const idx = newInfra.findIndex(i => i.id === inf.id);
                                                       if (idx >= 0) newInfra[idx].durability = newInfra[idx].maxDurability;
                                                       const timeStr = getRealmTimeStr(prev.currentLocation, prev.knownRealms);
                                                       return {
                                                           ...prev,
                                                           sect: { ...prev.sect, wealth: prev.sect.wealth - repairCost, infrastructure: newInfra },
                                                           sectWealthHistory: [{ time: timeStr, reason: `Bảo trì ${inf.name}`, change: -repairCost, breakdown: [{ action: `Chi phí sửa chữa`, change: `-${repairCost}`, time: timeStr }] }, ...(prev.sectWealthHistory || [])].slice(0, 100)
                                                       };

                                                   showAlert(`Đã khôi phục Độ Bền ${inf.name}!`);

                                               Bảo Trì ({formatLargeNumber(repairCost)})

                       </div>

               {sectModalTab === 'members' && (
                   <div className="flex flex-col h-full animate-in slide-in-from-right-4 max-w-6xl mx-auto">
                       {/* V1.25.1: Thanh Tìm Kiếm Môn Đồ Tông Môn */}
                       <div className="relative mb-3 shrink-0">
                           <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-purple-400/50" />
                           <input 
                               type="text" 
                               placeholder="Tìm kiếm tên môn đồ, chức vụ, cảnh giới..." 
                               value={sectMemberSearch} 
                               onChange={e => setSectMemberSearch(e.target.value)} 
                               className="w-full bg-[var(--bg-surface)] border border-purple-900/40 rounded-xl pl-9 pr-8 py-2 text-xs text-gray-200 focus:outline-none focus:border-purple-500 shadow-inner" 
                           />
                           {sectMemberSearch && (
                               <button onClick={() => setSectMemberSearch('')} className="absolute right-3 top-1/2 -translate-y-1/2 text-purple-400/50 hover:text-purple-300 p-0.5" title="Xóa tìm kiếm">
                                   <X size={14} />
                               </button>
                           )}
                       </div>

                       {/* Tab Bộ Lọc */}
                       <div className={`flex gap-2 overflow-x-auto custom-scrollbar pb-3 mb-4 border-b shrink-0 ${gameState.sect.isDynasty ? 'border-yellow-900/30' : 'border-purple-900/30'}`}>
                           {(gameState.sect.isDynasty ? [
                               { id: 'all', label: 'Toàn Bộ Dân Quốc', color: 'text-gray-300 bg-gray-800' },
                               { id: 'Thái Thượng', label: 'Hộ Quốc Thần Vật', color: 'text-red-400 bg-red-900/20' },
                               { id: 'Trưởng Lão', label: 'Đại Tướng / Thái Thú', color: 'text-yellow-500 bg-yellow-900/20' },

                               { id: 'Nội Môn', label: 'Cấm Vệ Quân', color: 'text-teal-400 bg-teal-900/20' },
                               { id: 'Ngoại Môn', label: 'Quân Đội / Tướng Sĩ', color: 'text-emerald-400 bg-emerald-900/20' },

                               { id: 'Chấp Sự', label: 'Chấp Sự', color: 'text-orange-400 bg-orange-900/20' },
                               { id: 'Nội Môn', label: 'Nội Môn Đệ Tử', color: 'text-teal-400 bg-teal-900/20' },
                               { id: 'Ngoại Môn', label: 'Ngoại Môn Đệ Tử', color: 'text-emerald-400 bg-emerald-900/20' },
                               { id: 'Tạp Dịch', label: 'Tạp Dịch', color: 'text-gray-500 bg-gray-900/40' }

                              let rawMembers = [
                                  ...(gameState.sect?.npcs || []),
                                  ...((gameState.journal?.npcs || []).filter((n: any) =>

                              let members = rawMembers.filter((m: any) => {
                                  const cName = String(m.name || '').trim().toLowerCase();

                                  seenMNames.add(cName);
                                  return true;
                              });
                              
                              if (hideDeadNpcs) members = members.filter(n => !n.isDead);
                              
                              if (npcFilter !== 'all') {
                                  members = members.filter(n => new RegExp(npcFilter, 'i').test(n.role || ''));
                              }

                              if (sectMemberSearch.trim()) {
                                  const q = sectMemberSearch.toLowerCase().trim();
                                  members = members.filter((m: any) => `${m.name || ''} ${m.role || ''} ${m.level || ''}`.toLowerCase().includes(q));
                              }
                              
                              // Ưu tiên Sống -> Chết, và Role quan trọng lên đầu
                              const roleWeight = (r) => {
                                  const text = String(r).toLowerCase();
                                  if (text.includes('thái thượng')) return 100;
                                  if (text.includes('trưởng lão')) return 90;
                                  if (text.includes('khách khanh')) return 80;
                                  if (text.includes('chấp sự')) return 70;
                                  if (text.includes('nội môn')) return 60;
                                  if (text.includes('ngoại môn')) return 50;
                                  return 0; // Tạp dịch
                              };

                              members.sort((a, b) => {
                                  if (a.isDead && !b.isDead) return 1;  
                                  if (!a.isDead && b.isDead) return -1; 
                                  const wA = roleWeight(a.role);
                                  const wB = roleWeight(b.role);
                                  if (wA !== wB) return wB - wA;
                                  return (b.lastUpdated || 0) - (a.lastUpdated || 0);
                              });

                              if (members.length === 0) return (
                                  <div className="flex flex-col items-center justify-center py-20 opacity-50 px-4 text-center">
                                      <Users size={60} className="text-purple-500 mb-4"/>
                                      <p className="text-lg italic text-purple-300">Chưa có ai thuộc chức vụ này được ghi nhận.</p>

                                  <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
                                      {displayList.map((npc, i) => {

                                          // Format Thẻ Tông Môn / Quan Lại
                                          return (
