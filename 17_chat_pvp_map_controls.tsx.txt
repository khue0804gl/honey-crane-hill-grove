  // V5.57: Thuật toán Bố cục Sơ đồ dạng Phân Tầng (Radial Tree) lấy Vị trí hiện tại làm trung tâm
  const mapLayoutData = useMemo(() => {
      const locations = (gameState.encyclopedia || []).filter(e => e.type === 'location');
      
      // V7.60 FIX: Đảm bảo Vị trí hiện tại luôn tồn tại trong danh sách để thuật toán vẽ không bị sập (Tương thích save cũ)
      const currentLoc = gameState.currentLocation || 'Vùng đất vô danh';
      if (!locations.find(l => l.name === currentLoc)) {
          locations.push({ type: 'location', name: currentLoc, desc: 'Vị trí hiện tại.' });
      }

      if (locations.length === 0) return { nodes: [], edges: [], center: { cx: 0, cy: 0 }, mapW: 4000, mapH: 4000, padX: 500, padY: 500 };

      const nodeNames = locations.map(l => l.name);
      const edges = [];
      
      // Chỉ trích xuất các đường đi ĐÃ TỪNG ĐI QUA từ Ký ức Lộ trình
      Object.keys(gameState.travelRecords || {}).forEach(key => {
          const [u, v] = key.split('|');
          if (nodeNames.includes(u) && nodeNames.includes(v)) {
              const edgeId = [u, v].sort().join('|');
              if (!edges.find(e => e.id === edgeId)) edges.push({ id: edgeId, u, v });
          }
      });

      // Dùng BFS (Tìm kiếm theo chiều rộng) để phân tầng (Level)
      const levels = { [currentLoc]: 0 };
      const queue = [currentLoc];
      const adj = {};
      nodeNames.forEach(n => adj[n] = []);
      edges.forEach(e => { adj[e.u].push(e.v); adj[e.v].push(e.u); });

      while (queue.length > 0) {
          const curr = queue.shift();
          // V7.60 FIX: Thêm điều kiện kiểm tra adj[curr] tồn tại để chống crash
          if (adj[curr]) {
              adj[curr].forEach(neighbor => {
                  if (levels[neighbor] === undefined) {
                      levels[neighbor] = levels[curr] + 1;
                      queue.push(neighbor);
                  }
              });
          }
      }

      // Các điểm chưa đi tới bị vứt ra vòng ngoài cùng
      let maxLevel = Math.max(0, ...Object.values(levels).filter(l => l !== undefined));
      nodeNames.forEach(n => {
          if (levels[n] === undefined) levels[n] = maxLevel + 1;
      });

      const levelGroups = {};
      Object.entries(levels).forEach(([name, lvl]) => {
          if (!levelGroups[lvl]) levelGroups[lvl] = [];
          levelGroups[lvl].push(name);
      });

      // Không gian vũ trụ khổng lồ 4000x4000
      const mapW = 4000; const mapH = 4000; const padX = 500; const padY = 500;
      const centerX = mapW / 2;
      const centerY = mapH / 2;

      const nodes = locations.map(loc => {
          const lvl = levels[loc.name];
          const group = levelGroups[lvl];
          const idx = group.indexOf(loc.name);
          const count = group.length;

          let cx, cy;
          if (lvl === 0) {
              cx = centerX; cy = centerY;
          } else {
              const radius = lvl * 400; // Mỗi vòng cách nhau 400px (đủ rộng để đọc nhãn)
              const angleOffset = (lvl % 2 === 0) ? 0 : (Math.PI / count); // So le góc giữa các vòng
              const angle = (idx / count) * 2 * Math.PI + angleOffset;
              cx = centerX + radius * Math.cos(angle);
              cy = centerY + radius * Math.sin(angle);
          }
          return { ...loc, cx, cy, lvl };
      });

      return { nodes, edges, center: { cx: centerX, cy: centerY }, mapW, mapH, padX, padY };
  }, [gameState.encyclopedia, gameState.travelRecords, gameState.currentLocation]);

  // V4.15 & V5.57: Hàm tự động căn giữa bản đồ vào vị trí hiện tại
  const centerMapOnPlayer = () => {
      const activeGraph = getActiveWorldGraph(gameState.timelineId);
      const curLocNodeId = (gameState.currentLocationNodeId && activeGraph[gameState.currentLocationNodeId])
          ? gameState.currentLocationNodeId
          : (Object.keys(activeGraph).find(k => activeGraph[k].name === gameState.currentLocation) || Object.keys(activeGraph)[0]);
      setSelectedNodeId(curLocNodeId);

      const curX = gameState.coordinates?.x ?? 0;
      const curY = gameState.coordinates?.y ?? 0;
      const curZ = gameState.coordinates?.z ?? currentPlaneZ;
      const cacheKey = `${curX},${curY},${curZ}`;
      const targetCell = gridCache[cacheKey] || generateWorldGridCell(curX, curY, curZ, gameState.activeDLCs || [], gameState.mapSeed || 0);
      setSelectedGridCell(targetCell);
  };

  // V4.18: Hiển thị thời gian từ phút thực tế (Ký ức)
  const formatTravelTimeFromMinutes = (minutes) => {
      if (!minutes || minutes <= 0) return "Thuấn di";
      const hours = minutes / 60;
      if (hours < 1) return `${Math.ceil(minutes)} phút`;
      if (hours < 24) return `${Math.floor(hours)} giờ ${Math.ceil(minutes % 60)} phút`;
      const days = Math.floor(hours / 24);
      const remHours = Math.floor(hours % 24);
      if (days > 365) return `${Math.floor(days/365)} năm ${days%365} ngày`;
      return `${days} ngày ${remHours > 0 ? remHours + ' giờ' : ''}`;
  };
