      {isStoneConverterOpen && (

                {(() => {

                        const totalYield = (item.quantity || 1) * rate;
                        return (
                          <div key={idx} className="flex items-center justify-between p-2.5 bg-[var(--bg-body)] rounded-xl border border-[var(--border-main)]">
                            <div>
                              <div className="font-bold text-xs text-[var(--text-main)]">{item.name}</div>

                            </div>
                            <button onClick={() => {
                              setGameState(prev => {
                                const newInv = prev.inventory.filter(x => x !== item);
                                const timeStr = getRealmTimeStr(prev.currentLocation || 'Thế giới hiện tại', prev.knownRealms);
                                const newStonesHistory = [{ time: timeStr, reason: 'Rã linh thạch', change: totalYield, breakdown: [{ action: `Rã [${item.name} x${item.quantity}] thành +${totalYield} linh thạch`, change: `+${totalYield}`, time: timeStr }] }, ...(prev.stonesHistory || [])].slice(0, 30);

                                  spiritStones: (Number(prev.spiritStones) || 0) + totalYield,
                                  stonesHistory: newStonesHistory,

                              Rã Hết

              <button onClick={() => setIsDataModalOpen(false)} className="text-[var(--text-muted)] hover:text-red-500"><X size={20}/></button>
            </div>
            <div className="space-y-4">
               <div className="bg-[var(--bg-surface)] p-4 rounded-xl border border-[var(--border-main)]">
                  <div className="flex justify-between text-sm mb-2"><span className="text-[var(--text-muted)]">Bộ nhớ cục bộ (Local)</span><span className="font-bold text-[var(--text-main)]">{storageData.localKB} KB</span></div>

      {isExpansionInfoOpen && (

                   </div>
                   <button onClick={() => setIsExpansionInfoOpen(false)} className="p-2 bg-[var(--bg-body)] hover:bg-[var(--bg-hover)] border border-[var(--border-main)] rounded text-[var(--text-muted)]"><X size={20}/></button>

              <div className="flex flex-wrap justify-between items-center text-xs text-[var(--text-muted)] font-mono mb-2 pb-4 border-b border-[var(--border-main)] gap-2">
                 <span className="bg-[var(--bg-surface)] px-3 py-1.5 rounded-lg border border-[var(--border-main)]">Tổng sự kiện: <span className="text-purple-400 font-bold text-sm">{gameState.hiddenChronicles?.length || 0}</span></span>
                 <span className="bg-[var(--bg-surface)] px-3 py-1.5 rounded-lg border border-[var(--border-main)]">Sáng Thế Tự Động: <span className={isAutoWorldExpansion ? "text-emerald-400 font-bold" : "text-red-400 font-bold"}>{isAutoWorldExpansion ? "Đang Bật" : "Đang Tắt"}</span></span>

                                                        <span className="leading-relaxed">{line.replace(/^[-*•]\s*/, '').trim()}</span>
                                                    </p>
                                                );
                                            })}
                                        </div>

                        const content = splitIndex !== -1 ? strC.substring(splitIndex + 2) : strC;

                            <div key={originalIndex} className="p-4 bg-[var(--bg-surface)] border border-[var(--border-main)] rounded-xl leading-relaxed text-[var(--text-muted)] italic border-l-2 border-l-gray-600 shadow-sm mb-4 relative overflow-hidden" style={{ fontSize: `${expansionFontSize - 2}px` }}>

                                <div className={`relative z-10 ${relatedSuppressedNpcFallback ? 'blur-sm select-none opacity-30' : ''}`}>
                                    {yearTag && <span className="text-gray-400 font-mono text-[10px] font-bold mr-2 bg-gray-800 px-2 py-0.5 rounded border border-gray-700">{yearTag}</span>}

            </div>

               <div className="p-5 border-b border-[var(--border-main)] flex justify-between items-center bg-[var(--bg-surface)]">

'oldest' : 'newest')} className="flex items-center gap-1.5 text-sm font-bold text-[var(--text-muted)] hover:text-orange-500 bg-[var(--bg-body)] px-3 py-1.5 rounded-lg border border-[var(--border-main)] transition-colors">

               <div className="p-6 overflow-y-auto custom-scrollbar space-y-4">
                  {(updateSortOrder === 'newest' ? UPDATE_HISTORY : [...UPDATE_HISTORY].reverse()).map((update) => (

                        {update.date && <span className="text-[11px] text-[var(--text-muted)] font-mono bg-[var(--bg-surface)] px-2 py-0.5 rounded border border-[var(--border-main)] shadow-inner">📅 {update.date}</span>}
                      </div>
                      <p className="text-xs text-[var(--text-main)] leading-relaxed">{update.desc}</p>
                    </div>
                  ))}
               </div>
            </div>
         </div>
      )}

      {/* RECONSTRUCTION MODAL */}
      {isReconstructionModalOpen && reconstructionData && (
          <div className="fixed inset-0 z-[100] flex items-center justify-center bg-[var(--bg-overlay)] p-4">
              <div className="bg-[var(--bg-panel)] p-6 md:p-8 rounded-2xl max-w-lg w-full border border-[var(--border-main)] shadow-[0_0_50px_rgba(0,0,0,0.8)] text-center relative max-h-[90vh] overflow-y-auto custom-scrollbar">
                  <div className="flex justify-center mb-4"><Sparkles size={48} className="text-purple-500 animate-pulse"/></div>
                  <h3 className="text-2xl font-bold mb-4 text-[var(--text-bold)]">Tái Cấu Trúc Khí Vận</h3>
                  <p className="text-[var(--text-main)] mb-6 leading-relaxed">Phát hiện <span className="text-red-500 font-bold">{reconstructionData.missingTraits.length} Khí Vận</span> của nhân vật này không còn tồn tại trong kho dữ liệu Thiên Đạo ở phiên bản hiện tại.</p>
                  <div className="bg-[var(--bg-surface)] p-4 rounded-xl border border-[var(--border-main)] mb-6 text-left">
                      <h4 className="font-bold text-sm text-[var(--text-muted)] mb-2 uppercase">Khí Vận Thất Truyền:</h4>
                      <ul className="list-disc pl-5 text-red-400 text-sm mb-4 space-y-1">
                          {reconstructionData.missingTraits.map((t, i) => <li key={i}>{t.name}</li>)}
                      </ul>
                      <div className="border-t border-[var(--border-main)] pt-3">
                          <span className="text-sm text-[var(--text-muted)]">Thiên Đạo sẽ bồi thường: </span>
                          <span className="font-bold text-yellow-500">{reconstructionData.refundedPoints * 50} Linh Thạch</span>

                      setIsReconstructionModalOpen(false);
                      executeLoadProfile(reconstructionData.profile, reconstructionData.bonusLog);
                  }} className="w-full bg-purple-600 hover:bg-purple-500 text-white font-bold py-3 px-6 rounded-xl text-lg shadow-lg transition-transform hover:scale-105">Chấp Nhận & Phục Hồi</button>

      {legacyPersonalityModal && (
          <div className="fixed inset-0 z-[120] flex items-center justify-center bg-[var(--bg-overlay)] p-4 animate-in fade-in zoom-in-95">
              <div className="bg-[var(--bg-panel)] p-6 md:p-8 rounded-2xl max-w-lg w-full border border-purple-500/50 shadow-[0_0_50px_rgba(168,85,247,0.3)] relative max-h-[90vh] overflow-y-auto custom-scrollbar">
                  <div className="flex justify-center mb-4"><Brain size={48} className="text-purple-500 animate-pulse"/></div>
                  <h3 className="text-2xl font-bold text-center mb-3 text-purple-400 uppercase tracking-widest">Khắc Ấn Đạo Tâm</h3>
                  <p className="text-[var(--text-main)] text-sm mb-6 text-center leading-relaxed">Luật lệ Đa Vũ Trụ vừa được củng cố (v12.42)! Nhân vật <span className="font-bold text-white">[{legacyPersonalityModal.profile.name}]</span> của bạn đến từ chiều không gian cũ nên chưa có Tính Cách Cốt Lõi.<br/><br/>Hãy xác định rõ Bản Ngã trước khi tiếp tục. Lựa chọn này là <span className="text-red-400 font-bold">BẤT BIẾN</span>!</p>
                  
                  <div className="space-y-3 mb-6">
                      {PERSONALITIES_POOL.map(p => (
                          <div 
                              key={p.name} 
                              onClick={() => setSelectedLegacyPersonality(p.name)}
                              className={`p-4 rounded-xl border cursor-pointer transition-all ${selectedLegacyPersonality === p.name ? 'bg-purple-900/30 border-purple-500 shadow-md' : 'bg-[var(--bg-surface)] border-[var(--border-main)] hover:border-purple-400/50'}`}
                          >
                              <div className="flex items-center gap-2 mb-1.5">
                                  <div className={`w-4 h-4 rounded-full border flex items-center justify-center ${selectedLegacyPersonality === p.name ? 'bg-purple-500 border-purple-500' : 'bg-[var(--bg-body)] border-[var(--border-main)]'}`}>
                                      {selectedLegacyPersonality === p.name && <Check size={10} className="text-white"/>}
                                  </div>
                                  <span className={`font-bold ${selectedLegacyPersonality === p.name ? 'text-purple-400' : 'text-[var(--text-bold)]'}`}>{p.name}</span>
                              </div>
                              <p className="text-xs text-[var(--text-muted)] pl-6 leading-relaxed">{p.desc}</p>
                          </div>
                      ))}
                  </div>

                  <div className="flex gap-3">
                      <button onClick={() => setLegacyPersonalityModal(null)} className="flex-1 py-3 rounded-xl border border-[var(--border-main)] bg-[var(--bg-surface)] hover:bg-[var(--bg-hover)] font-bold text-[var(--text-main)]">Hủy Bỏ</button>
                      <button onClick={() => {
                          if (!selectedLegacyPersonality) return showAlert("Vui lòng chọn 1 tính cách!");
                          const currentPool = PERSONALITIES_POOL;
                          const selected = currentPool.find(p => p.name === selectedLegacyPersonality);
                          const fullDaoHeart = `${selected.name} - ${selected.desc}`;
                          
                          const newRawState = { ...legacyPersonalityModal.rawGameState, daoHeart: fullDaoHeart };
                          let newCompressedStr = null;
                          try { newCompressedStr = LZString.compressToBase64(JSON.stringify(newRawState)); } catch(e){}
                          const newProfile = { ...legacyPersonalityModal.profile, gameState: newRawState, gameStateStr: newCompressedStr || legacyPersonalityModal.profile.gameStateStr };
                          
                          setLegacyPersonalityModal(null);
                          
                          // Kiểm tra xem có cần ném qua màn hình Tái Cấu Trúc Khí Vận không
                          const savedTraits = newRawState.traits || [];
                          let validTraits = [], missingTraits = [], costDifference = 0;
                          savedTraits.forEach(st => {
                             if (!st) return;
                             const currentTrait = TRAITS_POOL.find(t => t.id === st.id || t.name === st.name);
                             if (currentTrait) { validTraits.push(currentTrait); costDifference += ((currentTrait.cost || 0) - (st.cost || 0)); } 
                             else { missingTraits.push(st); }
                          });
                          
                          let newSpiritStones = Number(newRawState.spiritStones) || 0;
                          if (costDifference !== 0) newSpiritStones = Math.max(0, newSpiritStones - (costDifference * 50));
                          
                          if (missingTraits.length > 0) {
                             const refundedPoints = missingTraits.reduce((sum, t) => sum + (Number(t.cost) || 0), 0);
                             const refundStones = refundedPoints * 50;
                             const compensatedStones = newSpiritStones + refundStones;
                             setReconstructionData({ profile: { ...newProfile, gameState: { ...newRawState, spiritStones: compensatedStones, traits: validTraits } }, missingTraits, refundedPoints, validTraits, bonusLog: null });
                             setIsReconstructionModalOpen(true); 
                          } else {
                             executeLoadProfile({ ...newProfile, gameState: { ...newRawState, spiritStones: newSpiritStones, traits: validTraits } });
                          }
                      }} className="flex-1 bg-purple-600 hover:bg-purple-500 text-white font-bold py-3 rounded-xl shadow-lg transition-transform hover:scale-105">Khắc Ấn</button>
                  </div>
              </div>
          </div>
      )}
