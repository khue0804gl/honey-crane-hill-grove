  const handleUpgradeTavern = () => {
    const tv = getTavern(gameState);
    if (tv.level >= 5) {
      return showAlert("[TỬU LÂU CỰC ĐẠI] Tửu quán của bạn đã đạt Cấp 5: Vạn Giới Đệ Nhất Lâu, chấn nhiếp chư thiên!");
    }
    const nextCfg = TAVERN_TIERS[tv.level];
    const cost = nextCfg.upgradeCost;
    if ((Number(gameState.spiritStones) || 0) < cost) {
      return showAlert(`Không đủ Linh Thạch! Cần ${cost.toLocaleString()} Linh Thạch trong túi để nâng cấp lên [${nextCfg.name}].`);
    }

    setGameState((prev: any) => {
      const curTv = getTavern(prev);
      return {
        ...prev,
        spiritStones: Math.max(0, (Number(prev.spiritStones) || 0) - cost),
        tavern: {
          ...curTv,
          level: curTv.level + 1,
          tier: nextCfg.name,
          tables: nextCfg.maxTables,
          reputation: curTv.reputation + 100,
          dailyRevenue: curTv.dailyRevenue + nextCfg.baseIncome,
          financeLogs: [
            { time: `Năm ${prev.age || 18}`, type: 'Chi', amount: cost, desc: `Nâng cấp tửu quán lên [${nextCfg.name}]` },
            ...(curTv.financeLogs || [])
          ].slice(0, 30)
        }
      };
    });
    showAlert(`[THĂNG CẤP TỬU LÂU] Chúc mừng! Tửu quán đã nâng cấp lên Cấp ${nextCfg.level}: [${nextCfg.name}]! Tăng lên ${nextCfg.maxTables} bàn khách và mở rộng quy mô.`);
  };

  const handleDepositTavernVault = (amount: number) => {

      return showAlert(`Không đủ Linh Thạch! Túi đồ chỉ có ${Number(gameState.spiritStones || 0).toLocaleString()} Linh Thạch.`);

      };

  const handleDrinkWine = (wineId: string) => {
    const tv = getTavern(gameState);
    const wine = (tv.wines || []).find((w: any) => w.id === wineId);
    if (!wine || (wine.stock || 0) <= 0) {
      return showAlert("Loại rượu này trong hầm đã cạn bình! Hãy ủ thêm mẻ mới.");
    }

    setGameState((prev: any) => {
      const curTv = getTavern(prev);
      const updatedWines = curTv.wines.map((w: any) => w.id === wineId ? { ...w, stock: Math.max(0, w.stock - 1) } : w);
      let hpBuff = 0;
      let mpBuff = 0;
      let expBuff = 0;
      let corruptionReduce = 0;

      if (wine.name === 'Trúc Diệp Thanh') {
        mpBuff = Math.floor((Number(prev.maxSpiritualEnergy) || 50) * 0.15);
      } else if (wine.name === 'Bách Thảo Quỳnh Tương') {
        hpBuff = Math.floor((Number(prev.maxHealth) || 100) * 0.25);
        corruptionReduce = 5;
      } else if (wine.name === 'Cửu Chuyển Tiên Tửu') {
        expBuff = 100;
      }

      return {
        ...prev,
        health: Math.min(Number(prev.maxHealth) || 100, (Number(prev.health) || 0) + hpBuff),
        spiritualEnergy: Math.min(Number(prev.maxSpiritualEnergy) || 50, (Number(prev.spiritualEnergy) || 0) + mpBuff),
        cultivationProgress: (Number(prev.cultivationProgress) || 0) + expBuff,
        corruption: Math.max(0, (Number(prev.corruption) || 0) - corruptionReduce),
        tavern: {
          ...curTv,
          wines: updatedWines
        }
      };
    });
    showAlert(`[THƯỞNG TỬU GIANG HỒ]
Bạn nhấp một ngụm [${wine.name}] (${wine.ageYears} năm ủ).
• Hương vị: ${wine.potency}
• Công hiệu: ${wine.effect}`);
  };

  const handleBrewWine = (wineName: string) => {
    const tv = getTavern(gameState);
    if ((Number(tv.vault) || 0) < 50) {
      return showAlert("Ngân quỹ tửu quán không đủ 50 Linh Thạch để mua men rượu và dược dẫn ủ mới!");
    }
    setGameState((prev: any) => {
      const curTv = getTavern(prev);
      const updatedWines = curTv.wines.map((w: any) => {
        if (w.name === wineName) {
          return { ...w, stock: (w.stock || 0) + 5, ageYears: (w.ageYears || 1) + 2 };
        }
        return w;
      });
      return {
        ...prev,
        tavern: {
          ...curTv,
          vault: Math.max(0, (Number(curTv.vault) || 0) - 50),
          wines: updatedWines,
          financeLogs: [
            { time: `Năm ${prev.age || 18}`, type: 'Chi', amount: 50, desc: `Ủ thêm 5 vò [${wineName}] vào hầm rượu` },
            ...(curTv.financeLogs || [])
          ].slice(0, 30)
        }
      };
    });
    showAlert(`[Ủ RƯỢU THÀNH CÔNG] Đã ủ thành công thêm 5 vò [${wineName}] lưu vào Hầm Rượu của tửu quán!`);
  };

  const handleSlaughterBeast = (beastId: string) => {
    const tv = getTavern(gameState);
    const beast = (tv.beasts || []).find((b: any) => b.id === beastId);
    if (!beast || (beast.count || 0) <= 0) {
      return showAlert("Khu chuồng trại hiện không còn linh thú loại này!");
    }

    setGameState((prev: any) => {
      const curTv = getTavern(prev);
      const updatedBeasts = curTv.beasts.map((b: any) => b.id === beastId ? { ...b, count: Math.max(0, b.count - 1) } : b);
      const meatYield = beast.tier * 8 + 10;
      const updatedIngredients = curTv.ingredients.map((ing: any) => {
        if (ing.type === 'Thịt Yêu Thú') {
          return { ...ing, quantity: (ing.quantity || 0) + meatYield };
        }
        return ing;
      });

      return {
        ...prev,
        tavern: {
          ...curTv,
          beasts: updatedBeasts,
          ingredients: updatedIngredients
        }
      };
    });
    showAlert(`[CHẾ BIẾN LINH THÚ] Đã đưa 1 con [${beast.name}] vào phòng bếp! Thu được thêm ${beast.tier * 8 + 10} Cân Thịt Yêu Thú nhập kho.`);
  };

  const handleHarvestFruit = (fruitId: string) => {
    const tv = getTavern(gameState);
    const fruit = (tv.fruits || []).find((f: any) => f.id === fruitId);
    if (!fruit) return;

    setGameState((prev: any) => {
      const curTv = getTavern(prev);
      const updatedFruits = curTv.fruits.map((f: any) => {
        if (f.id === fruitId) {
          return { ...f, count: (f.count || 0) + (f.yieldPerHarvest || 5), harvestDaysLeft: 3 };
        }
        return f;
      });
      return {
        ...prev,
        tavern: {
          ...curTv,
          fruits: updatedFruits
        }
      };
    });
    showAlert(`[THU HOẠCH LINH QUẢ] Vườn quả đã thu hoạch thêm +${fruit.yieldPerHarvest || 5} quả [${fruit.name}] tươi mọng!`);
  };
