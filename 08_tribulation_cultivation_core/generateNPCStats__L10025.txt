const generateNPCStats = (levelName, npcData = {}) => {
    // V1.22.2: NPC CỐ ĐỊNH trong Vạn Linh Ký có chỉ số định sẵn, bảo toàn nguyên gốc, CẤM auto-scale ngẫu nhiên
    if (npcData && (npcData.id || npcData.vlkId)) {
        const checkId = npcData.vlkId || npcData.id;
        if (typeof VAN_LINH_KY !== 'undefined' && (VAN_LINH_KY as any)[checkId]) {
            const vlk = (VAN_LINH_KY as any)[checkId];
            let fixedHp = vlk.hp || 100;
            let fixedMp = vlk.mp || 0;
            let fixedAtk = vlk.atk || 10;
            let fixedDef = vlk.def || 10;
            let fixedSpd = vlk.spd || 10;
            let fixedSpp = vlk.spp || 10;
            if (npcData.isNightmare) {
                const nm = getNightmareStats(vlk.canh || levelName);
                fixedAtk = Math.max(fixedAtk, nm.atk);
                fixedDef = Math.max(fixedDef, nm.def);
                fixedSpd = Math.max(fixedSpd, nm.spd);
                fixedSpp = Math.max(fixedSpp, nm.spp);
                fixedHp = Math.max(fixedHp, nm.hp);
                fixedMp = Math.max(fixedMp, nm.mp);
            }
            return {
                maxHp: fixedHp, hp: fixedHp,
                maxMp: fixedMp, mp: fixedMp,
                attack: fixedAtk,
                defense: fixedDef,
                speed: fixedSpd,
                spiritPower: fixedSpp,
                equipped: { weapon: null, armor: null, head: null, boots: null, artifact: null, accessory: null },
                activeEffects: []
            };
        }
    }
    if (npcData && npcData.name && typeof VAN_LINH_KY !== 'undefined') {
        const cleanN = String(npcData.name).trim().toLowerCase();
        const found = Object.values(VAN_LINH_KY as Record<string, any>).find(v => v && (String(v.ten).toLowerCase() === cleanN || (v.bietDanh && String(v.bietDanh).toLowerCase() === cleanN)));
        if (found) {
            let fixedHp = found.hp || 100;
            let fixedMp = found.mp || 0;
            let fixedAtk = found.atk || 10;
            let fixedDef = found.def || 10;
            let fixedSpd = found.spd || 10;
            let fixedSpp = found.spp || 10;
            if (npcData.isNightmare) {
                const nm = getNightmareStats(found.canh || levelName);
                fixedAtk = Math.max(fixedAtk, nm.atk);
                fixedDef = Math.max(fixedDef, nm.def);
                fixedSpd = Math.max(fixedSpd, nm.spd);
                fixedSpp = Math.max(fixedSpp, nm.spp);
                fixedHp = Math.max(fixedHp, nm.hp);
                fixedMp = Math.max(fixedMp, nm.mp);
            }
            return {
                maxHp: fixedHp, hp: fixedHp,
                maxMp: fixedMp, mp: fixedMp,
                attack: fixedAtk,
                defense: fixedDef,
                speed: fixedSpd,
                spiritPower: fixedSpp,
                equipped: { weapon: null, armor: null, head: null, boots: null, artifact: null, accessory: null },
                activeEffects: []
            };
        }
    }
    let stageCfg = getStageConfig(levelName);
    // V1.39 THÀNH KHÔN: Khóa cứng cảnh giới Thần Sao C'tan tối thiểu Đạo Tổ (Stage 20)
    const _checkRaceRole = (String(npcData.race || '') + ' ' + String(npcData.role || '')).toLowerCase();
    if (/(thần sao|c'tan|ctan)/.test(_checkRaceRole) && stageCfg.id < 20) {
        stageCfg = getStageConfig('Đạo Tổ');
    }
    const mult = stageCfg.statMult;
    
    // Sửa lỗi: Cân bằng Cực Cảnh - Cho tăng tiến mượt mà từ đầu thay vì nhảy vọt ở Nguyên Anh
    let cucCanhMult = stageCfg.id; 
    
    // Tỷ lệ ngẫu nhiên dao động từ 85% đến 115% sức mạnh chuẩn
    const variance = () => 0.85 + Math.random() * 0.3; 
    
    // --- MA TRẬN HỆ SỐ ---
    let hpDefMult = 1.0;
    let atkSpdMult = 1.0;
    let globalMult = 1.0;

    const raceStr = String(npcData.race || 'Nhân Tộc').toLowerCase();
    const roleStr = String(npcData.role || '').toLowerCase();
    const alignStr = String(npcData.alignment || '').toLowerCase();
    
    // FIX LỖI OBJECT: Chuyển đổi mảng Khí vận thành mảng Chuỗi (String) để so sánh
    const traitsArr = Array.isArray(npcData.traits) ? npcData.traits : [];
    const traitNames = traitsArr.map(t => typeof t === 'object' ? (t.name || '') : String(t));

    // V1.39 THÀNH KHÔN: ĐỒNG BỘ TOÀN DIỆN 27 XUẤT THÂN & CHỦNG TỘC VÀO MA TRẬN CHỈ SỐ NPC
    if (/(cự long|viễn cổ cự long|farum azula)/i.test(raceStr) || /(cự long|viễn cổ cự long)/i.test(roleStr)) {
        hpDefMult = 2.5;
        atkSpdMult = 1.8;
        globalMult *= 1.5;
    } else if (/(thần sao|c'tan|ctan)/i.test(raceStr) || /(thần sao|c'tan|ctan)/i.test(roleStr)) {
        hpDefMult = 5.0;
        atkSpdMult = 3.0;
        globalMult *= 3.0;
    } else if (/(chân long|long tộc)/i.test(raceStr) || /(chân long|long tộc)/i.test(roleStr)) {
        hpDefMult = 2.2;
        atkSpdMult = 1.6;
        globalMult *= 1.4;
    } else if (/(côn bằng)/i.test(raceStr) || /(côn bằng)/i.test(roleStr)) {
        hpDefMult = 2.0;
        atkSpdMult = 2.2;
        globalMult *= 1.5;
    } else if (/(thanh khâu|hồ tộc)/i.test(raceStr) || /(thanh khâu|hồ tộc)/i.test(roleStr)) {
        hpDefMult = 0.9;
        atkSpdMult = 1.5;
        globalMult *= 1.3;
    } else if (/(thái thản|titan|vu tộc|thập nhị vu)/i.test(raceStr) || /(thái thản|titan|vu tộc)/i.test(roleStr)) {
        hpDefMult = 3.2;
        atkSpdMult = 1.3;
        globalMult *= 1.4;
    } else if (/(bỉ mông|behemoth)/i.test(raceStr) || /(bỉ mông|behemoth)/i.test(roleStr)) {
        hpDefMult = 2.8;
        atkSpdMult = 1.4;
        globalMult *= 1.3;
    } else if (/(thao thiết)/i.test(raceStr) || /(thao thiết)/i.test(roleStr)) {
        hpDefMult = 2.5;
        atkSpdMult = 1.5;
        globalMult *= 1.3;
    } else if (/(tiên tộc|tinh linh|elf)/i.test(raceStr) || /(tiên tộc|tinh linh|elf)/i.test(roleStr)) {
        hpDefMult = 0.9;
        atkSpdMult = 1.6;
        globalMult *= 1.2;
    } else if (/(dực tộc|thiên thần|angel)/i.test(raceStr) || /(dực tộc|thiên thần|angel)/i.test(roleStr)) {
        hpDefMult = 1.2;
        atkSpdMult = 1.8;
        globalMult *= 1.3;
    } else if (/(huyết tộc|vampire|blood angel)/i.test(raceStr) || /(huyết tộc|vampire|blood angel)/i.test(roleStr)) {
        hpDefMult = 1.4;
        atkSpdMult = 1.7;
        globalMult *= 1.3;
    } else if (/(mị ma|succubus|sắc nghiệt)/i.test(raceStr) || /(mị ma|succubus|sắc nghiệt)/i.test(roleStr)) {
        hpDefMult = 1.0;
        atkSpdMult = 1.6;
        globalMult *= 1.3;
    } else if (/(ác mộng|gian kỳ|nightmare)/i.test(raceStr) || /(ác mộng|gian kỳ|nightmare)/i.test(roleStr)) {
        hpDefMult = 0.9;
        atkSpdMult = 1.9;
        globalMult *= 1.3;
    } else if (/(thi tộc|vong linh|undead)/i.test(raceStr) || /(thi tộc|vong linh|undead)/i.test(roleStr)) {
        hpDefMult = 2.0;
        atkSpdMult = 1.1;
        globalMult *= 1.2;
    } else if (/(cơ khí|cyborg|mechanicus|contingency)/i.test(raceStr) || /(cơ khí|cyborg|mechanicus)/i.test(roleStr)) {
        hpDefMult = 2.2;
        atkSpdMult = 1.6;
        globalMult *= 1.4;
    } else if (/(goblin|green tide)/i.test(raceStr) || /(goblin|green tide)/i.test(roleStr)) {
        hpDefMult = 0.8;
        atkSpdMult = 1.2;
        globalMult *= 0.9;
    } else if (/(yêu|ma|vu|thú|hung|cự nhân)/.test(raceStr)) {
        hpDefMult = 1.5;
    } else if (/(nhân|tiên|thần)/.test(raceStr)) {
        hpDefMult = 0.8;
        atkSpdMult = 1.3;
    }

    // 2. Xuất Thân / Lai Lịch (Background)
    if (/(tông|thánh địa|gia tộc|hoàng|đế|cung|điện|các)/.test(roleStr) || /(tông|thánh địa|gia tộc|hoàng|đế|cung|điện|các)/.test(alignStr)) {
        globalMult *= 1.4; // Giàu có, đồ xịn
    } else if (/(tán tu|nô lệ|khất cái|ăn mày|tạp dịch|thường dân)/.test(roleStr)) {
        globalMult *= 0.8; // Nghèo khổ
    }

    // 3. Khí Vận (Traits)
    if (traitNames.includes('Thiên Mệnh Chi Tử') || traitNames.includes('Khí Vận Chi Tử') || traitNames.includes('Khí Vận Chi Nữ') || traitNames.includes('Thiên Mệnh Ma Tôn') || traitNames.includes('Thiên Chùy Bách Luyện') || traitNames.includes('Đại Khí Vận') || traitNames.includes('Thiên Mệnh Hồng Nhan')) {
        globalMult *= 2.0; // Con cưng thiên đạo x2
    }
    if (traitNames.includes('Thiên Sát Cô Tinh') || traitNames.includes('Ách Vận Bám Thân') || traitNames.includes('Thiên Sinh Phế Vật')) {
        globalMult *= 0.6; // Xui xẻo, phế vật
    }

    const finalHpDefMult = cucCanhMult * hpDefMult * globalMult;
    const finalAtkSpdMult = cucCanhMult * atkSpdMult * globalMult;
    
    // Công thức HP Hàm Mũ kết hợp Hệ số (HP/MP là chỉ số Tổng nên vẫn giữ mult)
    const expectedMaxHp = Math.floor(100 * mult * finalHpDefMult * Math.pow(1.15, stageCfg.id - 1) * variance());
    const expectedMaxMp = Math.floor(50 * mult * cucCanhMult * globalMult * Math.pow(1.15, stageCfg.id - 1) * variance());
    
    // V2.24 ĐẠI TU: CHỈ SỐ THỂ PHÁCH NPC THEO ĐÚNG CẢNH GIỚI (BÁO LỖI DARK)
    // Hệ số đột phá x2.5 lũy tiến theo từng đại cảnh giới, tiệm cận ngưỡng trần HardCap của cảnh giới đó
    const stageHardCap = 2625 * Math.pow(stageCfg.id, 3);
    const growthExp = Math.floor(10 * Math.pow(2.5, stageCfg.id - 1));
    const stageBaseStat = Math.min(Math.floor(stageHardCap * 0.75), growthExp);
    
    // Bóc tách tiểu cảnh giới (Sơ kỳ x1.0, Trung kỳ x1.25, Hậu kỳ x1.5, Viên Mãn / Đỉnh Phong x1.8)
    let minorMult = 1.0;
    const lvlStr = String(npcData.level || levelName || '').toLowerCase();
    if (lvlStr.includes('trung kỳ') || lvlStr.includes('giai đoạn 2')) minorMult = 1.25;
    else if (lvlStr.includes('hậu kỳ') || lvlStr.includes('giai đoạn 3')) minorMult = 1.5;
    else if (lvlStr.includes('viên mãn') || lvlStr.includes('đỉnh phong') || lvlStr.includes('cực hạn')) minorMult = 1.8;
    else {
        const tangMatch = lvlStr.match(/tầng\s+(\d+)/);
        if (tangMatch) {
            const tang = parseInt(tangMatch[1]);
            minorMult = 1.0 + Math.min(0.8, Math.max(0, (tang - 1) * 0.1));
        }
    }
    
    const realmBaseline = Math.floor(stageBaseStat * minorMult);
    const baseAtk = Math.max(10, Math.floor(realmBaseline * atkSpdMult * globalMult * variance()));
    const baseDef = Math.max(10, Math.floor(realmBaseline * hpDefMult * globalMult * variance()));
    const baseSpd = Math.max(10, Math.floor(realmBaseline * atkSpdMult * globalMult * variance()));
    const baseSpp = Math.max(10, Math.floor(realmBaseline * globalMult * variance()));

    let finalAtk = baseAtk, finalDef = baseDef, finalSpd = baseSpd, finalSpp = baseSpp;
    let expectedFinalHp = expectedMaxHp, expectedFinalMp = expectedMaxMp;
    if (npcData.isNightmare) {
        const nm = getNightmareStats(levelName);
        finalAtk = Math.max(finalAtk, nm.atk);
        finalDef = Math.max(finalDef, nm.def);
        finalSpd = Math.max(finalSpd, nm.spd);
        finalSpp = Math.max(finalSpp, nm.spp);
        expectedFinalHp = Math.max(expectedFinalHp, nm.hp);
        expectedFinalMp = Math.max(expectedFinalMp, nm.mp);
    }

    const baseStats = npcData.baseStats;

    if (baseStats) {
        return {
            maxHp: Math.max(baseStats.maxHp || 100, expectedFinalHp),
            hp: Math.max(baseStats.maxHp || 100, expectedFinalHp), // Hồi full máu khi lên cấp
            maxMp: Math.max(baseStats.maxMp || 50, expectedFinalMp),
            mp: Math.max(baseStats.maxMp || 50, expectedFinalMp),
            attack: Math.max(baseStats.attack || 10, finalAtk),
            defense: Math.max(baseStats.defense || 10, finalDef),
            speed: Math.max(baseStats.speed || 10, finalSpd),
            spiritPower: Math.max(baseStats.spiritPower || 10, finalSpp),
            equipped: baseStats.equipped || { weapon: null, armor: null, head: null, boots: null, artifact: null, accessory: null },
            activeEffects: baseStats.activeEffects || []
        };
    }
    
    return {
        maxHp: expectedFinalHp, hp: expectedFinalHp,
        maxMp: expectedFinalMp, mp: expectedFinalMp,
        attack: finalAtk,
        defense: finalDef,
        speed: finalSpd,
        spiritPower: finalSpp,
        equipped: { weapon: null, armor: null, head: null, boots: null, artifact: null, accessory: null },
        activeEffects: []
    };
};
