const getGridDistanceLi = (c1: { x: number; y: number; z?: number }, c2: { x: number; y: number; z?: number }) => {
  const dx = Math.abs(c1.x - c2.x);
  const dy = Math.abs(c1.y - c2.y);
  const dz = Math.abs((c1.z || 0) - (c2.z || 0));
  return Math.max(1, Math.round(Math.sqrt(dx * dx + dy * dy) * 10 + dz * 50));
};
