const getBaiGiaBreakthroughCost = (stageId: number, minorLevel: number, isMajor: boolean) => {
  if (stageId <= 1) return 5000;
  const baseCosts: Record<number, number> = {
    2: 5000, 3: 20000, 4: 100000, 5: 500000, 6: 2000000,
    7: 10000000, 8: 50000000, 9: 200000000, 10: 1000000000,
    11: 3000000000, 12: 10000000000, 13: 30000000000, 14: 100000000000, 15: 300000000000
  };
  const base = baseCosts[stageId] || (stageId >= 15 ? 500000000000 : 5000);
  return isMajor ? base * 5 : Math.floor(base * (1 + (minorLevel - 1) * 0.25));
};

const getBaiGiaUpgradeCost = (currentLevel: number) => {
  const lvl = Math.max(1, currentLevel || 1);
  return Math.floor(10000 * lvl * Math.pow(1.5, lvl - 1));
};

const getItemPerkInfo = (perkName: string) => {
  if (!perkName) return { name: 'Đặc Tính Vô Danh', desc: 'Hiệu quả gia trì thần bí cho pháp bảo.', category: 'Bổ Trợ', color: 'text-purple-300 border-purple-500/30 bg-purple-900/20' };
  const cleanName = String(perkName).trim();
  if (ITEM_PERK_DICTIONARY[cleanName]) return ITEM_PERK_DICTIONARY[cleanName];
  for (const k of Object.keys(ITEM_PERK_DICTIONARY)) {
    if (cleanName.toLowerCase().includes(k.toLowerCase()) || k.toLowerCase().includes(cleanName.toLowerCase())) {
      return { ...ITEM_PERK_DICTIONARY[k], name: cleanName };
    }
  }
  return {
    name: cleanName,
    desc: `Gia trì đặc tính [${cleanName}] vào khí cụ. Tăng cường uy lực thực chiến và cộng hưởng cùng đạo pháp của chủ nhân.`,
    category: 'Bổ Trợ',
    color: 'text-purple-300 border-purple-500/30 bg-purple-900/20'
  };
};
