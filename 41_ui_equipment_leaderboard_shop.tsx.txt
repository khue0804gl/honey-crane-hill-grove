                <button onClick={() => setGlobalBoardTab('arena')} className={`flex-1 py-3 text-sm md:text-base font-bold flex items-center justify-center gap-2 transition-all ${globalBoardTab === 'arena' ? 'text-red-400 border-b-2 border-red-400 bg-red-900/10' : 'text-[var(--text-muted)] hover:text-[var(--text-main)]'}`}><Swords size={18}/> Đấu Trường Sinh Tử</button>
                <button onClick={() => setGlobalBoardTab('chapter')} className={`flex-1 py-3 text-sm md:text-base font-bold flex items-center justify-center gap-2 transition-all ${globalBoardTab === 'chapter' ? 'text-emerald-400 border-b-2 border-emerald-400 bg-emerald-900/10' : 'text-[var(--text-muted)] hover:text-[var(--text-main)]'}`}><BookOpen size={18}/> Cần Mẫn Tu Đạo</button>
            </div>

            {globalBoardTab === 'chapter' && (
                <div className="flex bg-[var(--bg-body)] justify-center gap-4 py-2 border-b border-[var(--border-main)] shrink-0">
                    <button onClick={() => setGlobalBoardSubTab('week')} className={`px-4 py-1.5 text-xs font-bold rounded-full transition-all ${globalBoardSubTab === 'week' ? 'bg-emerald-600 text-white' : 'bg-[var(--bg-surface)] text-[var(--text-muted)] border border-[var(--border-main)]'}`}>Tuần Này</button>
                    <button onClick={() => setGlobalBoardSubTab('month')} className={`px-4 py-1.5 text-xs font-bold rounded-full transition-all ${globalBoardSubTab === 'month' ? 'bg-emerald-600 text-white' : 'bg-[var(--bg-surface)] text-[var(--text-muted)] border border-[var(--border-main)]'}`}>Tháng Này</button>
                </div>
            )}

            <div className="flex-1 overflow-y-auto p-4 custom-scrollbar bg-[var(--bg-body)] relative">
                {globalBoardData.isLoading ? (
                    <div className="h-full flex flex-col items-center justify-center text-yellow-500/50">
                        <Activity size={40} className="animate-spin mb-4"/>
                        <p className="font-bold">Đang kết nối Đám Mây Thiên Đạo...</p>
                    </div>
                ) : (
                    <div className="space-y-3">
                        {(() => {
                            let list = [];
                            let valLabel = "";
                            let colorTheme = "";
                            
                            if (globalBoardTab === 'tower') {
                                list = globalBoardData.tower;
                                valLabel = "Tầng";
                                colorTheme = "text-indigo-400";
                            } else if (globalBoardTab === 'arena') {
                                list = globalBoardData.arena;
                                valLabel = "Hạng";
                                colorTheme = "text-red-400";
                            } else if (globalBoardTab === 'chapter') {
                                colorTheme = "text-emerald-400";
                                valLabel = "Chương";
                                const curW = getCurrentWeekStr();
                                const curM = getCurrentMonthStr();
                                if (globalBoardSubTab === 'week') {
                                    list = globalBoardData.chapters.filter(r => r.weekStr === curW && (r.weekCount || 0) > 0).sort((a,b) => b.weekCount - a.weekCount);
                                } else {
                                    list = globalBoardData.chapters.filter(r => r.monthStr === curM && (r.monthCount || 0) > 0).sort((a,b) => b.monthCount - a.monthCount);
                                }
                            }

                            if (list.length === 0) return <div className="text-center py-20 text-[var(--text-muted)] italic">Bảng Phong Thần hiện đang bỏ trống. Chờ người khai mở!</div>;

                            return list.slice(0, 50).map((p, i) => {
                                const rank = i + 1;
                                let rankStyle = 'bg-[var(--bg-surface)] border-[var(--border-main)] text-[var(--text-muted)]';
                                let rankIcon = <span className="font-mono text-lg font-black">{rank}</span>;
                                
                                if (rank === 1) { rankStyle = 'bg-yellow-900/20 border-yellow-500/50 text-yellow-500 shadow-[0_0_15px_rgba(234,179,8,0.2)]'; rankIcon = <Crown size={24} className="text-yellow-400"/>; }
                                else if (rank === 2) { rankStyle = 'bg-gray-800/50 border-gray-400/50 text-gray-300'; rankIcon = <Crown size={20} className="text-gray-300"/>; }
                                else if (rank === 3) { rankStyle = 'bg-orange-900/20 border-orange-700/50 text-orange-600'; rankIcon = <Crown size={20} className="text-orange-500"/>; }

                                const isMe = p.uid === user?.uid && p.profileId === currentProfileId;

                                let displayValue = 0;
                                if (globalBoardTab === 'tower') displayValue = p.towerFloor;
                                else if (globalBoardTab === 'arena') displayValue = p.rankPosition;
                                else if (globalBoardTab === 'chapter') displayValue = globalBoardSubTab === 'week' ? p.weekCount : p.monthCount;

                                return (
                                    <div key={i} className={`flex items-center p-3 rounded-xl border transition-all ${isMe ? 'border-teal-500 bg-teal-900/10' : rankStyle} ${rank <= 3 ? 'py-4' : ''}`}>
                                        <div className={`w-12 h-12 shrink-0 flex items-center justify-center ${rank <= 3 ? '' : 'bg-[var(--bg-body)] rounded-full border border-[var(--border-main)]'}`}>
                                            {rankIcon}
                                        </div>
                                        
                                        <div className="flex-1 px-4 overflow-hidden">
                                            <div className="flex items-center gap-2">
                                                <span className={`font-bold text-lg truncate ${rank === 1 ? 'text-yellow-500' : rank === 2 ? 'text-gray-200' : rank === 3 ? 'text-orange-500' : 'text-[var(--text-bold)]'}`}>
                                                    {p.name}
                                                </span>
                                                {isMe && <span className="text-[11px] leading-5 bg-teal-600 text-white px-1.5 py-0.5 rounded uppercase" style={{ letterSpacing: '0.03em' }}>Bản Tôn</span>}
                                            </div>
                                            <div className="text-[11px] text-[var(--text-muted)] flex items-center gap-2 mt-0.5">
                                                <span className="bg-[var(--bg-body)] px-1.5 py-0.5 rounded border border-[var(--border-main)]">{p.level}</span>

                                            </div>

                                        <div className="shrink-0 flex flex-col items-end pr-2">
                                            <span className={`text-2xl font-black font-mono ${colorTheme}`}>{displayValue}</span>
                                            <span className="text-[10px] uppercase font-bold text-[var(--text-muted)] tracking-widest">{valLabel}</span>
                                        </div>
                                    </div>
                                );
                            });
                        })()}
                    </div>
                )}
            </div>
          </div>
        </div>
      )}
      
      {/* V2.16: CỬA HÀNG BẠI GIA BIG MODAL (ETSUO ĐỀ XUẤT) */}
      {isBaiGiaShopOpen && (() => {
        const curSysLvl = gameState.baiGiaSystemLevel || 1;
        const catalog = getBaiGiaShopCatalog(curSysLvl);
        const filteredItems = catalog.filter(it => {
          if (baiGiaShopTab === 'equipment') return it.type === 'equipment';
          if (baiGiaShopTab === 'pills') return it.type === 'pill';
          if (baiGiaShopTab === 'maps') return it.type === 'map';
          if (baiGiaShopTab === 'intel') return it.type === 'intel';
          if (baiGiaShopTab === 'physiques') return it.type === 'physique';
          return true;
        });

        const handleBuyItem = (item: BaiGiaShopItem) => {
          if ((gameState.baiGiaPoints || 0) < item.cost) {
            showAlert(`Chưa đủ Điểm Bại Gia! Cần ${formatLargeNumber(item.cost)} ĐBG để đổi [${item.name}] (Hiện có: ${formatLargeNumber(gameState.baiGiaPoints || 0)} ĐBG).`);
            return;
          }
          if (curSysLvl < item.reqLevel) {
            showAlert(`Hệ Thống chưa đạt cấp yêu cầu! Cần Hệ Thống Cấp ${item.reqLevel} để mở khóa vật phẩm này (Hiện tại: Cấp ${curSysLvl}).`);
            return;
          }

          showConfirm(`Xác nhận tiêu hao ${formatLargeNumber(item.cost)} Điểm Bại Gia để mua [${item.name}]?`, () => {
            setGameState(prev => {
              const newPoints = (prev.baiGiaPoints || 0) - item.cost;
              let updatedInv = [...(prev.inventory || [])];
              let updatedTraits = [...(prev.traits || [])];
              let updatedKnownLocations = [...(prev.knownLocations || [])];
              let extraHp = 0, extraMp = 0, extraAtk = 0, extraDef = 0, extraSpd = 0, extraSp = 0, extraEnlight = 0, extraTalent = 0;

              if (item.type === 'equipment') {
                updatedInv.push({
                  name: item.name,
                  category: 'Trang Bị',
                  rarity: item.rarity,
                  desc: item.desc,
                  quantity: 1,
                  perks: item.perks || [],
                  ...(item.stats || {})
                });
              } else if (item.type === 'pill') {
                if (item.stats?.healHp) {
                  extraHp = Number(item.stats.maxHealthBonus) || 0;
                }
                if (item.stats?.enlightenment) {
                  extraEnlight = Number(item.stats.enlightenment) || 0;
                  extraTalent = Number(item.stats.talent) || 0;
                }
                if (item.stats?.maxLifespanBonus) {
                  extraHp += 200;
                }
                updatedInv.push({
                  name: item.name,
                  category: 'Đan Dược',
                  rarity: item.rarity,
                  desc: item.desc,
                  quantity: 1
                });
              } else if (item.type === 'map') {
                if (item.actionEffect === 'unlock_ancient_realm' && !updatedKnownLocations.includes('Bí Cảnh Thượng Cổ')) {
                  updatedKnownLocations.push('Bí Cảnh Thượng Cổ');
                } else if (item.actionEffect === 'unlock_immortal_cave' && !updatedKnownLocations.includes('Động Phủ Chân Tiên')) {
                  updatedKnownLocations.push('Động Phủ Chân Tiên');
                } else if (item.actionEffect === 'unlock_void_stars' && !updatedKnownLocations.includes('Vực Ngoại Hư Không')) {
                  updatedKnownLocations.push('Vực Ngoại Hư Không');
                }
                extraSp += 200;
              } else if (item.type === 'intel') {
                extraEnlight += 30;
                extraSp += 100;
              } else if (item.type === 'physique') {
                let traitId = 7020;
                if (item.actionEffect === 'trait_7021') traitId = 7021;
                if (item.actionEffect === 'trait_7022') traitId = 7022;

                if (updatedTraits.some(t => t.id === traitId || t.name === item.name)) {
                  showAlert(`Đạo hữu đã thức tỉnh thể chất [${item.name}] rồi, không thể nhận trùng!`);
                  return prev;
                }

                updatedTraits.push({
                  id: traitId,
                  name: item.name,
                  rarity: 'Đỏ',
                  cost: 0,
                  desc: item.desc,
                  isDynamic: true
                });

                if (traitId === 7020) {
                  extraHp += Math.floor((prev.maxHealth || 100) * 0.3);
                  extraAtk += Math.floor((prev.attack || 10) * 0.3);
                  extraDef += Math.floor((prev.defense || 10) * 0.3);
                } else if (traitId === 7021) {
                  extraAtk += 2000;
                  extraDef += 2500;
                  extraHp += 20000;
                } else if (traitId === 7022) {
                  extraAtk += 5000;
                  extraDef += 5000;
                  extraSp += 5000;
                  extraHp += 50000;
                }
              }

              return {
                ...prev,
                baiGiaPoints: newPoints,
                inventory: updatedInv,
                traits: updatedTraits,
                knownLocations: updatedKnownLocations,
                maxHealth: (prev.maxHealth || 100) + extraHp,
                health: (prev.health || 100) + extraHp,
                attack: (prev.attack || 10) + extraAtk,
                defense: (prev.defense || 10) + extraDef,
                speed: (prev.speed || 10) + extraSpd,
                spiritPower: (prev.spiritPower || 10) + extraSp,
                enlightenment: (prev.enlightenment || 10) + extraEnlight,
                talent: (prev.talent || 10) + extraTalent,
                logs: [...(prev.logs || []), {
                  type: 'system',
                  text: `💰 [Cửa Hàng Bại Gia] Giao dịch thành công! Tiêu hao ${formatLargeNumber(item.cost)} ĐBG đổi được [${item.name}]!`,
                  chapter: prev.chapter || 1
                }]
              };
            });
            showAlert(`Mua thành công [${item.name}]! Tiêu hao ${formatLargeNumber(item.cost)} Điểm Bại Gia.`);
          });
        };

        return (
          <div className="fixed inset-0 z-[120] flex flex-col bg-[var(--bg-body)] animate-in fade-in duration-200">
            {/* Header */}
            <div className="p-3 md:p-4 border-b border-amber-900/50 bg-gradient-to-b from-amber-950/60 to-transparent flex justify-between items-center shrink-0 shadow-lg z-10">
              <div className="flex items-center gap-3">
                <div className="bg-amber-900/40 p-2.5 rounded-lg border border-amber-700/50 shadow-inner">
                  <Coins className="text-amber-400" size={26} />
                </div>
                <div>
                  <div className="flex items-center gap-2">
                    <h3 className="text-lg md:text-xl font-bold text-amber-300 uppercase tracking-widest">
                      Cửa Hàng Bại Gia • Thần Hào Các
                    </h3>
                    <span className="text-xs bg-amber-900/70 text-amber-300 border border-amber-500/50 px-2 py-0.5 rounded font-bold">
                      Hệ Thống Cấp {curSysLvl}
                    </span>
                  </div>
                  <p className="text-[11px] md:text-xs text-amber-300/70 mt-0.5">
                    Tiêu xài hoang phí • Tán tài chứng đạo • Vòng 2 do Etsuo đề xuất
                  </p>
                </div>
              </div>
              <div className="flex items-center gap-3">
                <div className="text-right hidden sm:block">
                  <div className="text-[11px] text-[var(--text-muted)]">Số dư hiện có</div>
                  <div className="text-sm font-bold text-amber-400">{formatLargeNumber(gameState.baiGiaPoints || 0)} ĐBG</div>
                </div>
                <button
                  onClick={() => setIsBaiGiaShopOpen(false)}
                  className="p-1.5 md:p-2 rounded-lg bg-gray-800/80 hover:bg-gray-700 text-gray-300 hover:text-white transition-colors"
                >
                  <X size={20} />
                </button>
              </div>
            </div>

            {/* Navigation Tabs */}
            <div className="flex border-b border-amber-900/40 bg-[var(--bg-panel)] px-2 md:px-4 shrink-0 overflow-x-auto gap-1 py-1">
              {[
                { id: 'equipment', label: 'Trang Bị', icon: Swords },
                { id: 'pills', label: 'Đan Dược', icon: FlaskRound },
                { id: 'maps', label: 'Bản Đồ & Bí Cảnh', icon: Map },
                { id: 'intel', label: 'Tình Báo & Tin Đồn', icon: BookOpen },
                { id: 'physiques', label: 'Thể Chất Cực Phẩm', icon: Crown },
              ].map(tab => {
                const Icon = tab.icon;
                const isActive = baiGiaShopTab === tab.id;
                return (
                  <button
                    key={tab.id}
                    onClick={() => setBaiGiaShopTab(tab.id as any)}
                    className={`flex items-center gap-1.5 px-3 py-2 text-xs md:text-sm font-bold rounded-lg transition-colors whitespace-nowrap ${
                      isActive

                        : 'text-[var(--text-muted)] hover:text-amber-200 hover:bg-amber-900/20'

                    <Icon size={16} />
                    <span>{tab.label}</span>

            </div>

              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3 md:gap-4">
                {filteredItems.map(item => {
                  const isLocked = curSysLvl < item.reqLevel;
                  const canAfford = (gameState.baiGiaPoints || 0) >= item.cost;
                  const rarityColors: Record<string, string> = {
                    'Trắng': 'border-gray-500/30 text-gray-300 bg-gray-900/20',
                    'Xanh': 'border-emerald-500/30 text-emerald-300 bg-emerald-950/20',

                    'Vàng': 'border-yellow-500/30 text-yellow-300 bg-yellow-950/20',
                    'Cam': 'border-orange-500/30 text-orange-300 bg-orange-950/20',
                    'Đỏ': 'border-red-500/40 text-red-300 bg-red-950/30 shadow-[0_0_15px_rgba(239,68,68,0.2)]'

                          <h4 className="font-bold text-sm md:text-base leading-tight">
                            {item.name}

                            {item.rarity}

                          {item.desc}

                          <div className="flex flex-wrap gap-1.5 mb-3">
                            {Object.entries(item.stats).map(([k, v]) => (
                              <span key={k} className="text-[11px] font-mono bg-black/40 px-2 py-0.5 rounded text-amber-200 border border-white/5">
                                {k}: +{typeof v === 'number' ? formatLargeNumber(v) : v}
                              </span>
                            ))}
                          </div>
                        )}
                        {item.perks && item.perks.length > 0 && (
                          <div className="flex flex-wrap gap-1 mb-3">
                            {item.perks.map(p => (
                              <span key={p} className="text-[10px] font-bold bg-purple-900/40 text-purple-300 px-1.5 py-0.5 rounded border border-purple-500/30">
                                ★ {p}
