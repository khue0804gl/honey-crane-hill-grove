  const runDiagnostics = async () => {
      setIsDiagnosing(true);
      setDiagnosticLogs([]);
      let currentLogs = [];

      const addLog = async (status, msg, percent, stepText) => {
          if (stepText) setDiagnosticProgress({ percent, text: stepText });
          await new Promise(r => setTimeout(r, 600)); // Tạo độ trễ đọc dữ liệu
          currentLogs = [...currentLogs, { status, msg }];
          setDiagnosticLogs(currentLogs);
      };

      setDiagnosticProgress({ percent: 5, text: 'Khởi tạo luồng kiểm tra hệ thống...' });
      await new Promise(r => setTimeout(r, 400));

      // 1. Check API Key
      await addLog('success', 'KẾT NỐI AI: Thiên Đạo đã được liên kết ngầm thành công (Bỏ qua rà soát Key cứng).', 20, 'Kiểm tra đường truyền kết nối AI...');

      // 2. Check Firebase Config
      let fbMsg = 'ĐÁM MÂY FIREBASE: Cấu hình hệ thống hợp lệ.';
      let fbStatus = 'success';
      if (myFirebaseConfig.apiKey === "ĐIỀN_API_KEY_CỦA_ANH" && typeof __firebase_config === 'undefined') {
          fbStatus = 'warn';
          fbMsg = 'CẢNH BÁO ĐÁM MÂY: Firebase đang dùng dữ liệu ảo. Hãy điền thông tin thật nếu mang game ra host ngoài.';
      }
      await addLog(fbStatus, fbMsg, 40, 'Xác thực cấu hình Đám Mây Firebase...');

      // 3. Check State Integrity
      let stateMsg = 'NHỤC THÂN: Cấu trúc dữ liệu nhân vật ổn định.';
      let stateStatus = 'success';
      if (!gameState || typeof gameState !== 'object') {
          stateStatus = 'error';
          stateMsg = 'LỖI NHỤC THÂN: Biến gameState (nhân vật) bị rỗng hoặc hỏng cấu trúc!';
      }
      await addLog(stateStatus, stateMsg, 60, 'Phân tích dữ liệu Nhục Thân (State)...');

      // 4. Check Storage Limits
      try {
          const storage = getStorageData();
          if (Number(storage.localPct) >= 90) {
              await addLog('warn', `BỘ NHỚ TRÀN: LocalStorage đã chiếm ${storage.localPct}%. Dễ gây lỗi liệt nút Save/Load.`, 75, 'Đo lường sức chứa Bộ Nhớ (Storage)...');
          } else {
              await addLog('success', `BỘ NHỚ TRÌNH DUYỆT: An toàn (${storage.localPct}% / 5MB).`, 75, 'Đo lường sức chứa Bộ Nhớ (Storage)...');
          }
      } catch (e) {
          await addLog('error', 'LỖI BỘ NHỚ: Không thể đọc dung lượng lưu trữ. ' + e.message, 75, 'Đo lường sức chứa Bộ Nhớ (Storage)...');
      }

      // 5. Thử nghiệm Logic Lõi
      try {
          const testStage = getStageConfig('Đại Thừa Kỳ');
          if (!testStage || isNaN(testStage.statMult)) throw new Error("Cảnh giới trả về NaN");
          await addLog('success', 'LÕI THIÊN ĐẠO: Hàm tính toán Cảnh Giới & Chỉ số (StatMult) hoạt động trơn tru.', 90, 'Chạy giả lập Thuật toán Lõi...');
      } catch(e) {
          await addLog('error', 'LỖI LÕI THIÊN ĐẠO: Thuật toán hệ số sức mạnh bị sập. ' + e.message, 90, 'Chạy giả lập Thuật toán Lõi...');
      }

      // 6. Test UI / Giao diện & Nút bấm theo Ngữ Cảnh
      setDiagnosticProgress({ percent: 95, text: 'Rà soát liên kết Giao diện (UI) & Nút bấm...' });
      await new Promise(r => setTimeout(r, 600));

      if (!gameStarted) {
          await addLog('info', 'GIAO DIỆN: Đạo hữu chưa Bước vào Luân Hồi. Các nút Đấu Trường, Bách Khoa, Sổ Tay đang được ẩn đi một cách hợp lý (Trạng thái Bình thường).', 100, 'Hoàn tất rà soát toàn diện!');
      } else {
          try {
              const modals = {
                  "Cẩm Nang Tu Tiên": typeof isHelpOpen === 'boolean',
                  "Lịch Sử Cập Nhật": typeof isUpdateHistoryOpen === 'boolean',
                  "Tu Tiên Bách Khoa": typeof isEncyclopediaOpen === 'boolean',
                  "Chiến Trường Xếp Hạng": typeof isArenaOpen === 'boolean',
                  "Sổ Tử Thần": typeof isDeathNoteOpen === 'boolean',
                  "Kho Dữ Liệu": typeof isDataModalOpen === 'boolean'
              };
              const missing = Object.entries(modals).filter(([_, exists]) => !exists).map(([name]) => name);
              if (missing.length > 0) {
                  await addLog('error', `❌ LỖI NÚT BẤM: Khuyết thiếu biến trạng thái cho bảng [${missing.join(', ')}]. Nút bấm sẽ bị liệt!`, 100, 'Hoàn tất rà soát toàn diện!');
              } else {
                  await addLog('success', '✅ GIAO DIỆN & NÚT BẤM: 100% Nguyên vẹn. Mọi bảng chức năng đều đã sẵn sàng.', 100, 'Hoàn tất rà soát toàn diện!');
              }
          } catch (e) {
              await addLog('error', 'LỖI GIAO DIỆN: Quá trình rà soát bị gián đoạn.', 100, 'Hoàn tất rà soát toàn diện!');
          }
      }

      setTimeout(() => {
          setDiagnosticProgress({ percent: 0, text: '' });
          setIsDiagnosing(false);
      }, 2000);
  };
