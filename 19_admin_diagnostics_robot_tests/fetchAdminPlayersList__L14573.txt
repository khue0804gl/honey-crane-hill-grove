  const fetchAdminPlayersList = async () => {
      setIsAdminFetching(true);
      try {
          const snap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'global_records'));
          let list = [];
          snap.forEach(doc => list.push({ id: doc.id, ...doc.data() }));
          list.sort((a,b) => (b.lastUpdated || 0) - (a.lastUpdated || 0));
          setAdminPlayersList(list.slice(0, 300));
      } catch(e) { showAlert("Lỗi tải danh sách người chơi: " + e.message); }
      setIsAdminFetching(false);
  };
