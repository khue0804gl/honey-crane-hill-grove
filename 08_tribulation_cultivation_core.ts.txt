const getTribulationTier = (intendedStageId, karma, spiritualRootName) => {
  const id = Number(intendedStageId) || 2;
  let baseWaves = 9;
  let tierName = 'Tam Cửu Thiên Lôi';
  if (id >= 15)      { baseWaves = 81; tierName = 'Cửu Cửu Thiên Kiếp'; }
  else if (id >= 11) { baseWaves = 54; tierName = 'Lục Cửu Thiên Kiếp'; }
  else if (id >= 8)  { baseWaves = 36; tierName = 'Tứ Cửu Thiên Kiếp'; }
  else if (id >= 5)  { baseWaves = 27; tierName = 'Nhị Thập Thất Thiên Lôi'; }

  const k = Number(karma) || 0;
  const karmaBonus = k < 0 ? Math.min(27, Math.floor(Math.abs(k) / 10) * 3) : 0;

  const rootName = String(spiritualRootName || '');
  const isChaos = /hỗn độn|hủy diệt/i.test(rootName);
  const isHeaven = /thiên linh/i.test(rootName);
  const rootBonus = isChaos ? 9 : 0;

  let totalWaves = baseWaves + karmaBonus + rootBonus;
  if (isHeaven) totalWaves -= 9;
  totalWaves = Math.max(9, totalWaves);

  let damageType = 'Thái Ất Thần Lôi';
  if (isChaos && k <= -40) damageType = 'Hỗn Độn Diệt Thế Hắc Lôi';
  else if (k <= -80)       damageType = 'Diệt Thế Hắc Lôi';
  else if (k <= -40)       damageType = 'Hủy Diệt Tử Lôi';
  else if (isChaos)        damageType = 'Hỗn Độn Diệt Thế Lôi';
  else if (k >= 80)        damageType = 'Tẩy Tủy Cam Lôi';

  return { tierName, totalWaves, damageType, isChaos, isHeaven, karmaBonus, rootBonus };
};

const getTribulationWaveDamage = (intendedStageId, maxHp, bgId = 'bg_0') => {
  const id = Number(intendedStageId) || 2;
  const hp = Math.max(1, Number(maxHp) || 1);
  // V1.39 DARK & SƯ PHỤ: Cân bằng giảm mạnh sát thương lôi kiếp 5 cảnh giới đầu chống chết sốc
  let pct;
  if (id <= 2) {
    pct = 0.035; // 3.5% HP/đạo cho Phàm Nhân -> Luyện Khí
  } else if (id === 3) {
    pct = 0.045; // 4.5% HP/đạo cho Trúc Cơ
  } else if (id === 4) {
    // V1.49: HÓA HÌNH LÔI KIẾP X2 CHO YÊU TỘC THUẦN CHỦNG (Góp ý Ethan195)
    pct = (typeof bgId !== 'undefined' && isPureYeuTocBg(bgId)) ? 0.11 : 0.055;
  } else if (id === 5) {
    pct = 0.065; // 6.5% HP/đạo cho Nguyên Anh
  } else {
    pct = 0.07 + Math.min(0.06, Math.max(0, id - 6) * 0.005);
  }
  let resDmg = Math.max(1, Math.floor(hp * pct));
  // V1.79: THIÊN ĐẠO KHƯỚC TỪ (Góp ý Kha) - Lôi kiếp x2 sát thương chuẩn
  if (typeof bgId !== 'undefined' && bgId === 'bg_thien_dao_khuoc_tu') {
    resDmg = resDmg * 2;
  }
  return resDmg;
};

const parseDeclaredTribulationWaves = (text) => {
  const t = String(text || '').toLowerCase();
  let m = t.match(/(\d{1,3})\s*(?:đạo|đợt|tia|luồng|nhịp|đòn)/);
  if (!m) m = t.match(/(?:lãnh|chịu|đỡ|gồng|chống|hứng|nhận|vượt|lướt qua|đón)\s*(\d{1,3})/);
  if (!m) m = t.match(/(\d{1,3})\s*(?:lôi|sét|kiếp)/);
  if (!m) return 0;
  const n = parseInt(m[1], 10);
  return (isFinite(n) && n > 0) ? n : 0;
};

// V1.24: Frontend quyet dinh CA luot do kiep — bao nhieu dao, bao nhieu sat

const resolveTribulationTurn = (trib, health, maxHp, declaredWaves, traits = [], bgId = 'bg_0', innerWorld = null, defense = 0, cultivationPath = 'tu_tien', spiritualRoot = null, equippedCoreMethod = null) => {
  const total = Math.max(1, Number(trib && trib.totalWaves) || 1);
  const cur = Math.max(0, Number(trib && trib.currentWave) || 0);
  const left = Math.max(0, total - cur);
  const hp = Math.max(0, Number(health) || 0);
  let perWave = getTribulationWaveDamage(trib && trib.intendedStageId, maxHp, bgId);

  const tNames = Array.isArray(traits) ? traits.map(t => typeof t === 'string' ? t : (t?.name || '')) : [];
  const hasLoiThanMoc = tNames.includes('Vạn Cổ Lôi Thần Mộc Khu') || (Array.isArray(traits) && traits.some(t => t?.id === 9805));
  const hasThienLinhCan = tNames.some(n => n.includes('Thiên Linh Căn'));
  const hasQuyKhu = tNames.includes('Đại Đạo Quy Khư');
  const hasThienLoiThe = tNames.includes('Thiên Lôi Thể') || (Array.isArray(traits) && traits.some(t => t?.id === 1315));
  const hasDiaThoMoc = tNames.includes('Địa Thổ Mộc Thể') || (Array.isArray(traits) && traits.some(t => t?.id === 1316));
  const hasThanhQuang = tNames.includes('Thánh Quang Thần Thể') || (Array.isArray(traits) && traits.some(t => t?.id === 1314));

  // V2.12 GÓP Ý DARK - HỆ THỐNG GIẢM TRỪ SÁT THƯƠNG LÔI KIẾP (THỂ TU, LINH CĂN & CÔNG PHÁP)
  let totalMitigationPct = 0;

  // 1. Giảm trừ từ chỉ số Phòng Thủ (Defense) & Đạo Lộ Thể Tu
  const defVal = Math.max(0, Number(defense) || 0);
  const isTheTu = cultivationPath === 'the_tu' || tNames.some(n => /thể|bàn cổ|kim thân|luyện bì|luyện cốt/i.test(n));
  if (defVal > 0 && maxHp > 0) {
    const defRatio = defVal / (defVal + maxHp * 0.8);
    if (isTheTu) {
      // Thể Tu nhục thân tôi luyện cực hạn: giảm tối đa 60% sát thương lôi kiếp từ chỉ số thủ
      const theTuMitigation = Math.min(0.60, defRatio * 1.2 + 0.15);
      totalMitigationPct += theTuMitigation;
    } else {
      // Các phái khác giảm tối đa 35% sát thương từ phòng thủ/hộ thể
      const normalMitigation = Math.min(0.35, defRatio * 0.7);
      totalMitigationPct += normalMitigation;
    }
  }

  // 2. Giảm trừ từ Linh Căn (Lôi, Mộc, Quang)
  const srElements = (spiritualRoot && Array.isArray(spiritualRoot.elements)) ? spiritualRoot.elements : [];
  const srName = String(spiritualRoot?.name || '');
  if (srElements.includes('Lôi') || /lôi/i.test(srName)) {
    totalMitigationPct += 0.40; // Lôi Linh Căn: Lôi dẫn nhập thể giảm 40% sát thương
  }
  if (srElements.includes('Quang') || /quang/i.test(srName)) {
    totalMitigationPct += 0.25; // Quang Linh Căn: Thánh quang tiêu tán giảm 25% sát thương
  }
  if (srElements.includes('Mộc') || /mộc/i.test(srName)) {
    totalMitigationPct += 0.20; // Mộc Linh Căn: Sinh cơ dẫn lôi nhập địa giảm 20% sát thương
  }

  // 3. Giảm trừ từ Thể Chất đặc thù
  if (hasThienLoiThe) {
    totalMitigationPct += 0.20; // Thiên Lôi Thể miễn 20% sát thương lôi kiếp
  }
  if (hasThanhQuang) {
    totalMitigationPct += 0.25; // Thánh Quang Thần Thể miễn 25% sát thương lôi kiếp
  }
  if (hasDiaThoMoc) {
    totalMitigationPct += 0.20; // Địa Thổ Mộc Thể miễn 20% sát thương lôi kiếp
  }

  // 4. Giảm trừ từ Công Pháp phòng ngự / Thần thông hộ thể
  const methodStr = String(equippedCoreMethod || '').toLowerCase();
  if (/(hộ thể|kim chung|quy giáp|bất diệt|lôi pháp|lôi thần|toái ngọc|huyền giáp|phòng ngự|trận đồ|lạc tinh|kiên bích)/i.test(methodStr)) {
    totalMitigationPct += 0.20; // Công pháp phòng thủ giảm 20% sát thương lôi kiếp
  }

  // Áp dụng trần giảm thương tối đa 85% để giữ độ thử thách
  totalMitigationPct = Math.min(0.85, totalMitigationPct);
  if (totalMitigationPct > 0) {
    perWave = Math.max(1, Math.floor(perWave * (1 - totalMitigationPct)));
  }

  if (hasThienLinhCan) {
    perWave = Math.max(1, Math.floor(perWave * 0.7)); // V1.89 THIÊN ĐẠO ƯU TIÊN: Giảm 30% sát thương lôi kiếp
  }

  // V2.81: Tiểu Thế Giới đã gỡ bỏ khỏi cơ chế cản lôi kiếp

  const hasThienMenhMaTon = Array.isArray(traits) && traits.some(t => t && (t.name === 'Thiên Mệnh Ma Tôn' || t.id === 2088));
  // V1.79: THIÊN ĐẠO KHƯỚC TỪ (Góp ý Kha) - Lôi kiếp x2 sát thương chuẩn
  if (bgId === 'bg_thien_dao_khuoc_tu') {
    perWave = perWave * 2;
  }
  if (hasThienMenhMaTon) {
    perWave = perWave * 2; // V1.70 THIÊN ĐẠO CĂM GHÉT: Lôi kiếp x2 uy lực!
  }

  if (left <= 0 || hp <= 0) {
    return { waves: 0, perWave: perWave, damage: 0, lethal: false, mercy: false, left: left, isLoiThanMoc: hasLoiThanMoc };
  }

  const dec = Math.max(0, Math.floor(Number(declaredWaves) || 0));
  let waves = 0;
  let mercy = false;

  if (hasLoiThanMoc) {
    // V1.36 FIX: Vạn Cổ Lôi Thần Mộc Khu — Cắn nuốt lôi kiếp, không nhận sát thương, không thể tử vong!
    waves = dec > 0 ? Math.min(left, dec) : Math.min(left, Math.max(1, Math.min(3, left)));
    return { waves: waves, perWave: perWave, damage: 0, lethal: false, mercy: false, left: left, isLoiThanMoc: true };
  }
  if (hasQuyKhu) {
    // V1.89 ĐẠI ĐẠO QUY KHƯ: Quy khư cội nguồn hóa giải lôi kiếp!
    waves = dec > 0 ? Math.min(left, dec) : Math.min(left, Math.max(1, Math.min(3, left)));
    return { waves: waves, perWave: 0, damage: 0, lethal: false, mercy: false, left: left, isLoiThanMoc: true };
  }

  if (dec > 0) {
    // Nguoi choi tu khai so -> KHONG thuong xot, dung dung so do.
    waves = Math.min(left, dec);
  } else {
    // Khong khai -> Thien Dao gieo xu 50/50.
    mercy = Math.random() < TRIB_MERCY_CHANCE;
    if (mercy) {
      waves = Math.max(1, Math.min(left, Math.floor(Math.max(0, hp - 1) / perWave)));
    } else {
      waves = Math.max(1, Math.min(left, Math.ceil(hp / perWave) + (Math.random() < 0.5 ? 1 : 0)));
    }
  }

  let damage = waves * perWave;
  // V1.39 DARK & SƯ PHỤ: Giữ ngưỡng sinh tử 5% HP cho 5 cảnh giới đầu (id <= 5), chống chết sốc đột ngột trừ khi chủ động khai số tự sát
  const stageId = Number(trib && trib.intendedStageId) || 2;
  if (stageId <= 5 && dec === 0) {
    const survivalThreshold = Math.max(1, Math.floor(maxHp * 0.05));
    if (damage >= hp) {
      damage = Math.max(1, hp - survivalThreshold);
      mercy = true;
    }
  } else if (mercy && damage >= hp) {
    damage = Math.max(1, hp - 1);
  }

  return { waves: waves, perWave: perWave, damage: damage, lethal: damage >= hp, mercy: mercy, left: left, isLoiThanMoc: false };
};

export const isPhysicalTrait = (t: any): boolean => {
    if (!t) return false;
    const name = typeof t === 'string' ? t : (t.name || '');
    const desc = typeof t === 'object' ? (t.desc || '') : '';
    if (typeof MASTER_FUSION_MATRIX !== 'undefined' && MASTER_FUSION_MATRIX.some(r => r.name.toLowerCase() === name.toLowerCase())) {
        return true;
    }
    if (/(thể|thân|cốt|huyết mạch|chân thân|nhục thân|thánh thể|thần thể|bá thể|ma thể|bất hoại|kim thân|khu\b|nhãn\b|mị cốt|thiên đố|phôi)/i.test(name)) {
        return true;
    }
    if (/(thể chất|nhục thân|huyết mạch|thần thể|thánh thể|cốt tủy)/i.test(desc)) {
        return true;
    }
    return false;
};

export const countPhysicalTraitsWithCores = (traits: any[]): number => {
    if (!traits || !Array.isArray(traits)) return 1;
    let total = 0;
    traits.forEach(t => {
        if (!t) return;
        const name = typeof t === 'string' ? t : (t.name || '');
        if (typeof MASTER_FUSION_MATRIX !== 'undefined') {
            const recipe = MASTER_FUSION_MATRIX.find(r => r.name.toLowerCase() === name.toLowerCase());
            if (recipe) {
                const ings = typeof getRecursiveFusionIngredients === 'function' ? getRecursiveFusionIngredients(name) : [];
                total += Math.max(1, ings.length);
                return;
            }
        }
        if (isPhysicalTrait(t)) {
            total += 1;
        }
    });
    return Math.max(1, total);
};

const hasWoodSpiritualRoot = (sr: any): boolean => {
    if (!sr) return false;
    if (String(sr.name || '').toLowerCase().includes('mộc')) return true;
    if (Array.isArray(sr.elements) && sr.elements.some((e: any) => String(e).toLowerCase().includes('mộc'))) return true;
    return false;
};

const getNightmareStats = (levelString) => {
  const lv = String(levelString || '').toLowerCase().trim();
  if (!lv || lv.includes('phàm nhân') || lv.includes('người thường')) {
    return { atk: 100, def: 100, spd: 100, spp: 100, hp: 1000, mp: 1000, valueOf() { return this.atk; } };
  }

  let subKey = 'default';
  const tangM = lv.match(/tầng\s*(\d+)/);
  if (tangM) {
    const t = parseInt(tangM[1], 10);
    subKey = 't' + Math.max(1, Math.min(10, t));
  } else if (lv.includes('sơ kỳ')) subKey = 'so';
  else if (lv.includes('trung kỳ')) subKey = 'trung';
  else if (lv.includes('hậu kỳ')) subKey = 'hau';
  else if (lv.includes('viên mãn') || lv.includes('cực cảnh') || lv.includes('đỉnh phong')) subKey = 'vien';

  for (let i = NIGHTMARE_DATA.length - 1; i >= 0; i--) {
    const item = NIGHTMARE_DATA[i];
    if (item.regex.test(lv)) {
      const v4 = item.r4[subKey] || item.r4.default || 100;
      const v3 = item.r3[subKey] || item.r3.default || 1000;
      return {
        atk: v4, def: v4, spd: v4, spp: v4,
        hp: v3, mp: v3,
        valueOf() { return this.atk; }
      };
    }
  }

  return { atk: 100, def: 100, spd: 100, spp: 100, hp: 1000, mp: 1000, valueOf() { return this.atk; } };
};

const applyNpcStatWithSaturation = (statName, prevVal, aiAbsoluteVal, isBonus = false, statKey, npc, forceBypass = false) => {
    if (aiAbsoluteVal === undefined || aiAbsoluteVal === null) return Number(prevVal) || 0;
    const targetVal = parseAIStat(aiAbsoluteVal);
    const baseValue = Number(prevVal) || (isBonus ? 0 : 10);
    const change = targetVal - baseValue;
    
    if (change <= 0) return targetVal;

    // 1. Ngoại Lệ Vĩ Mô, Đặc Quyền Admin, Chế Độ Ác Mộng & 10 Siêu Thực Thể Ngoại Hạn (Bỏ qua khóa trần hoàn toàn)
    const npcCleanName = String(npc?.name || '').toLowerCase();
    const isUncappedCosmicEntity = [
        'azathoth', 'phong đô đại đế', 'neoth', 'beyond anafabula', 'ma thần gurren lagann',
        'void demonbane', 'featherine', 'featherine augustus aurora', 'tự sự chí tôn', 'atroposx', 'bạch hoàng',
        'anos voldigoad', 'anos', 'ruphas mafahl', 'ruphas', 'bá hoàng nữ vương',
        'shallow vernal', 'bạch ngưng chi chủ', 'bất nghiệt tự điển',
        'orsted', 'long thần orsted', 'cổ trần sa', 'sailor galaxia', 'kha thần lục',
        'rimuru', 'rimuru tempest', 'rimura', 'hư vô thao thiết', 'vĩnh hằng cổ lĩnh chủ',
        'klein moretti', 'klein', 'ngu giả'
    ].some(n => npcCleanName.includes(n));

    if (forceBypass || isUncappedCosmicEntity || npc?.isNightmare) return baseValue + change;

    const currentStageConfig = getStageConfig(npc.level || 'Phàm Nhân');
    const hasTrait = (name) => npc.traits?.some(t => (t.name || t) === name);
    
    if (isBonus) {
        let finalLimit = 10 + (currentStageConfig.id * 5); 
        if (statKey === 'maxHp') finalLimit *= 20; 
        if (statKey === 'maxMp') finalLimit *= 10; 
        finalLimit *= 3; 
        
        if (statKey === 'maxHp' && hasTrait('Bàn Cổ Huyết Mạch')) finalLimit *= 5;

        if (change > finalLimit) {
            const excess = change - finalLimit;
            const absorbedExcess = Math.min(Math.floor(excess * 0.05), finalLimit * 2);
            const saturatedChange = finalLimit + absorbedExcess; 
            return baseValue + saturatedChange;
        }
        return baseValue + change;
    }

    let minorCapPct = 1.0;
    const lvlStr = String(npc.level || '').toLowerCase();
    if (lvlStr.includes('sơ kỳ')) minorCapPct = 0.30;
    else if (lvlStr.includes('trung kỳ')) minorCapPct = 0.55;
    else if (lvlStr.includes('hậu kỳ')) minorCapPct = 0.80;
    else if (lvlStr.includes('viên mãn')) minorCapPct = 1.0;
    else {
        const match = lvlStr.match(/tầng\s+(\d+)/);
        if (match) {
            const tang = parseInt(match[1]);
            minorCapPct = Math.min(1.0, Math.max(0.1, tang / 9));
        }
    }

    let HARD_CAP = 2625 * Math.pow(currentStageConfig.id, 3) * minorCapPct; 
    if (statKey === 'defense' && hasTrait('Bàn Cổ Huyết Mạch')) HARD_CAP *= 1.5;
    if (statKey === 'spiritPower' && hasTrait('Ngộ Tính Siêu Phàm')) HARD_CAP *= 1.5;
    if (statKey === 'speed' && hasTrait('Phong Ảnh Thể')) HARD_CAP *= 1.5;
    if (statKey === 'attack' && hasTrait('Tuyệt Thế Kiếm Tu')) HARD_CAP *= 1.2;

    const capRatio = baseValue / HARD_CAP;
    
    let effectiveChange = change;
    let multiplier = 1.0;
    if (capRatio >= 0.99) multiplier = 0.01;
    else if (capRatio >= 0.95) multiplier = 0.02;
    else if (capRatio >= 0.90) multiplier = 0.10;
    else if (capRatio >= 0.80) multiplier = 0.30;
    
    effectiveChange = change * multiplier;

    let maxChangePerTurn = Math.floor(15 * Math.pow(currentStageConfig.id, 1.8)) * 3;
    if (hasTrait('Thôn Kim Phệ Cốt Thể')) maxChangePerTurn *= 5;

    if (effectiveChange > maxChangePerTurn) {
        const excess = effectiveChange - maxChangePerTurn;
        const absorbedExcess = Math.min(Math.floor(excess * 0.05), maxChangePerTurn * 2);
        effectiveChange = maxChangePerTurn + absorbedExcess; 
    }
    
    return Math.floor(baseValue + effectiveChange);
};

  const safeTraits = Array.isArray(traits) ? traits : [];
  const getTName = (t: any): string => (typeof t === 'string' ? t : (t?.name || ''));
  
  // V1.75: THỂ CHẤT THIÊN ĐỐ - Khóa trần thọ nguyên cơ bản 300 năm (chỉ tăng qua cắn đan diên thọ)
  if (hasThienDoTrait(safeTraits)) {
      maxAge = 300;
  }
  const raceStr = String(race).toLowerCase();
  
  // V1.22.4: BẢO TOÀN VÀ TĂNG THỌ CHO CÁC THẦN THỂ DUNG HỢP
  if (safeTraits.some(t => getTName(t) === 'Vạn Kiếp Chân Tôn Thể')) maxAge += 500000;
  if (safeTraits.some(t => getTName(t) === 'Tiên Thiên Bách Thảo Bất Tử Khu')) maxAge += 10000;
  if (safeTraits.some(t => (getTName(t) === 'Hỗn Độn Bàn Cổ Khu' || getTName(t) === 'Thập Nhị Đô Thiên Thần Sát Chân Thân'))) maxAge += 500;
  if (safeTraits.some(t => (getTName(t) === 'Hoang Cổ Chí Tôn Thể' || getTName(t) === 'Hỗn Độn Vạn Khí Cốt'))) maxAge += 200;
  if (safeTraits.some(t => getTName(t) === 'Đạo Tổ Toàn Tri Thể')) maxAge += 1000000;
  // V1.22.11b FIX: Thêm thọ nguyên cho 5 Thần Thể fusion thiếu
  if (safeTraits.some(t => getTName(t) === 'Bồ Đề Ma La Bất Hoại Thân')) maxAge += 9999999; // Bất hoại = gần bất tử
  if (safeTraits.some(t => (getTName(t) === 'Cơ Giới Hắc Quang Thể' || getTName(t) === 'Hạch Tâm Cơ Thần Khôi Lỗi' || getTName(t) === 'Chung Yên Điểm Kỳ Dị Thể'))) maxAge += 5000; // Cơ giới thể = bền bỉ
  if (safeTraits.some(t => getTName(t) === 'Vạn Trận Quy Tông Thể')) maxAge += 500; // Trận sư

  if (safeTraits.some(t => getTName(t) === 'Bàn Cổ Huyết Mạch')) maxAge += 50;
  if (safeTraits.some(t => getTName(t) === 'Trường Sinh Thể')) maxAge += 100;
  // V10.76 FIX: Đặc quyền thọ nguyên của Đạo Tổ hóa phàm
  if (safeTraits.some(t => getTName(t) === 'Đạo Tổ Niết Bàn')) maxAge += 100000; 
  if (safeTraits.some(t => getTName(t) === 'Trường Sinh Bất Lão')) maxAge += 9999999;
  if (safeTraits.some(t => getTName(t) === 'Tiên Phong Đạo Cốt')) maxAge += 10;
  if (safeTraits.some(t => getTName(t) === 'Đại Nạn Bất Tử')) maxAge += 20;
  if (safeTraits.some(t => getTName(t) === 'Tai Kiếp Liên Miên')) maxAge -= 20;
  if (safeTraits.some(t => getTName(t) === 'Ách Vận Bám Thân')) maxAge -= 15;
  // Dù có bị giảm nửa tuổi thọ, thì cộng với Vô Hạn vẫn là Vô Hạn
  if (safeTraits.some(t => getTName(t) === 'Thiên Đố Anh Tài')) maxAge = Math.floor(maxAge / 2);

  if (safeTraits.some(t => getTName(t) === 'Bán Tinh Linh (Huyết Mạch Lai)')) maxAge += 150;
  if (safeTraits.some(t => getTName(t) === 'Bán Long (Huyết Mạch Lai)')) maxAge += 300;
  if (safeTraits.some(t => getTName(t) === 'Bán Thần (Huyết Mạch Lai)')) maxAge += 500;
  if (safeTraits.some(t => (getTName(t) === 'Bán Yêu (Huyết Mạch Lai)' || getTName(t) === 'Bán Ma (Huyết Mạch Lai)' || getTName(t) === 'Bán Mị Ma (Huyết Mạch Lai)'))) maxAge += 50;

  // V10.94: Áp dụng Lớp Giáp Thứ 2 - Hệ Số Tuổi Thọ Chủng Tộc
if (/(rồng|long|thần|thái thản|titan)/.test(raceStr)) maxAge *= 100;
else if (/(cây|thảo mộc|thụ|linh sâm|mộc)/.test(raceStr)) maxAge *= 10;
else if (/(yêu|thú|cáo|sói|hổ|báo|hung|cự nhân)/.test(raceStr)) maxAge *= 5;

// Nhận diện kép (Đề phòng file bị mất ID xuất thân trong lúc tính toán)
const isPlant = bgId === 'bg_20' || safeTraits.some(t => getTName(t) === 'Bản Thể Thảo Mộc');
const isTreasure = bgId === 'bg_24' || safeTraits.some(t => getTName(t) === 'Thiên Tài Địa Bảo');
const isDemon = bgId === 'bg_25' || safeTraits.some(t => getTName(t) === 'Tiên Thiên Đại Yêu');
// V7.16 & V10.64 & V2.55 FIX: Bất tử cảnh giới (Chân Tiên ID >= 9 trở lên, Đạo Tổ, v.v.) và các dị loại Thạch Thần, Thi Tộc
const hasImmortalTrait = safeTraits.some(t => ['Thạch Thần Bất Diệt', 'Cơ Quan Hạch Tâm', 'Bảo Vật Hóa Hình', 'Trường Sinh Cửu Thị', 'Trường Sinh Bất Lão', 'Đạo Tổ Toàn Tri Thể'].includes(getTName(t)));
const isImmortal = stageCfg.id >= 9 || /(đạo tổ|tiên đế|tiên vương|tiên hoàng|tiên tôn|đại la|thái ất|kim tiên|huyền tiên|chân tiên|hư tiên|độ kiếp|đại thừa|chí tôn)/i.test(String(level || '')) || ['bg_21', 'bg_22', 'bg_23', 'bg_undead', 'bg_ctan'].includes(bgId) || hasImmortalTrait || /(đá|thạch|máy|khôi lỗi|kiếm|bảo vật|vong linh|cương thi)/.test(raceStr);

if (isPlant && maxAge < 500) maxAge += 500; 
if (isTreasure && maxAge < 1000) maxAge += 1000; 
if (isDemon && maxAge < 300) maxAge += 300; 

// V1.49: THỌ NGUYÊN YÊU TỘC THUẦN CHỦNG (Long tộc 200, Phượng hoàng 150 - Góp ý Ethan195)
if (bgId === 'bg_yeu_thuan_long') {
    if (stageCfg.id <= 2 && maxAge < 200) maxAge = 200;
    else maxAge += 100;
} else if (bgId === 'bg_yeu_thuan_phuong') {
    if (stageCfg.id <= 2 && maxAge < 150) maxAge = 150;
    else maxAge += 50;
} else if (['bg_yeu_thuan_mieu', 'bg_yeu_thuan_ho', 'bg_yeu_thuan_dieu', 'bg_yeu_thuan_di_toc', 'bg_yeu_thuan_ngu'].includes(bgId)) {
    if (stageCfg.id <= 2 && maxAge < 120) maxAge = 120;
    else maxAge += 20;
} 
if (isImmortal) maxAge += 9999999; // Bất tử tuyệt đối (10 triệu năm)

// V2.22 ETHAN195 FIX & V3.27 DARK FIX: Khóa sàn thọ nguyên bảo hiểm toàn diện
const effectiveBonus = Number(bonusMaxAge) || 0;
let finalMaxAge = maxAge + effectiveBonus;
// V3.27 FIX: Các bậc Bất Tử (Cấp 9+, Đạo Tổ, Vạn Ma Chi Chủ...) khóa cứng sàn thọ nguyên tối thiểu 10 triệu năm
if (isImmortal) {
    finalMaxAge = Math.max(10000000, finalMaxAge);
}
if (currentAge && Number(currentAge) > 0) {
    finalMaxAge = Math.max(Math.floor(Number(currentAge)) + 1, finalMaxAge);
}
return Math.max(10, finalMaxAge);
};
