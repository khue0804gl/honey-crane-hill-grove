const getSpiritualRoot = (traits, bgId = 'bg_0') => {
  // V1.79: XUẤT THÂN THIÊN ĐẠO KHƯỚC TỪ (Góp ý Kha)
  if (bgId === 'bg_thien_dao_khuoc_tu') {
      return { name: 'Phế Linh Căn (Ngũ Hành Tạp Căn Khuyết)', elements: ['Kim Khuyết', 'Mộc Tàn', 'Thủy Cạn', 'Hỏa Lụi', 'Thổ Nứt'], purity: 'Phế Phẩm Bị Thiên Đạo Ruồng Bỏ', color: 'text-gray-500 font-bold drop-shadow-[0_0_5px_rgba(107,114,128,0.8)]' };
  }
  // V1.49: YÊU TỘC THUẦN CHỦNG KHỞI ĐẦU VÔ LINH CĂN (Chờ Kim Đan hóa hình thức tỉnh - Góp ý Ethan195)
  if (isPureYeuTocBg(bgId)) {
      if (bgId === 'bg_yeu_thuan_ngu') {
          return { name: 'Đơn Thủy Linh Căn (Hải Uyên Thần Chủng)', elements: ['Thủy'], purity: 'Thuần Khiết Hải Vực', color: 'text-cyan-400 font-bold drop-shadow-[0_0_8px_rgba(34,211,238,0.8)]' };
      }
      if (bgId === 'bg_yeu_thuan_phuong') {
          return { name: 'Vô Linh Căn (Chờ Thức Tỉnh Linh Hỏa Kim Đan)', elements: ['Phượng Hỏa Yêu Khí'], purity: 'Phượng Hoàng Thuần Chủng', color: 'text-orange-400 font-bold drop-shadow-[0_0_5px_rgba(251,146,60,0.8)]' };
      }
      return { name: 'Vô Linh Căn (Yêu Khí Bẩm Sinh)', elements: ['Yêu Khí'], purity: 'Yêu Tộc Chờ Hóa Hình', color: 'text-purple-400 font-bold drop-shadow-[0_0_5px_rgba(192,132,252,0.8)]' };
  }
    const tNames = (traits || []).map(t => t?.name || '');
    
    if (tNames.includes('Thiên Sinh Rác Rưởi')) {
        if (typeof __GlobalPremadeWorld !== 'undefined' && __GlobalPremadeWorld === 'dau_pha') {
            return { name: 'Vô Thuộc Tính', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
        }
        return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
    }

    if (tNames.includes('Thiên Sinh Phế Vật')) {
        if (Math.random() < 0.70) {
            if (typeof __GlobalPremadeWorld !== 'undefined' && __GlobalPremadeWorld === 'dau_pha') {
                return { name: 'Vô Thuộc Tính', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
            }
            return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
        } else {
            if (typeof __GlobalPremadeWorld !== 'undefined' && __GlobalPremadeWorld === 'dau_pha') {
                const dpEls = ['Hỏa', 'Mộc', 'Thủy', 'Lôi', 'Băng', 'Phong', 'Thổ', 'Kim', 'Ám'];
                const shuffled = [...dpEls].sort(() => 0.5 - Math.random());
                const myEls = shuffled.slice(0, 5);
                return { name: `Ngũ Thuộc Tính (${myEls.join('-')})`, elements: myEls, purity: 'Kém', color: 'text-[var(--text-muted)] font-medium' };
            } else {
                const els = ['Kim', 'Mộc', 'Thủy', 'Hỏa', 'Thổ'];
                const shuffled = [...els].sort(() => 0.5 - Math.random());
                return { name: `Ngũ (Tạp) Linh Căn (${shuffled.join('-')})`, elements: shuffled, purity: 'Kém', color: 'text-[var(--text-muted)] font-medium' };
            }
        }
    }

    // V13: Độc tôn Thuộc Tính Đấu Khí (Thay thế Linh Căn)
    if (typeof __GlobalPremadeWorld !== 'undefined' && __GlobalPremadeWorld === 'dau_pha') {
        const dpEls = ['Hỏa', 'Mộc', 'Thủy', 'Lôi', 'Băng', 'Phong', 'Thổ', 'Kim', 'Ám'];
        const isAlchemist = Math.random() < 0.15 || tNames.includes('Học Đồ Luyện Dược') || tNames.includes('Luyện Dược Kì Tài');
        
        if (isAlchemist) {
            return { name: 'Hỏa Mang Mộc Khí (Luyện Dược Sư)', elements: ['Hỏa', 'Mộc'], purity: 'Cực Phẩm', color: 'text-red-500 font-bold drop-shadow-[0_0_5px_rgba(239,68,68,0.8)]' };
        }
        const myEl = dpEls[Math.floor(Math.random() * dpEls.length)];
        return { name: `Thuộc Tính (${myEl})`, elements: [myEl], purity: 'Bình Thường', color: 'text-orange-400 font-bold' };
    }

    const els = ['Kim', 'Mộc', 'Thủy', 'Hỏa', 'Thổ'];
    const rares = ['Băng', 'Phong', 'Lôi', 'Ám', 'Quang'];
    
    if (tNames.includes('Hư Không Lậu Thể')) return { name: 'Phản Vật Chất Hố Đen', elements: ['Không Gian', 'Thời Gian'], purity: 'Độc Nhất', color: 'text-gray-900 border border-gray-600 bg-black font-black drop-shadow-[0_0_5px_rgba(255,255,255,0.5)] px-1 rounded' };
    if (tNames.includes('Thủy Băng Hàn Thể')) return { name: 'Thủy Băng Linh Căn', elements: ['Thủy', 'Băng', 'Hàn'], purity: 'Tuyệt Đối', color: 'text-cyan-300 font-black drop-shadow-[0_0_5px_rgba(103,232,249,0.8)]' };
    if (tNames.includes('Tuyệt Thế Kiếm Tu') || tNames.includes('Tuyệt Thế Đao Tu') || tNames.includes('Tuyệt Thế Thương Tu') || tNames.includes('Tuyệt Thế Kích Tu')) return { name: 'Cực Kim Linh Căn', elements: ['Kim'], purity: 'Tuyệt Đối', color: 'text-yellow-400 font-black drop-shadow-[0_0_5px_rgba(250,204,21,0.5)]' };
    if (tNames.includes('Tuyệt Thế Cung Tu') || tNames.includes('Phong Ảnh Thể')) return { name: 'Cực Phong Linh Căn', elements: ['Phong'], purity: 'Tuyệt Đối', color: 'text-emerald-300 font-black drop-shadow-[0_0_5px_rgba(110,231,183,0.5)]' };
    if (tNames.includes('Tuyệt Thế Phủ Tu') || tNames.includes('Tuyệt Thế Chùy Tu')) return { name: 'Cực Thổ Linh Căn', elements: ['Thổ'], purity: 'Tuyệt Đối', color: 'text-amber-600 font-black drop-shadow-[0_0_5px_rgba(217,119,6,0.8)]' };
    if (tNames.includes('Tuyệt Thế Roi Tu')) return { name: 'Cực Lôi Linh Căn', elements: ['Lôi'], purity: 'Tuyệt Đối', color: 'text-purple-400 font-black drop-shadow-[0_0_5px_rgba(192,132,252,0.8)]' };
    if (tNames.includes('Tuyệt Thế Nhạc Tu')) return { name: 'Cực Thủy Linh Căn', elements: ['Thủy'], purity: 'Tuyệt Đối', color: 'text-blue-400 font-black drop-shadow-[0_0_5px_rgba(96,165,250,0.8)]' };
    if (tNames.includes('Thánh Quang Thần Thể')) return { name: 'Cực Quang Linh Căn', elements: ['Quang'], purity: 'Tuyệt Đối', color: 'text-yellow-200 font-black drop-shadow-[0_0_5px_rgba(254,240,138,0.8)]' };
    if (tNames.includes('Thiên Lôi Thể')) return { name: 'Cực Lôi Linh Căn', elements: ['Lôi'], purity: 'Tuyệt Đối', color: 'text-purple-400 font-black drop-shadow-[0_0_5px_rgba(192,132,252,0.8)]' };
    if (tNames.includes('Địa Thổ Mộc Thể')) return { name: 'Cực Thổ Mộc Linh Căn', elements: ['Thổ', 'Mộc'], purity: 'Tuyệt Đối', color: 'text-amber-600 font-black drop-shadow-[0_0_5px_rgba(217,119,6,0.8)]' };
    if (tNames.includes('Hỗn Độn Diệt Thế Ma Căn')) return { name: 'Diệt Thế Ma Căn', elements: ['Hỗn Độn', 'Hủy Diệt', 'Hủ Hóa', 'Phóng Xạ', 'Ám'], purity: 'Tối Thượng', color: 'text-red-600 font-black drop-shadow-[0_0_8px_rgba(220,38,38,0.8)]' };
    if (tNames.includes('Hỗn Độn Đạo Thể')) return { name: 'Hỗn Độn Linh Căn', elements: ['Kim', 'Mộc', 'Thủy', 'Hỏa', 'Thổ', 'Âm', 'Dương'], purity: 'Tuyệt Đối', color: 'text-rose-500 font-black drop-shadow-[0_0_5px_rgba(244,63,94,0.5)]' };
    if (tNames.includes('Thiên Linh Căn')) {
        const isRare = Math.random() > 0.8;
        const el = isRare ? rares[Math.floor(Math.random()*rares.length)] : els[Math.floor(Math.random()*els.length)];
        return { name: `Thiên Linh Căn (${el})`, elements: [el], purity: 'Thuần Khiết', color: 'text-red-500 font-bold' };
    }
    
    if (bgId === 'bg_0') {
        if (Math.random() < 0.40) {
            return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
        } else {
            const rootRoll = Math.random();
            if (rootRoll < 0.10) { 
                const isRare = Math.random() > 0.8;
                const el = isRare ? rares[Math.floor(Math.random()*rares.length)] : els[Math.floor(Math.random()*els.length)];
                return { name: `Thiên Linh Căn (${el})`, elements: [el], purity: 'Thuần Khiết', color: 'text-red-500 font-bold' };
            } else if (rootRoll < 0.20) { 
                if (Math.random() > 0.5) {
                    const el = rares[Math.floor(Math.random()*rares.length)];
                    return { name: `Dị Linh Căn (${el})`, elements: [el], purity: 'Biến Dị', color: 'text-purple-400 font-bold' };
                } else {
                    const shuffled = [...els].sort(() => 0.5 - Math.random());
                    const myEls = shuffled.slice(0, 2);
                    return { name: `Song Linh Căn (${myEls.join('-')})`, elements: myEls, purity: 'Thượng Phẩm', color: 'text-blue-400 font-bold' };
                }
            } else { 
                const countRoll = Math.random();
                const count = countRoll > 0.6 ? 3 : countRoll > 0.2 ? 4 : 5;
                const shuffled = [...els].sort(() => 0.5 - Math.random());
                const myEls = shuffled.slice(0, count);
                const prefix = count === 3 ? 'Tam' : count === 4 ? 'Tứ' : 'Ngũ (Tạp)';
                let color = count === 3 ? 'text-emerald-500 font-bold' : 'text-[var(--text-muted)] font-medium';
                return { name: `${prefix} Linh Căn (${myEls.join('-')})`, elements: myEls, purity: count === 3 ? 'Trung Phẩm' : 'Kém', color };
            }
        }
    }

    const mortalBgs = ['bg_17', 'bg_tieu_nhi', 'bg_lang_y', 'bg_ngu_phu', 'bg_tieu_thuong', 'bg_ca_ki', 'bg_linh_quen', 'bg_no_le', 'bg_tu_tu', 'bg_kiem_no', 'bg_dao_tac', 'bg_vo_su_pham'];
    const isMortalBg = mortalBgs.includes(bgId) || tNames.includes('Kẻ Ăn Mày') || tNames.includes('Thường Dân');
    const isGodBg = ['bg_8', 'bg_9', 'bg_10', 'bg_19', 'bg_20', 'bg_21', 'bg_22', 'bg_23', 'bg_24', 'bg_25', 'bg_long_toc', 'bg_con_bang', 'bg_ho_toc', 'bg_vu_toc', 'bg_chan_thien_menh', 'bg_elf', 'bg_angel', 'bg_blood_angel', 'bg_succubus', 'bg_nightmare', 'bg_undead', 'bg_titan', 'bg_behemoth', 'bg_taotie', 'bg_ctan', 'bg_vien_co_cu_long'].includes(bgId);
    
    if (isMortalBg && Math.random() < 0.90) {
        return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
    }
    if (!isGodBg && !isMortalBg && Math.random() < 0.40) {
        return { name: 'Vô Linh Căn', elements: ['Không Có'], purity: 'Phế Vật', color: 'text-gray-500 font-medium' };
    }

    const countRoll = Math.random();
    const count = countRoll > 0.8 ? 2 : countRoll > 0.4 ? 3 : countRoll > 0.1 ? 4 : 5;
    
    if (count === 2 && Math.random() > 0.8) {
        const el = rares[Math.floor(Math.random()*rares.length)];
        return { name: `Dị Linh Căn (${el})`, elements: [el], purity: 'Biến Dị', color: 'text-purple-400 font-bold' };
    }
    
    const shuffled = [...els].sort(() => 0.5 - Math.random());
    const myEls = shuffled.slice(0, count);
    const prefix = count === 2 ? 'Song' : count === 3 ? 'Tam' : count === 4 ? 'Tứ' : 'Ngũ (Tạp)';
    
    let color = 'text-[var(--text-main)]';
    if (count === 2) color = 'text-blue-400 font-bold';
    else if (count === 3) color = 'text-emerald-500 font-bold';
    else if (count >= 4) color = 'text-[var(--text-muted)] font-medium';

    const calcPurity = count === 2 ? 'Thượng Phẩm' : count === 3 ? 'Trung Phẩm' : 'Kém';
    return { name: `${prefix} Linh Căn (${myEls.join('-')})`, elements: myEls, purity: calcPurity, color };
};
