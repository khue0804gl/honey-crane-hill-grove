const parseKnowledgeLines = (knowledgeStr, currentAge = 18) => {
    if (!knowledgeStr || typeof knowledgeStr !== 'string') return [];
    const lines = knowledgeStr.split('\n').map(l => l.trim()).filter(l => l.startsWith('-'));
    
    return lines.map((line, idx) => {
        let text = line.replace(/^-\s*/, '').trim();
        let status = 'Đã Kiểm Chứng';
        let statusTag = '🟢 [Đã Kiểm Chứng]';
        let statusColor = 'text-emerald-400 bg-emerald-950/40 border-emerald-500/40';
        let timeTag = '';
        let chapterTag = '';
        let learnedAge = null;
        let content = text;

        if (text.includes('[Tin Đồn]') || text.includes('🟡')) {
            status = 'Tin Đồn';
            statusTag = '🟡 [Tin Đồn]';
            statusColor = 'text-yellow-400 bg-yellow-950/40 border-yellow-500/40';
        } else if (text.includes('[Đã Đính Chính]') || text.includes('🟣')) {
            status = 'Đã Đính Chính';
            statusTag = '🟣 [Đã Đính Chính]';
            statusColor = 'text-purple-400 bg-purple-950/40 border-purple-500/40';
        } else if (text.includes('[Nguy Cơ Lỗi Thời]') || text.includes('🔴')) {
            status = 'Nguy Cơ Lỗi Thời';
            statusTag = '🔴 [Nguy Cơ Lỗi Thời]';
            statusColor = 'text-red-400 bg-red-950/40 border-red-500/40';
        }

        const ageMatch = text.match(/Tuổi\s*(\d+)/i);
        if (ageMatch) {
            learnedAge = parseInt(ageMatch[1], 10);
            if (currentAge - learnedAge >= 30 && status !== 'Đã Đính Chính') {
                status = 'Nguy Cơ Lỗi Thời';
                statusTag = '🔴 [Nguy Cơ Lỗi Thời]';
                statusColor = 'text-red-400 bg-red-950/40 border-red-500/40';
            }
        }

        const timeMatch = text.match(/\[(.*?)\]/g);
        if (timeMatch && timeMatch.length >= 2) {
            timeTag = timeMatch[1] || '';
            if (timeMatch[2]) chapterTag = timeMatch[2];
            content = text.replace(/\[.*?\]\s*/g, '').trim();
        }

        return {
            id: idx,
            raw: line,
            status,
            statusTag,
            statusColor,
            timeTag: timeTag.replace(/[\[\]]/g, ''),
            chapterTag: chapterTag.replace(/[\[\]]/g, ''),
            learnedAge,
            content
        };
    });
};

const pruneKnowledgeIntelligent = (knowledgeStr, currentAge = 18, maxChars = 50000) => {
    if (!knowledgeStr || knowledgeStr.length <= maxChars) return knowledgeStr;

    let lines = knowledgeStr.split('\n').filter(l => l.trim().length > 0);
    
    // Ưu tiên 1: Tỉa bỏ các dòng mẩu tin dạng "Tin Đồn" hoặc "Nguy Cơ Lỗi Thời" cũ nhất từ trên xuống
    let prunedLines = [];
    for (let i = 0; i < lines.length; i++) {
        const line = lines[i];
        const ageMatch = line.match(/Tuổi\s*(\d+)/i);
        const lAge = ageMatch ? parseInt(ageMatch[1], 10) : 0;
        const isOutdatedOrRumor = line.includes('Tin Đồn') || line.includes('Nguy Cơ Lỗi Thời') || (currentAge - lAge >= 30 && !line.includes('Đã Đính Chính'));

        if (isOutdatedOrRumor && knowledgeStr.length > maxChars) {
            knowledgeStr = lines.slice(i + 1).join('\n');
            continue;
        }
        prunedLines.push(line);
    }

    // Ưu tiên 2: Nếu vẫn đầy quá 50.000 ký tự, bỏ các dòng mẩu tin cũ nhất ở đầu, luôn giữ nguyên vẹn 100% từng dòng
    while (prunedLines.join('\n').length > maxChars && prunedLines.length > 1) {
        prunedLines.shift();
    }

    return prunedLines.join('\n');
};
