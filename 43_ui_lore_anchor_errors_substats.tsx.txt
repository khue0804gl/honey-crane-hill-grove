                                    if (!cleanTitle) return showAlert("Vui lòng nhập tiêu đề sự kiện!");
                                    if (!cleanDesc) return showAlert("Vui lòng nhập nội dung chi tiết sự kiện!");

                                    const currentChap = gameState.chapterCount || 0;
                                    const newEvent = {
                                        id: 'ev_' + Date.now(),
                                        title: cleanTitle,
                                        location: cleanLoc,
                                        desc: cleanDesc,
                                        startChapter: currentChap,
                                        expireChapter: currentChap + Number(eventDuration || 3)
                                    };

                                    setGameState((prev: any) => ({
                                        ...prev,
                                        timedEvents: [...(prev.timedEvents || []), newEvent]
                                    }));

                                    setStoryLog((prev: any) => {
                                        const newLog = [...prev, {
                                            type: 'system',
                                            text: `[Mỏ Neo Sự Kiện] Đã thiết lập sự kiện [${cleanTitle}] tại [${cleanLoc}] (Thời hạn ${eventDuration} chương).`,

                                        }];
                                        return newLog.length > 50 ? newLog.slice(-50) : newLog;
                                    });

                                    showAlert(`Đã tạo thành công sự kiện "${cleanTitle}"!`);
                                    setEventTitle('');
                                    setEventLocation('');
                                    setEventDesc('');

          </div>

                   <button onClick={() => { showConfirm('Chắc chắn xóa toàn bộ báo cáo lỗi?', () => setErrorLogs([])); }} className="text-xs bg-red-900/20 text-red-400 hover:bg-red-500 hover:text-white border border-red-500/30 px-3 py-1.5 rounded transition-colors shadow-sm flex items-center gap-1">
                       <Trash2 size={14}/> Xóa Cả
                   </button>
                   <button onClick={() => setIsErrorLogModalOpen(false)} className="p-2 hover:bg-red-900/30 hover:text-red-500 border border-transparent hover:border-red-900/50 rounded-lg text-[var(--text-muted)] transition-colors"><X size={20}/></button>
               </div>
            </div>
            
            <div className="flex-1 overflow-y-auto p-4 md:p-6 custom-scrollbar bg-[var(--bg-body)] space-y-4">
                {errorLogs.length === 0 ? (
                    <div className="h-full flex flex-col items-center justify-center text-emerald-500/50 py-10">
                        <Check size={60} className="mb-4 opacity-50"/>
                        <p className="text-lg italic font-bold text-emerald-500">Hệ thống đang hoạt động trơn tru. Chưa ghi nhận lỗi nào!</p>
                    </div>
                ) : (
                    errorLogs.map((log, i) => (
                        <div key={i} className="bg-[var(--bg-body)] rounded-xl border border-red-900/30 hover:border-red-500/50 transition-colors p-4 shadow-sm flex flex-col gap-2">
                            <div className="flex justify-between items-start border-b border-red-900/30 pb-2 mb-1">
                                <div className="font-bold text-red-400 text-sm flex items-center gap-2">
                                    <AlertTriangle size={16} className="shrink-0"/> <span className="break-all">{log.message}</span>
                                </div>
                                <span className="text-[10px] font-mono text-gray-500 shrink-0 ml-4">{log.time}</span>
                            </div>
                            {log.source && (
                                <div className="text-[11px] text-gray-400 font-mono flex gap-2">
                                    <span className="text-blue-400">Tệp:</span> <span className="break-all">{log.source}</span>

                                </div>

                                <div className="bg-[var(--bg-body)] p-3 rounded-lg border border-red-900/50 text-[10px] md:text-[11px] text-gray-400 font-mono whitespace-pre-wrap overflow-x-auto max-h-60 custom-scrollbar">

                                </div>

                 <div className="flex items-center gap-3">

                  <p className="text-[11px] text-[var(--text-muted)] leading-relaxed italic mb-3 bg-[var(--bg-surface)] p-3 rounded-lg border border-[var(--border-main)] shadow-inner">💡 Đạo hữu có thể tùy chọn ghim các chỉ số nghề nghiệp/đời sống ra ngoài màn hình chính để dễ bề theo dõi. Thiên Đạo sẽ luôn nội suy các thông số ẩn này để làm cơ sở logic.</p>
